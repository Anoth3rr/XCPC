### 多项式封装 Poly

```c++
template <int G = 3> struct Poly {
    static void ntt(vector<Z> &a, bool iv) {
        int n = a.size();
        assert(n && !(n & (n - 1)));
        vector<int> rv(n);
        int k = __builtin_ctz(static_cast<unsigned>(n));
        if (n > 1)
            for (int i = 0; i < n; ++i) rv[i] = (rv[i >> 1] >> 1) | ((i & 1) << (k - 1));
        for (int i = 0; i < n; ++i)
            if (i < rv[i]) swap(a[i], a[rv[i]]);
        for (int len = 2; len <= n; len <<= 1) {
            Z wn = power(Z(G), (Z::getMod() - 1) / len);
            if (iv) wn = wn.inv();
            for (int i = 0; i < n; i += len) {
                Z w = 1;
                for (int j = 0; j < len / 2; ++j) {
                    Z x = a[i + j], y = a[i + j + len / 2] * w;
                    a[i + j] = x + y;
                    a[i + j + len / 2] = x - y;
                    w *= wn;
                }
            }
        }
        if (iv) {
            Z x = Z(n).inv();
            for (Z &v : a) v *= x;
        }
    }

    static vector<Z> add(vector<Z> a, vector<Z> b) {
        int n = max(a.size(), b.size());
        a.resize(n), b.resize(n);
        for (int i = 0; i < n; ++i) a[i] += b[i];
        return a;
    }

    static vector<Z> sub(vector<Z> a, vector<Z> b) {
        int n = max(a.size(), b.size());
        a.resize(n), b.resize(n);
        for (int i = 0; i < n; ++i) a[i] -= b[i];
        return a;
    }

    static vector<Z> mul(vector<Z> a, vector<Z> b) {
        if (a.empty() || b.empty()) return {};
        int m = a.size() + b.size() - 1, n = 1;
        while (n < m) n <<= 1;
        a.resize(n), b.resize(n);
        ntt(a, false), ntt(b, false);
        for (int i = 0; i < n; ++i) a[i] *= b[i];
        ntt(a, true);
        a.resize(m);
        return a;
    }

    static vector<Z> inv(const vector<Z> &a, int n) {
        assert(n >= 0);
        if (!n) return {};
        assert(!a.empty() && a[0] != Z(0));
        if (n == 1) return {a[0].inv()};
        vector<Z> b = inv(a, (n + 1) >> 1);
        int m = 1;
        while (m < n << 1) m <<= 1;
        vector<Z> c(a.begin(), a.begin() + min<int>(a.size(), n));
        c.resize(m), b.resize(m);
        ntt(c, false), ntt(b, false);
        for (int i = 0; i < m; ++i) b[i] *= Z(2) - c[i] * b[i];
        ntt(b, true);
        b.resize(n);
        return b;
    }

    static vector<Z> div(const vector<Z> &a, const vector<Z> &b) {
        assert(!b.empty() && b.back() != Z(0));
        int n = a.size(), m = b.size();
        if (n < m) return {Z(0)};
        vector<Z> x(a.rbegin(), a.rend()), y(b.rbegin(), b.rend());
        y = inv(y, n - m + 1);
        vector<Z> q = mul(x, y);
        q.resize(n - m + 1);
        reverse(q.begin(), q.end());
        return q;
    }

    static vector<Z> mod(const vector<Z> &a, const vector<Z> &b) {
        vector<Z> r = sub(a, mul(div(a, b), b));
        r.resize(min<int>(r.size(), b.size() - 1));
        return r;
    }

    static vector<Z> der(const vector<Z> &a) {
        int n = a.size();
        if (n <= 1) return {};
        vector<Z> b(n - 1);
        for (int i = 1; i < n; ++i) b[i - 1] = a[i] * Z(i);
        return b;
    }

    static vector<Z> integ(const vector<Z> &a) {
        int n = a.size();
        vector<Z> b(n + 1);
        for (int i = 0; i < n; ++i) b[i + 1] = a[i] / Z(i + 1);
        return b;
    }

    static vector<Z> ln(const vector<Z> &a, int n) {
        assert(n >= 0);
        if (!n) return {};
        assert(!a.empty() && a[0] != Z(0));
        vector<Z> b = integ(mul(der(a), inv(a, n)));
        b.resize(n);
        return b;
    }

    static vector<Z> exp(const vector<Z> &a, int n) {
        assert(n >= 0);
        if (!n) return {};
        assert(a.empty() || a[0] == Z(0));
        vector<Z> b{Z(1)};
        for (int k = 2; k < n << 1; k <<= 1) {
            int m = min(k, n);
            vector<Z> c(a.begin(), a.begin() + min<int>(a.size(), m));
            c.resize(m);
            c = sub(c, ln(b, m));
            c[0] += Z(1);
            b = mul(b, c);
            b.resize(m);
        }
        b.resize(n);
        return b;
    }

    static void build(vector<vector<Z>> &t, const vector<Z> &x, int u, int l, int r) {
        if (r - l == 1) {
            t[u] = {-x[l], Z(1)};
            return;
        }
        int m = (l + r) >> 1;
        build(t, x, u << 1, l, m);
        build(t, x, u << 1 | 1, m, r);
        t[u] = mul(t[u << 1], t[u << 1 | 1]);
    }

    static void eval0(const vector<Z> &f, const vector<vector<Z>> &t, vector<Z> &y, int u, int l, int r) {
        if (r - l == 1) {
            if (f.empty())
                y[l] = Z(0);
            else if (f.size() == 1)
                y[l] = f[0];
            else
                y[l] = mod(f, t[u])[0];
            return;
        }
        vector<Z> g = mod(f, t[u]);
        int m = (l + r) >> 1;
        eval0(g, t, y, u << 1, l, m);
        eval0(g, t, y, u << 1 | 1, m, r);
    }

    static vector<Z> eval(const vector<Z> &f, const vector<Z> &x) {
        int n = x.size();
        if (!n) return {};
        vector<vector<Z>> t(n << 2 | 1);
        build(t, x, 1, 0, n);
        vector<Z> y(n);
        eval0(f, t, y, 1, 0, n);
        return y;
    }

    static vector<Z> interp(const vector<Z> &x, const vector<Z> &y) {
        int n = x.size();
        assert(y.size() == n);
        if (!n) return {};
        vector<vector<Z>> t(n << 2 | 1);
        build(t, x, 1, 0, n);
        vector<Z> d = eval(der(t[1]), x), w(n);
        for (int i = 0; i < n; ++i) w[i] = y[i] / d[i];
        function<vector<Z>(int, int, int)> dfs = [&](int u, int l, int r) -> vector<Z> {
            if (r - l == 1) return {w[l]};
            int m = (l + r) >> 1;
            return add(mul(dfs(u << 1, l, m), t[u << 1 | 1]), mul(dfs(u << 1 | 1, m, r), t[u << 1]));
        };
        return dfs(1, 0, n);
    }

    static vector<Z> shift(vector<Z> a, int k) {
        int n = a.size();
        if (k >= 0) a.insert(a.begin(), k, Z(0));
        else if (k <= -n) return {};
        else a.erase(a.begin(), a.begin() - k);
        return a;
    }
    static vector<Z> trunc(vector<Z> a, int n) {
        assert(n >= 0);
        a.resize(n);
        return a;
    }
    static vector<Z> mulT(const vector<Z> &a, vector<Z> b) {
        if (a.empty() || b.empty()) return {};
        int n = b.size();
        reverse(b.begin(), b.end());
        return shift(mul(a, b), 1 - n);
    }
    static vector<Z> pow(const vector<Z> &a, int k, int n) {
        assert(k >= 0 && n >= 0);
        vector<Z> r(n);
        if (!n) return r;
        if (!k) { r[0] = 1; return r; }
        int i = 0;
        while (i < a.size() && a[i] == Z(0)) ++i;
        if (i == a.size() || i128(i) * k >= n) return r;
        int d = i * k, m = n - d;
        Z v = a[i];
        vector<Z> b(a.begin() + i, a.end());
        b.resize(m);
        Z iv = v.inv();
        for (auto &x : b) x *= iv;
        b = ln(b, m);
        for (auto &x : b) x *= Z(k);
        b = exp(b, m);
        v = power(v, k);
        for (int j = 0; j < m; ++j) r[j + d] = b[j] * v;
        return r;
    }
    static vector<Z> sqrt(const vector<Z> &a, int n) {
        assert(n >= 0);
        if (!n) return {};
        assert(!a.empty() && a[0] == Z(1));
        vector<Z> b{Z(1)};
        Z half = Z(2).inv();
        for (int k = 2; k < n * 2; k *= 2) {
            int m = min(k, n);
            auto c = trunc(mul(trunc(a, m), inv(b, m)), m);
            b.resize(m);
            for (int i = 0; i < m; i++) b[i] = (b[i] + c[i]) * half;
        }
        return trunc(b, n);
    }
};
```

### 离散傅里叶变换 dft 与其逆变换 idft

$P$ 需要为奇素数。

```cpp
vector<int> rev;
template<int P> vector<MInt<P>> roots{0, 1};

template<int P> constexpr MInt<P> findRoot() {
    MInt<P> i = 2;
    int k = __builtin_ctzll(P - 1);
    while (true) {
        if (power(i, (P - 1) / 2) != 1) {
            break;
        }
        i += 1;
    }
    return power(i, (P - 1) >> k);
}

template<int P> constexpr MInt<P> Root = findRoot<P>();
template<> constexpr MInt<998244353> Root<998244353>{31};

template<int P> constexpr void dft(vector<MInt<P>> &a) {
    int n = a.size();

    if (int(rev.size()) != n) {
        int k = __builtin_ctz(n) - 1;
        rev.resize(n);
        for (int i = 0; i < n; i++) {
            rev[i] = rev[i >> 1] >> 1 | (i & 1) << k;
        }
    }

    for (int i = 0; i < n; i++) {
        if (rev[i] < i) {
            swap(a[i], a[rev[i]]);
        }
    }
    if (roots<P>.size() < n) {
        int k = __builtin_ctz(roots<P>.size());
        roots<P>.resize(n);
        while ((1 << k) < n) {
            auto e = power(Root<P>, 1 << (__builtin_ctz(P - 1) - k - 1));
            for (int i = 1 << (k - 1); i < (1 << k); i++) {
                roots<P>[2 * i] = roots<P>[i];
                roots<P>[2 * i + 1] = roots<P>[i] * e;
            }
            k++;
        }
    }
    for (int k = 1; k < n; k *= 2) {
        for (int i = 0; i < n; i += 2 * k) {
            for (int j = 0; j < k; j++) {
                MInt<P> u = a[i + j];
                MInt<P> v = a[i + j + k] * roots<P>[k + j];
                a[i + j] = u + v;
                a[i + j + k] = u - v;
            }
        }
    }
}
template<int P> constexpr void idft(vector<MInt<P>> &a) {
    int n = a.size();
    reverse(a.begin() + 1, a.end());
    dft(a);
    MInt<P> inv = (1 - P) / n;
    for (int i = 0; i < n; i++) {
        a[i] *= inv;
    }
}
```

### 快速傅里叶变换 FFT

$\mathcal O(N\log N)$ 。

```c++
struct FFT {
    using cd = complex<double>;

    static void fft(vector<cd> &a, bool iv) {
        int n = a.size();
        assert(n && !(n & (n - 1)));
        if (n == 1) return;

        static vector<int> rev;
        static vector<cd> wn;
        if (rev.size() != n) {
            rev.assign(n, 0);
            for (int i = 1; i < n; i++)
                rev[i] = (rev[i >> 1] >> 1) | ((i & 1) ? n >> 1 : 0);

            wn.resize(n / 2);
            const double PI = acos(-1.0);
            for (int i = 0; i < n / 2; i++)
                wn[i] = polar(1.0, 2 * PI * i / n);
        }

        for (int i = 0; i < n; i++)
            if (i < rev[i]) swap(a[i], a[rev[i]]);

        for (int len = 2; len <= n; len <<= 1) {
            int step = n / len;
            for (int i = 0; i < n; i += len) {
                for (int j = 0; j < len / 2; j++) {
                    cd w = wn[j * step];
                    if (iv) w = conj(w);
                    cd x = a[i + j], y = a[i + j + len / 2] * w;
                    a[i + j] = x + y;
                    a[i + j + len / 2] = x - y;
                }
            }
        }

        if (iv)
            for (cd &x : a) x /= n;
    }
};
```

### 任意模卷积 MTT

```cpp
struct MTT {
    using Z = MInt<0>;

  private:
    using Z0 = MInt<1>;
    using Z1 = MInt<2>;
    using Z2 = MInt<3>;
    static constexpr int p0 = 1004535809;
    static constexpr int p1 = 998244353;
    static constexpr int p2 = 985661441;

    template <class T> static void ntt(vector<T> &a, bool iv) {
        int n = a.size();
        assert(n && !(n & (n - 1)) && (T::getMod() - 1) % n == 0);

        for (int i = 1, j = 0; i < n; i++) {
            int k = n >> 1;
            for (; j & k; k >>= 1) j ^= k;
            j ^= k;
            if (i < j) swap(a[i], a[j]);
        }

        for (int len = 2; len <= n; len <<= 1) {
            T wn = T(3).pow((T::getMod() - 1) / len);
            if (iv) wn = wn.inv();
            for (int i = 0; i < n; i += len) {
                T w = 1;
                for (int j = 0; j < len / 2; j++) {
                    T x = a[i + j], y = a[i + j + len / 2] * w;
                    a[i + j] = x + y;
                    a[i + j + len / 2] = x - y;
                    w *= wn;
                }
            }
        }

        if (iv) {
            T invn = T(n).inv();
            for (T &x : a) x *= invn;
        }
    }

    template <class T> static vector<T> mul0(const vector<Z> &a, const vector<Z> &b) {
        int m = a.size() + b.size() - 1, n = 1;
        while (n < m) n <<= 1;

        vector<T> x(n), y(n);
        for (int i = 0; i < a.size(); i++) x[i] = T(a[i].val());
        for (int i = 0; i < b.size(); i++) y[i] = T(b[i].val());

        bool same = x == y;
        ntt(x, false);
        if (same) {
            for (T &v : x) v *= v;
        } else {
            ntt(y, false);
            for (int i = 0; i < n; i++) x[i] *= y[i];
        }
        ntt(x, true);
        x.resize(m);
        return x;
    }

  public:
    static vector<Z> mul(vector<Z> a, vector<Z> b) {
        int n = a.size(), m = b.size();
        if (!n || !m) return {};
        assert(Z::getMod() > 1);

        if (min(n, m) <= 64) {
            vector<Z> c(n + m - 1);
            for (int i = 0; i < n; i++)
                for (int j = 0; j < m; j++) c[i + j] += a[i] * b[j];
            return c;
        }

        int cnt = 1;
        while (cnt < n + m - 1) cnt <<= 1;
        assert(cnt <= (1LL << 21));

        Z0::setMod(p0);
        Z1::setMod(p1);
        Z2::setMod(p2);

        auto c0 = mul0<Z0>(a, b);
        auto c1 = mul0<Z1>(a, b);
        auto c2 = mul0<Z2>(a, b);

        static const Z1 iv01 = Z1(p0).inv();
        static const Z2 iv02 = Z2(p0).inv();
        static const Z2 iv12 = Z2(p1).inv();

        Z w1 = Z(p0), w2 = w1 * Z(p1);
        vector<Z> c(n + m - 1);
        for (int i = 0; i < n + m - 1; i++) {
            Z1 x1 = (c1[i] - Z1(c0[i].val())) * iv01;
            Z2 x2 = ((c2[i] - Z2(c0[i].val())) * iv02 - Z2(x1.val())) * iv12;

            c[i] = Z(c0[i].val()) + Z(x1.val()) * w1 + Z(x2.val()) * w2;
        }
        return c;
    }
};
```

### FWT：OR、AND、XOR 卷积

计算下标按按位 OR、AND 或 XOR 合并的卷积。两个卷积输入必须落在同一个下标全集中，重载会将二者补到相同长度；XOR 逆变换要求 2 在模数下可逆。时间复杂度 $\mathcal O(N\log N)$。

$$\operatorname{FWT}(a)_i=\sum_{i\mid j=i}a_j.\\
\operatorname{FWT}(a)_i=\sum_{i\mathbin{\&}j=i}a_j.\\
\operatorname{FWT}(a)_i
=\sum_{i\circ j=0}a_j-\sum_{i\circ j=1}a_j,
\qquad
a\circ b=\operatorname{popcount}(a\mathbin{\&}b)\mathbin{\&}1.$$

```cpp
struct FWT {
    static void fwt(vector<Z> &a, char op, bool iv = false) {
        int n = a.size();
        assert(n && !(n & (n - 1)));
        assert(op == '|' || op == '&' || op == '^');
        for (int k = 1; k < n; k <<= 1) {
            for (int i = 0; i < n; i += k << 1) {
                for (int j = 0; j < k; j++) {
                    Z x = a[i + j], y = a[i + j + k];
                    if (op == '|') {
                        a[i + j + k] = iv ? y - x : y + x;
                    } else if (op == '&') {
                        a[i + j] = iv ? x - y : x + y;
                    } else {
                        a[i + j] = x + y;
                        a[i + j + k] = x - y;
                    }
                }
            }
        }
        if (op == '^' && iv) {
            Z invn = Z(n).inv();
            for (Z &x : a) x *= invn;
        }
    }

    static vector<Z> mul(vector<Z> a, vector<Z> b, char op) {
        if (a.empty() || b.empty()) return {};
        int n = 1;
        while (n < max(a.size(), b.size())) n <<= 1;
        a.resize(n), b.resize(n);
        fwt(a, op), fwt(b, op);
        for (int i = 0; i < n; i++) a[i] *= b[i];
        fwt(a, op, true);
        return a;
    }
};

vector<Z> operator|(vector<Z> a, vector<Z> b) {
    return FWT::mul(move(a), move(b), '|');
}
vector<Z> operator&(vector<Z> a, vector<Z> b) {
    return FWT::mul(move(a), move(b), '&');
}
vector<Z> operator^(vector<Z> a, vector<Z> b) {
    return FWT::mul(move(a), move(b), '^');
}
```

### 连续点拉格朗日插值

```cpp
Z Lagrange(const vector<Z> &f, Z x) {
    int n = f.size() - 1;
    vector<Z> pre(n + 2, 1);
    for (int i = 0; i <= n; i++) pre[i + 1] = pre[i] * (x - Z(i));
    Z ans = 0, suf = 1;
    for (int i = n; i >= 0; i--) {
        Z v = f[i] * pre[i] * suf * C.ifac(i) * C.ifac(n - i);
        if ((n - i) & 1)
            ans = ans - v;
        else
            ans = ans + v;

        suf = suf * (x - Z(i));
    }
    return ans;
}
```

### 任意点拉格朗日插值

$n + 1$ 个点可以唯一确定一个最高为 $n$ 次的多项式。$$\displaystyle f(k) = \sum_{i = 1}^{n + 1} y_i \prod_{i \neq j} \frac{x - x[j]}{x[i] - x[j]}$$

```c++
Z Lagrange(const vector<array<Z, 2>> &f, Z x) {
    int n = f.size();
    Z ans = 0;
    for (int i = 0; i < n; i++) {
        Z a = f[i][1], b = 1;
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            a *= x - f[j][0];
            b *= f[i][0] - f[j][0];
        }
        ans += a / b;
    }
    return ans;
}
```

### 最短线性递推 Berlekamp-Massey

求解数列的最短线性递推式，最坏复杂度为 $\mathcal O(NM)$，其中 $N$ 为数列长度，$M$ 为它的最短递推式的阶数。

```c++
template <class Z> vector<Z> BM(const vector<Z> &a) {
    vector<Z> c{Z(1)}, b{Z(1)};
    int l = 0, m = 1;
    Z d = 1;
    for (int i = 0; i < a.size(); ++i) {
        Z x = a[i];
        for (int j = 1; j <= l; ++j) x += c[j] * a[i - j];
        if (x == Z(0)) {
            ++m;
            continue;
        }
        vector<Z> t = c;
        Z q = x / d;
        if (c.size() < b.size() + m) c.resize(b.size() + m);
        for (int j = 0; j < b.size(); ++j) c[j + m] -= q * b[j];
        if (l * 2 <= i) {
            l = i + 1 - l;
            b = move(t);
            d = x;
            m = 1;
        } else {
            ++m;
        }
    }
    vector<Z> r(l);
    for (int i = 0; i < l; ++i) r[i] = -c[i + 1];
    return r;
}
```

### 常系数线性递推第 n 项 Linear-Recurrence

```c++
template<int P = 998244353> MInt<P> linearRecurrence(Poly<P> p, Poly<P> q, i64 n) {
    int m = q.size() - 1;
    while (n > 0) {
        auto newq = q;
        for (int i = 1; i <= m; i += 2) {
            newq[i] *= -1;
        }
        auto newp = p * newq;
        newq = q * newq;
        for (int i = 0; i < m; i++) {
            p[i] = newp[i * 2 + n % 2];
        }
        for (int i = 0; i <= m; i++) {
            q[i] = newq[i * 2];
        }
        n /= 2;
    }
    return p[0] / q[0];
}
```

### 分式第 n 项 Bostan–Mori

求 $[x^n]P(x)/Q(x)$，要求 $n\ge 0$、$Q(0)\ne 0$，允许分子次数不小于分母，也允许末尾零系数；依赖本章完整封装 `NTT::mul`。
每轮乘以 $Q(-x)$ 后按奇偶抽取系数，时间 $O(M(d)\log(n+2))$，其中 $d$ 为输入最大长度、$M(d)$ 为卷积复杂度。[参考](https://cirnonine9.github.io/algorithm-competition-template/模板/05-多项式/19-分式第k项.html)。

```cpp
Z bostanMori(vector<Z> p, vector<Z> q, i64 n) {
    assert(n >= 0 && !q.empty() && q[0] != Z(0));
    while (n && !p.empty()) {
        auto h = q;
        for (int i = 1; i < h.size(); i += 2) h[i] = -h[i];
        p = NTT::mul(p, h), q = NTT::mul(q, h);
        vector<Z> a, b;
        for (int i = n & 1; i < p.size(); i += 2) a.push_back(p[i]);
        for (int i = 0; i < q.size(); i += 2) b.push_back(q[i]);
        p = move(a), q = move(b);
        n >>= 1;
    }
    return p.empty() ? Z(0) : p[0] / q[0];
}
```

若 $a_n=\sum_{j=1}^k c_j a_{n-j}$，令 $Q(x)=1-\sum_{j=1}^k c_jx^j$、$P(x)=(\sum_{i=0}^{k-1}a_ix^i)Q(x)\bmod x^k$，则答案为 `bostanMori(p, q, n)`；本章 `BM` 返回的是 $c_1,\ldots,c_k$，构造 $Q$ 时要取负号并补常数项 $1$。

### 多点求值与快速插值

`eval(f, x)` 求各点 $f(x_i)$，`interp(x, y)` 返回通过这些点、次数小于点数的多项式系数，均复用本章完整封装 `NTT`；系数按低次到高次存储。
求值点允许重复，插值要求 `x`、`y` 等长且横坐标在模数下两两不同；系数数和点数均为 $O(n)$ 时，时间 $O(n\log^2 n)$，乘积树空间 $O(n\log n)$。

```cpp
auto y = NTT::eval(f, x);
auto g = NTT::interp(x, y);
```

空点集返回空数组；插值结果长度等于点数，可能带末尾零系数。由点值恢复原多项式时，需要点数严格大于原多项式次数。

### 多项式大指数幂

求 $f(x)^k\bmod x^n$，指数 `k` 为非负十进制字符串，允许前导零，约定 $f^0=1$；依赖本章完整封装 `NTT`，模数 $p$ 为 NTT 友好质数且 $n<p$，卷积长度须在模数支持范围内。
分别维护指数模 $p$、模 $p-1$ 及截断到 $n$ 的值，用于对数、常数项幂和最低次数偏移，时间 $O(|k|+n\log n)$。[参考](https://cirnonine9.github.io/algorithm-competition-template/模板/05-多项式/16-多项式快速幂.html)。

```cpp
vector<Z> polyPow(vector<Z> f, const string &k, int n) {
    assert(n >= 0 && !k.empty());
    if (!n) return {};
    int p = Z::getMod(), a = 0, b = 0, e = 0;
    assert(n < p);
    for (char c : k) {
        assert('0' <= c && c <= '9');
        int d = c - '0';
        a = (a * 10 + d) % p;
        b = (b * 10 + d) % (p - 1);
        e = min(n, e * 10 + d);
    }
    vector<Z> r(n);
    if (!e) { r[0] = 1; return r; }
    f.resize(n);
    int t = 0;
    while (t < n && f[t] == Z(0)) t++;
    if (t == n || i128(t) * e >= n) return r;

    int d = t * e, m = n - d;
    Z c = f[t], iv = c.inv();
    vector<Z> g(f.begin() + t, f.end());
    g.resize(m);
    for (Z &x : g) x *= iv;
    g = NTT::ln(g, m);
    for (Z &x : g) x *= Z(a);
    g = NTT::exp(g, m);
    c = power(c, b);
    for (int i = 0; i < m; i++) r[i + d] = g[i] * c;
    return r;
}
```

### 稀疏多项式除法

求 $f(x)/g(x)\bmod x^n$，要求 $g(0)\ne 0$；只依赖取模类 `Z`，输入不足 $n$ 项时补零。
适合分母非零项很少的情况，按次数递增递推，设截断后分母有 $s$ 个非零项，时间 $O(ns)$。[参考](https://cirnonine9.github.io/algorithm-competition-template/模板/05-多项式/11-稀疏多项式除法.html)。

```cpp
vector<Z> divSparse(vector<Z> f, const vector<Z> &g, int n) {
    assert(n >= 0);
    if (!n) return {};
    assert(!g.empty() && g[0] != Z(0));
    f.resize(n);
    vector<pair<int, Z>> a;
    for (int i = 1; i < g.size() && i < n; i++)
        if (g[i] != Z(0)) a.push_back({i, g[i]});
    Z iv = g[0].inv();
    for (int i = 0; i < n; i++) {
        for (auto [j, x] : a) {
            if (j > i) break;
            f[i] -= x * f[i - j];
        }
        f[i] *= iv;
    }
    return f;
}
```

### 稀疏多项式幂

求稀疏多项式 $f(x)$ 的非负整数次幂 $f(x)^k\bmod x^n$，允许前导零和任意非零首项，约定 $f^0=1$；依赖 `Z`、`power` 和组合数对象 `C`，要求质数模数下 $n<p$。
提取最低非零项并归一化后，由 $fh'=kf'h$ 递推，设截断后的非零项数为 $s$，时间 $O(n(s+1)+\log k)$，另需逆元预处理。[参考](https://cirnonine9.github.io/algorithm-competition-template/模板/05-多项式/16-多项式快速幂.html)。

```cpp
vector<Z> powSparse(vector<Z> f, int k, int n) {
    assert(k >= 0 && n >= 0 && n < Z::getMod());
    vector<Z> r(n);
    if (!n) return r;
    if (!k) { r[0] = 1; return r; }
    f.resize(n);
    int t = 0;
    while (t < n && f[t] == Z(0)) t++;
    if (t == n || i128(t) * k >= n) return r;

    int d = t * k, m = n - d;
    Z iv = f[t].inv(), e = Z(k) + Z(1);
    vector<pair<int, Z>> a;
    for (int i = 1; i < m && t + i < n; i++)
        if (f[t + i] != Z(0)) a.push_back({i, f[t + i] * iv});
    C.init(m - 1);
    r[d] = power(f[t], k);
    for (int i = 1; i < m; i++) {
        for (auto [j, x] : a) {
            if (j > i) break;
            r[d + i] += (e * Z(j) - Z(i)) * x * r[d + i - j];
        }
        r[d + i] *= C.inv(i);
    }
    return r;
}
```

### 连续点值平移

已知次数小于 $n$ 的多项式在 $0,\ldots,n-1$ 的点值 `f`，返回在 $c,\ldots,c+n-1$ 的点值，时间 $O(n\log n)$；依赖完整封装 `NTT` 和组合数对象 `C`，要求 $n<p$ 且卷积长度合法。
横坐标均按模 $p$ 理解，支持与原区间重叠及跨模数回绕；命中已知节点时直接取值，其余分母通过批量求逆处理。[参考](https://cirnonine9.github.io/algorithm-competition-template/模板/05-多项式/21-点值平移.html)。

```cpp
vector<Z> shiftValues(const vector<Z> &f, Z c) {
    int n = f.size();
    if (!n) return {};
    assert(n < Z::getMod());
    C.init(n - 1);
    vector<Z> a(n), b(n * 2 - 1), pre(n * 2, 1);
    for (int i = 0; i < n; i++) {
        a[i] = f[i] * C.ifac(i) * C.ifac(n - 1 - i);
        if ((n - 1 - i) & 1) a[i] = -a[i];
    }
    for (int i = 0; i < n * 2 - 1; i++) {
        b[i] = c + Z(i - n + 1);
        if (b[i] == Z(0)) b[i] = 1;
        pre[i + 1] = pre[i] * b[i];
    }
    Z iv = pre.back().inv();
    for (int i = n * 2 - 2; i >= 0; i--) {
        Z x = b[i];
        b[i] = iv * pre[i];
        iv *= x;
    }
    auto h = NTT::mul(a, b);
    vector<Z> r(n);
    Z w = pre[n];
    for (int i = 0; i < n; i++) {
        int t = (c + Z(i)).val();
        r[i] = t < n ? f[t] : h[n - 1 + i] * w;
        if (i + 1 < n) {
            Z x = c + Z(i + 1);
            if (x == Z(0)) x = 1;
            w *= x * b[i];
        }
    }
    return r;
}
```

### 原根

求通常意义下的原根。

```cpp
int power(int a, int b, int m) {
    int r = 1 % m;
    a %= m;
    for (; b; b >>= 1, a = a * a % m) if (b & 1) r = r * a % m;
    return r;
}
int Root(int n) {
    auto fact = [&](int v) {
        vector<int> fac;
        for (int i = 2; i <= v / i; ++i) {
            if (v % i != 0) continue;
            fac.push_back(i);
            do {
                v /= i;
            } while(v % i == 0);
        }
        if (v > 1) fac.push_back(v);
        return fac;
    };
    int phi = n;
    for (auto v : fact(n)) phi = phi / v * (v - 1);
    auto f = fact(phi);
    auto check = [&](int g) {
        if (gcd(g, n) != 1) return false;
        for (auto v : f) if (power(g, phi / v, n) == 1) return false;
        return true;
    };
    for (int g = 1; g < n; ++g) {
        if (check(g)) return g;
    }
    return -1;
}
```

### 常用结论

**普通生成函数 OGF**

数列 $a_n$ 的普通生成函数为 $A(x)=\sum_{n\ge 0}a_nx^n$，代码中第 $n$ 项直接存 $a_n$，主要用于按大小计数和普通卷积。
下式按形式幂级数理解，无需考虑解析收敛；分式的分母常数项必须可逆。

| 操作 | 对应生成函数或系数 |
| --- | --- |
| 普通卷积 $c_n=\sum_{i=0}^n a_i b_{n-i}$ | $C(x)=A(x)B(x)$ |
| 前缀和 $s_n=\sum_{i=0}^n a_i$ | $S(x)=A(x)/(1-x)$ |
| 后移 $k$ 项，前面补零 | $x^kA(x)$ |
| 系数乘下标 $b_n=na_n$ | $B(x)=xA'(x)$ |
| 每项下标乘 $k$ | $A(x^k)$ |

常用展开（$k\ge 1$，$m,r\ge 0$）：

$$
\frac{1}{1-x^k}=\sum_{j\ge 0}x^{kj},\qquad
\frac{1-x^m}{1-x}=\sum_{j=0}^{m-1}x^j,\qquad
\frac{1}{(1-x)^{r+1}}=\sum_{n\ge 0}\binom{n+r}{r}x^n.
$$

- 对数展开：$\displaystyle-\ln(1-x^k)=\sum_{j\ge 1}\frac{x^{kj}}{j}$（$k\ge 1$），即把 $-\ln(1-x)$ 的非零项下标乘以 $k$；模意义下分母 $j$ 须可逆。
- 若 $F_n=F_{n-1}+F_{n-2}$（$n\ge 2$），则 $\displaystyle\sum_{n\ge 0}F_nx^n=\frac{F_0+(F_1-F_0)x}{1-x-x^2}$。
- 卡特兰数 $C_n=\frac{1}{n+1}\binom{2n}{n}$ 的 OGF 为 $\displaystyle C(x)=\frac{1-\sqrt{1-4x}}{2x}$，取平方根常数项为 $1$，有 $C=1+xC^2$。
- 整数分拆数的 OGF 为 $\displaystyle\prod_{j\ge 1}\frac{1}{1-x^j}$；只求前 $n+1$ 项时，乘积保留到 $j=n$ 即可。

**指数生成函数 EGF**

数列 $a_n$ 的指数生成函数为 $A(x)=\sum_{n\ge 0}a_nx^n/n!$，适合有标号对象计数；代码中存 $a_n\cdot\texttt{C.ifac(n)}$，取回计数时乘 $n!$。
因此同一数列从 OGF 系数转为 EGF 系数要除以 $n!$，模质数 $p$ 时所用阶乘下标须小于 $p$。

$$
A(x)B(x)=\sum_{n\ge 0}\left(\sum_{i=0}^n\binom ni a_i b_{n-i}\right)\frac{x^n}{n!}.
$$

若 $B(0)=0$ 表示非空构件，$\exp B(x)$ 表示构件的无序集合，$1/(1-B(x))$ 表示有序序列，$-\ln(1-B(x))$ 表示循环排列。

| 计数对象 | EGF |
| --- | --- |
| 每个大小恰有一种对象，$a_n=1$ | $e^x$ |
| 全排列，$a_n=n!$ | $1/(1-x)$ |
| 非空循环排列，$a_n=(n-1)!$ | $-\ln(1-x)$ |
| 错排 | $e^{-x}/(1-x)$ |
| 集合划分（贝尔数） | $\exp(e^x-1)$ |
| 恰分成 $k$ 个非空无序块 | $(e^x-1)^k/k!$ |

- 连通有标号图的 EGF 记为 $C(x)$，则所有简单无向有标号图的 EGF 为 $\displaystyle G(x)=\exp C(x)=\sum_{n\ge 0}2^{\binom n2}\frac{x^n}{n!}$，所以 $C=\ln G$。
- 有根有标号树的 EGF 满足 $T=xe^T$；无根有标号树的 EGF 为 $\displaystyle U(x)=x+\sum_{n\ge 2}n^{n-2}\frac{x^n}{n!}$，森林的 EGF 为 $\exp U(x)$。

错排只允许长度至少为 $2$ 的循环，其构件 EGF 为 $-\ln(1-x)-x$，取指数得到 $e^{-x}/(1-x)$。

**常用多项式恒等式**

以下系数均按低次到高次排列，模意义下出现的整数分母必须可逆；形式对数取常数项为 $0$，形式指数的输入常数项为 $0$。

- 广义二项式定理：$\displaystyle(1+x)^\alpha=\sum_{n\ge 0}\binom{\alpha}{n}x^n$，其中 $\displaystyle\binom{\alpha}{n}=\frac{\alpha(\alpha-1)\cdots(\alpha-n+1)}{n!}$，$\binom{\alpha}{0}=1$；$\alpha=m\ge 0$ 为整数时，得到有限展开 $\displaystyle(1+x)^m=\sum_{j=0}^m\binom mjx^j$。
- 若 $F(0)=G(0)=1$，则 $\ln(FG)=\ln F+\ln G$、$(\ln F)'=F'/F$；若 $H(0)=0$，则 $\exp(\ln F)=F$、$\ln(\exp H)=H$、$(\exp H)'=H'\exp H$。
- 系数平移：若 $F(x)=\sum_{j=0}^n a_jx^j$、$F(x+c)=\sum_{i=0}^n b_ix^i$，则 $\displaystyle b_i=\sum_{j=i}^n\binom ji a_jc^{j-i}$，即 $\displaystyle i!b_i=\sum_{j=i}^n(j!a_j)\frac{c^{j-i}}{(j-i)!}$，反转后做一次卷积；取 $c=1$ 即得 $\displaystyle b_i=\sum_{j=i}^n\binom ji a_j$。
- 二项式变换：若 $\displaystyle b_n=\sum_{i=0}^n\binom ni a_i$，其 OGF 满足 $\displaystyle B(x)=\frac{1}{1-x}A\!\left(\frac{x}{1-x}\right)$；逆变换为 $\displaystyle a_n=\sum_{i=0}^n(-1)^{n-i}\binom ni b_i$。在 EGF 下，相应变换为乘以 $e^x$ 和 $e^{-x}$。
- 乘积转指数：令 $s_k=\sum_{i=1}^m a_i^k$，则 $\displaystyle\prod_{i=1}^m(1-a_ix)^{-1}=\exp\!\left(\sum_{k\ge 1}\frac{s_k}{k}x^k\right)$。
- 五边形数定理：$\displaystyle\prod_{j\ge 1}(1-x^j)=1+\sum_{k\ge 1}(-1)^k\left(x^{k(3k-1)/2}+x^{k(3k+1)/2}\right)$。
- NTT 的本原 $n$ 次单位根：若 $G$ 为模质数 $p$ 的原根且 $n\mid p-1$，则 $\omega_n=G^{(p-1)/n}$。

**拉格朗日反演**

从隐式方程 $T(x)=x\Phi(T(x))$ 提取复合函数系数，要求 $\Phi(0)\ne 0$，取满足 $T(0)=0$ 的解。
下式中出现的整数分母须可逆，常数项单独取 $[x^0]H(T(x))=H(0)$。[参考](https://cirnonine9.github.io/algorithm-competition-template/模板/05-多项式/17-拉格朗日反演.html)。

对 $n\ge 1$：

$$
[x^n]H(T(x))=\frac{1}{n}[t^{n-1}]H'(t)\Phi(t)^n.
$$

常用特例：

$$
[x^n]T(x)^k=\frac{k}{n}[t^{n-k}]\Phi(t)^n\quad(n\ge k\ge 1),
$$

$$
[x^n]\Phi(T(x))^k=\frac{k}{n+k}[t^n]\Phi(t)^{n+k}\quad(n\ge 0,\ k\ge 1).
$$

若 $F(0)=0$、$F'(0)\ne 0$，取 $\Phi(t)=t/F(t)$ 即可求复合逆 $T=F^{\langle-1\rangle}$ 的系数；这里的复合逆满足 $F(T(x))=x$。
例如 $T=x(1+T)^2$ 时，对 $n\ge 1$ 有 $\displaystyle[x^n]T=\frac1n\binom{2n}{n-1}=\frac{1}{n+1}\binom{2n}{n}$。

**单位根反演**

用于筛选生成函数中下标满足 $n\equiv r\pmod m$ 的项；要求存在本原 $m$ 次单位根 $\omega$ 且 $m$ 可逆，模质数 $p$ 时可在 $m\mid p-1$ 的条件下取 $\omega=g^{(p-1)/m}$，其中 $g$ 为原根。
单位根筛选得到同余类的系数之和，只有多项式次数小于 $m$ 时，才会单独留下一个系数。[参考](https://cirnonine9.github.io/algorithm-competition-template/模板/05-多项式/18-单位根反演.html)。

$$
\frac1m\sum_{j=0}^{m-1}\omega^{j(n-r)}=[n\equiv r\pmod m].
$$

对 $A(x)=\sum_{n\ge 0}a_nx^n$、$0\le r<m$：

$$
\sum_{q\ge 0}a_{qm+r}x^{qm+r}
=\frac1m\sum_{j=0}^{m-1}\omega^{-rj}A(\omega^jx).
$$

若 $A$ 为有限多项式，可以令 $x=1$ 求同余类系数和；例如：

$$
\sum_{\substack{0\le k\le N\\k\equiv r\pmod m}}\binom Nk z^k
=\frac1m\sum_{j=0}^{m-1}\omega^{-rj}(1+\omega^jz)^N.
$$

取 $m=2$、$\omega=-1$，得到保留偶数次项与奇数次项的公式：

$$
A_{\mathrm{even}}(x)=\frac{A(x)+A(-x)}2,\qquad
A_{\mathrm{odd}}(x)=\frac{A(x)-A(-x)}2.
$$
