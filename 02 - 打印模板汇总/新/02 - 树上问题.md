## 第二章：树上问题

### 树的直径

用于无权树，`getlen` 返回 `{边数, 端点1, 端点2}` 。

```c++
struct Tree {
    int n;
    vector<vector<int>> g;
    Tree(int n) : n(n), g(n + 1) {}
    void add(int u, int v) {
        g[u].push_back(v);
        g[v].push_back(u);
    }
    array<int, 3> getlen(int root) const {
        vector<int> dep(n + 1, -1);
        auto dfs = [&](int s, int fa) {
            vector<pii> st = {{s, fa}};
            dep[s] = 0;
            int r = s;
            for (; st.size();) {
                auto [u, fa] = st.back();
                st.pop_back();
                if (dep[u] > dep[r]) r = u;
                for (auto v : g[u]) {
                    if (v == fa) continue;
                    dep[v] = dep[u] + 1;
                    st.push_back({v, u});
                }
            }
            return r;
        };
        int st = dfs(root, 0);
        int ed = dfs(st, 0);
        return {dep[ed], st, ed};
    }
};
```

### 树论大封装（直径+重心+中心）

```c++
struct Tree {
    int n;
    vector<vector<pii>> g;
    vector<int> dep, f, mxd, d1, d2, s1, s2, up;
    Tree(int n) : n(n), g(n + 1), dep(n + 1), f(n + 1), mxd(n + 1), d1(n + 1), d2(n + 1), s1(n + 1), s2(n + 1), up(n + 1) {}
    void add(int u, int v, int w) {
        g[u].push_back({v, w});
        g[v].push_back({u, w});
    }
    void dfs(int u, int fa) {
        mxd[u] = dep[u];
        for (auto [v, w] : g[u]) {
            if (v == fa) continue;
            dep[v] = dep[u] + 1;
            f[v] = u;
            dfs(v, u);
            mxd[u] = max(mxd[u], mxd[v]);
        }
    }

    void dfs1(int u, int fa) {
        for (auto [v, w] : g[u]) {
            if (v == fa) continue;
            dfs1(v, u);
            int x = d1[v] + w;
            if (x > d1[u]) {
                d2[u] = d1[u], s2[u] = s1[u];
                d1[u] = x, s1[u] = v;
            } else if (x > d2[u]) {
                d2[u] = x, s2[u] = v;
            }
        }
    }
    void dfs2(int u, int fa) {
        for (auto [v, w] : g[u]) {
            if (v == fa) continue;
            up[v] = (s1[u] == v ? max(up[u], d2[u]) : max(up[u], d1[u])) + w;
            dfs2(v, u);
        }
    }

    int rad, dia, cen;
    void getCenter() {
        cen = 1; // 中心
        for (int i = 1; i <= n; i++) {
            if (max(d1[i], up[i]) < max(d1[cen], up[cen])) cen = i;
        }
        rad = max(d1[cen], up[cen]); // 距离最远点的距离的最小值
        dia = 0;                     // 直径
        for (int u = 1; u <= n; ++u) {
            dia = max(dia, d1[u] + d2[u]);
        }
    }

    int rem, cog;
    vector<bool> vis;
    void getCog() {
        vis.assign(n + 1, 0);
        rem = inf;
        cog = 1;
        dfsCog(1);
    }
    int dfsCog(int u) {
        vis[u] = true;
        int siz = 1, res = 0;
        for (auto [v, w] : g[u]) {
            if (vis[v]) continue;
            int sub = dfsCog(v);
            res = max(res, sub);
            siz += sub;
        }
        res = max(res, n - siz);
        if (res < rem) rem = res, cog = u;
        return siz;
    }
};
```

### 树同构（AHU）

判断两棵无根、无标号树是否同构，`iso(g,h)` 传入两棵无权树的邻接表并返回布尔值。时间 $O(n\log n)$，空间 $O(n)$。

```c++
bool iso(const vector<vector<int>> &g, const vector<vector<int>> &h) {
    if (g.size() != h.size()) return false;
    int n = g.size();
    n--;
    if (n <= 1) return true;
    auto cen = [&](const auto &e) {
        vector<int> deg(n + 1), r;
        queue<int> q;
        for (int u = 1; u <= n; u++) {
            deg[u] = e[u].size();
            if (deg[u] == 1) q.push(u);
        }
        int rem = n;
        while (rem > 2) {
            int cnt = q.size();
            rem -= cnt;
            while (cnt--) {
                int u = q.front();
                q.pop();
                for (int v : e[u]) if (--deg[v] == 1) q.push(v);
            }
        }
        while (q.size()) {
            r.push_back(q.front());
            q.pop();
        }
        return r;
    };
    auto a = cen(g), b = cen(h);
    if (a.size() != b.size()) return false;
    map<vector<int>, int> mp;
    auto get = [&](const auto &e, int rt) {
        vector<int> fa(n + 1), id(n + 1), ord{rt};
        for (int i = 0; i < ord.size(); i++) {
            int u = ord[i];
            for (int v : e[u]) if (v != fa[u]) {
                fa[v] = u;
                ord.push_back(v);
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            int u = ord[i];
            vector<int> key;
            for (int v : e[u]) if (v != fa[u]) key.push_back(id[v]);
            sort(key.begin(), key.end());
            int x = mp.size() + 1;
            id[u] = mp.emplace(move(key), x).first->second;
        }
        return id[rt];
    };
    int x = get(g, a[0]);
    for (int rt : b) if (get(h, rt) == x) return true;
    return false;
}
```

### 最近公共祖先 LCA

#### 树上倍增解法

预处理时间复杂度 $\mathcal O(N\log N)$ ；单次查询 $\mathcal O(\log N)$ ，但是常数比树链剖分解法更大。

**封装一：基础封装，针对无权图。**

```c++
struct Tree {
    int n, lg, tim;
    vector<vector<int>> g, up;
    vector<int> dep, tin, tout;

    Tree(int n) : n(n), tim(0), g(n + 1), dep(n + 1), tin(n + 1), tout(n + 1) {
        lg = 1;
        while ((1LL << lg) <= n) ++lg;
        up.assign(n + 1, vector<int>(lg));
    }

    void add(int u, int v) {
        g[u].push_back(v);
        g[v].push_back(u);
    }

    void dfs(int u, int fa) {
        up[u][0] = fa;
        tin[u] = ++tim;

        for (auto v : g[u]) {
            if (v == fa) continue;
            dep[v] = dep[u] + 1;
            dfs(v, u);
        }

        tout[u] = tim;
    }

    void work(int root = 1) {
        tim = 0;
        dep[root] = 0;
        dfs(root, 0);

        for (int k = 1; k < lg; ++k) {
            for (int u = 1; u <= n; ++u) {
                up[u][k] = up[up[u][k - 1]][k - 1];
            }
        }
    }

    int jmp(int u, int k) const {
        if (k < 0 || k > dep[u]) return 0;
        for (int i = 0; i < lg; ++i) {
            if ((k >> i) & 1) u = up[u][i];
        }
        return u;
    }

    int lca(int x, int y) const {
        if (dep[x] < dep[y]) swap(x, y);
        x = jmp(x, dep[x] - dep[y]);
        if (x == y) return x;

        for (int k = lg - 1; k >= 0; --k) {
            if (up[x][k] == up[y][k]) continue;
            x = up[x][k], y = up[y][k];
        }
        return up[x][0];
    }

    int dis(int x, int y) const {
        int p = lca(x, y);
        return dep[x] + dep[y] - 2 * dep[p];
    }

    int kth(int x, int y, int k) const {
        int p = lca(x, y);
        int a = dep[x] - dep[p], b = dep[y] - dep[p];
        if (k < 0 || k > a + b) return 0;
        return k <= a ? jmp(x, k) : jmp(y, a + b - k);
    }

    bool anc(int x, int y) const {
        return tin[x] <= tin[y] && tout[y] <= tout[x];
    }
};
```

**封装二：扩展封装，针对有权图，支持“倍增查询两点路径上的最大边权”功能**。

```c++
constexpr int inf = 2e18 + 9;

struct Tree {
    int n, lg, tim;
    vector<vector<int>> up, val;
    vector<vector<pii>> g;
    vector<int> dep, tin, tout, dist;

    Tree(int n) : n(n), tim(0), g(n + 1), dep(n + 1), tin(n + 1), tout(n + 1), dist(n + 1) {
        lg = 1;
        while ((1LL << lg) <= n) ++lg;
        up.assign(n + 1, vector<int>(lg));
        val.assign(n + 1, vector<int>(lg, inf));
    }

    void add(int u, int v, int w) {
        g[u].push_back({v, w});
        g[v].push_back({u, w});
    }

    void work(int root = 1) {
        tim = 0;
        dep[root] = 0;
        dist[root] = 0;
        val[root][0] = inf;
        auto dfs = [&](auto &&self, int u, int fa) -> void {
            up[u][0] = fa;
            for (int k = 1; k < lg; ++k) {
                up[u][k] = up[up[u][k - 1]][k - 1];
                val[u][k] = min(val[u][k - 1], val[up[u][k - 1]][k - 1]);
            }
            tin[u] = ++tim;
            for (auto [v, w] : g[u]) {
                if (v == fa) continue;
                dep[v] = dep[u] + 1;
                dist[v] = dist[u] + w;
                val[v][0] = w;
                self(self, v, u);
            }
            tout[u] = tim;
        };
        dfs(dfs, root, 0);
    }

    int jmp(int u, int k) const {
        if (k < 0 || k > dep[u]) return 0;
        for (int i = 0; i < lg; ++i) {
            if ((k >> i) & 1) u = up[u][i];
        }
        return u;
    }

    int lca(int x, int y) const {
        if (dep[x] < dep[y]) swap(x, y);
        x = jmp(x, dep[x] - dep[y]);
        if (x == y) return x;
        for (int k = lg - 1; k >= 0; --k) {
            if (up[x][k] == up[y][k]) continue;
            x = up[x][k], y = up[y][k];
        }
        return up[x][0];
    }

    int dis(int x, int y) const {
        int p = lca(x, y);
        return dist[x] + dist[y] - 2 * dist[p];
    }

    int getup(int u, int k) const {
        int r = inf;
        for (int i = 0; i < lg; ++i) {
            if ((k >> i) & 1) {
                r = min(r, val[u][i]);
                u = up[u][i];
            }
        }
        return r;
    }

    int get(int x, int y) const {
        if (x == y) return 0;
        int p = lca(x, y);
        return min(getup(x, dep[x] - dep[p]), getup(y, dep[y] - dep[p]));
    }

    int kth(int x, int y, int k) const {
        int p = lca(x, y);
        int a = dep[x] - dep[p], b = dep[y] - dep[p];
        if (k < 0 || k > a + b) return 0;
        return k <= a ? jmp(x, k) : jmp(y, a + b - k);
    }

    bool anc(int x, int y) const {
        return tin[x] <= tin[y] && tout[y] <= tout[x];
    }
};
```

#### 树链剖分解法

预处理时间复杂度 $\mathcal O(N)$ ；单次查询 $\mathcal O(\log N)$ ，常数较小。

```c++
struct Tree {
    int n, tim;
    vector<vector<int>> g;
    vector<int> fa, dep, siz, son, top, dfn, out, rev;

    Tree(int n) : n(n), tim(0), g(n + 1), fa(n + 1), dep(n + 1), siz(n + 1), son(n + 1), top(n + 1), dfn(n + 1), out(n + 1), rev(n + 1) {}

    void add(int u, int v) {
        g[u].push_back(v);
        g[v].push_back(u);
    }

    void dfs1(int u, int p) {
        fa[u] = p, siz[u] = 1, son[u] = 0;

        for (auto v : g[u]) {
            if (v == p) continue;
            dep[v] = dep[u] + 1;
            dfs1(v, u);
            siz[u] += siz[v];
            if (siz[v] > siz[son[u]]) son[u] = v;
        }
    }

    void dfs2(int u, int tp) {
        top[u] = tp;
        dfn[u] = ++tim;
        rev[tim] = u;

        if (son[u]) dfs2(son[u], tp);
        for (auto v : g[u]) {
            if (v == fa[u] || v == son[u]) continue;
            dfs2(v, v);
        }

        out[u] = tim;
    }

    void work(int root = 1) {
        tim = 0;
        dep[root] = 0;
        dfs1(root, 0);
        dfs2(root, root);
    }

    int lca(int u, int v) const {
        while (top[u] != top[v]) {
            if (dep[top[u]] < dep[top[v]]) swap(u, v);
            u = fa[top[u]];
        }
        return dep[u] < dep[v] ? u : v;
    }

    int dis(int u, int v) const {
        int p = lca(u, v);
        return dep[u] + dep[v] - 2 * dep[p];
    }

    bool anc(int u, int v) const {
        return dfn[u] <= dfn[v] && dfn[v] <= out[u];
    }

    int jmp(int u, int k) const {
        if (k < 0 || k > dep[u]) return 0;

        while (k) {
            int h = top[u];
            int len = dep[u] - dep[h];

            if (k <= len) return rev[dfn[u] - k];

            k -= len + 1;
            u = fa[h];
        }
        return u;
    }

    int kth(int u, int v, int k) const {
        int p = lca(u, v);
        int a = dep[u] - dep[p], b = dep[v] - dep[p];

        if (k < 0 || k > a + b) return 0;
        return k <= a ? jmp(u, k) : jmp(v, a + b - k);
    }

    pii subtree(int u) const {
        return {dfn[u], out[u]};
    }

    template <class F> void path(int u, int v, F f) const {
        while (top[u] != top[v]) {
            if (dep[top[u]] < dep[top[v]]) swap(u, v);
            f(dfn[top[u]], dfn[u]);
            u = fa[top[u]];
        }
        if (dep[u] > dep[v]) swap(u, v);
        f(dfn[u], dfn[v]);
    }
};

```

#### Euler 序列 + RMQ 解法

用于静态无权树，`add(u,v)` 连边后调用 `work(rt)`；`lca(u,v)` 求最近公共祖先，`dis(u,v)` 求边数距离，`anc(u,v)` 判断 `u` 是否为 `v` 的祖先（含自身）。预处理与空间均为 $O(n\log n)$，查询 $O(1)$，更换根后需重新 `work(rt)`。

```c++
struct Tree {
    int n, tim;
    vector<vector<int>> g, st;
    vector<int> fa, dep, dfn, out, rev, pos;

    Tree(int n) : n(n), tim(0), g(n + 1), fa(n + 1), dep(n + 1), dfn(n + 1), out(n + 1), rev(n + 1), pos(n + 1) {}

    void add(int u, int v) {
        g[u].push_back(v);
        g[v].push_back(u);
    }

    int get(int u, int v) const {
        return dep[u] < dep[v] ? u : v;
    }

    void work(int rt = 1) {
        tim = 0;
        fa[rt] = dep[rt] = 0;
        vector<int> ord(1), it(n + 1), stk{rt};
        while (stk.size()) {
            int u = stk.back();
            if (!it[u]) {
                dfn[u] = ++tim;
                rev[tim] = u;
                pos[u] = ord.size();
                ord.push_back(u);
            }
            if (it[u] == g[u].size()) {
                out[u] = tim;
                stk.pop_back();
                if (stk.size()) ord.push_back(stk.back());
            } else {
                int v = g[u][it[u]++];
                if (v == fa[u]) continue;
                fa[v] = u;
                dep[v] = dep[u] + 1;
                stk.push_back(v);
            }
        }
        int m = ord.size();
        m--;
        st.clear();
        st.push_back(move(ord));
        for (int k = 1; (1LL << k) <= m; k++) {
            int len = 1LL << k;
            st.emplace_back(m - len + 2);
            for (int i = 1; i + len - 1 <= m; i++) {
                st[k][i] = get(st[k - 1][i], st[k - 1][i + (len >> 1)]);
            }
        }
    }

    int lca(int u, int v) const {
        int l = pos[u], r = pos[v];
        if (l > r) swap(l, r);
        int k = __lg(r - l + 1);
        return get(st[k][l], st[k][r - (1LL << k) + 1]);
    }

    int dis(int u, int v) const {
        return dep[u] + dep[v] - 2 * dep[lca(u, v)];
    }

    bool anc(int u, int v) const {
        return dfn[u] <= dfn[v] && dfn[v] <= out[u];
    }
};
```

#### Tarjan 离线 LCA

`add(u,v)` 连边，`ask(u,v)` 登记询问并返回从 `0` 开始的编号，`work(rt)` 按登记顺序返回答案。时间 $O((n+q)\alpha(n))$，空间 $O(n+q)$，支持重新换根计算。

```c++
struct Tree {
    int n;
    vector<vector<int>> g;
    vector<vector<pii>> q;
    vector<int> ans;
    Tree(int n) : n(n), g(n + 1), q(n + 1) {}
    void add(int u, int v) {
        g[u].push_back(v);
        g[v].push_back(u);
    }
    int ask(int u, int v) {
        int id = ans.size();
        ans.push_back(0);
        q[u].push_back({v, id});
        q[v].push_back({u, id});
        return id;
    }
    vector<int> work(int rt = 1) {
        vector<int> f(n + 1), siz(n + 1, 1), anc(n + 1);
        vector<int> fa(n + 1), it(n + 1), st{rt};
        vector<bool> vis(n + 1);
        iota(f.begin(), f.end(), 0);
        anc = f;
        auto find = [&](int u) {
            while (u != f[u]) u = f[u] = f[f[u]];
            return u;
        };
        while (st.size()) {
            int u = st.back();
            if (it[u] < g[u].size()) {
                int v = g[u][it[u]++];
                if (v == fa[u]) continue;
                fa[v] = u;
                st.push_back(v);
            } else {
                vis[u] = true;
                for (auto [v, id] : q[u]) if (vis[v]) ans[id] = anc[find(v)];
                st.pop_back();
                if (fa[u]) {
                    int x = find(u), y = find(fa[u]);
                    if (siz[x] < siz[y]) swap(x, y);
                    f[y] = x;
                    siz[x] += siz[y];
                    anc[x] = fa[u];
                }
            }
        }
        return ans;
    }
};
```

#### 长链剖分：常数时间跳祖先

`work(rt)` 后，`jmp(u,k)` 返回第 `k` 级祖先，越界返回 `0`。预处理及空间 $O(n\log n)$，跳祖先 $O(1)$，`lca`、`dis` 为 $O(\log n)$。

```c++
struct Tree {
    int n, lg;
    vector<vector<int>> g, val, up, dn;
    vector<int> dep, fa, son, len, top;
    Tree(int n) : n(n), lg(__lg(n) + 1), g(n + 1), val(n + 1, vector<int>(lg)),
        dep(n + 1), fa(n + 1), son(n + 1), len(n + 1), top(n + 1) {}
    void add(int u, int v) {
        g[u].push_back(v);
        g[v].push_back(u);
    }
    void work(int rt = 1) {
        fill(son.begin(), son.end(), 0);
        fill(len.begin(), len.end(), 0);
        fa[rt] = dep[rt] = 0;
        vector<int> ord{rt};
        for (int i = 0; i < ord.size(); i++) {
            int u = ord[i];
            val[u][0] = fa[u];
            for (int k = 1; k < lg; k++) val[u][k] = val[val[u][k - 1]][k - 1];
            for (int v : g[u]) if (v != fa[u]) {
                fa[v] = u;
                dep[v] = dep[u] + 1;
                ord.push_back(v);
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            int u = ord[i];
            for (int v : g[u]) if (v != fa[u] && len[v] > len[son[u]]) son[u] = v;
            len[u] = len[son[u]] + 1;
        }
        up.assign(n + 1, {});
        dn.assign(n + 1, {});
        for (int u : ord) if (!fa[u] || son[fa[u]] != u) {
            for (int v = u, i = 0; i < len[u]; i++, v = fa[v]) up[u].push_back(v);
            for (int v = u; v; v = son[v]) {
                top[v] = u;
                dn[u].push_back(v);
            }
        }
    }
    int jmp(int u, int k) const {
        if (k < 0 || k > dep[u]) return 0;
        if (!k) return u;
        int b = __lg(k);
        u = val[u][b];
        k -= 1LL << b;
        int t = top[u], d = dep[u] - dep[t];
        return k <= d ? dn[t][d - k] : up[t][k - d];
    }
    int lca(int u, int v) const {
        if (dep[u] < dep[v]) swap(u, v);
        u = jmp(u, dep[u] - dep[v]);
        if (u == v) return u;
        for (int k = lg - 1; k >= 0; k--) if (val[u][k] != val[v][k]) {
            u = val[u][k];
            v = val[v][k];
        }
        return fa[u];
    }
    int dis(int u, int v) const {
        return dep[u] + dep[v] - 2 * dep[lca(u, v)];
    }
};
```

### 树上差分

离线路径加后统一求各点、各边的增量，复用上文“树链剖分解法”的 `Tree`，先执行 `tr.work(root)` 再创建对象，使用期间保持树与根不变。`askV()[u]` 为点增量，`askE()[u]` 为边 `(tr.fa[u],u)` 的增量（根位置为 `0`）；查询不改动差分数组，`clear()` 开始新一批。

```c++
struct TreeDiff {
    const Tree &tr;
    vector<int> tv, te;

    TreeDiff(const Tree &tr) : tr(tr), tv(tr.n + 1), te(tr.n + 1) {}
    void clear() {
        fill(tv.begin(), tv.end(), 0);
        fill(te.begin(), te.end(), 0);
    }
    void addV(int u, int v, int w) {
        int p = tr.lca(u, v);
        tv[u] += w;
        tv[v] += w;
        tv[p] -= w;
        if (tr.fa[p]) tv[tr.fa[p]] -= w;
    }
    void addE(int u, int v, int w) {
        int p = tr.lca(u, v);
        te[u] += w;
        te[v] += w;
        te[p] -= w;
        te[p] -= w;
    }
    vector<int> collect(vector<int> a) const {
        for (int i = tr.n; i >= 1; i--) {
            int u = tr.rev[i];
            if (tr.fa[u]) a[tr.fa[u]] += a[u];
        }
        return a;
    }
    vector<int> askV() const { return collect(tv); }
    vector<int> askE() const { return collect(te); }
};
```

```cpp
TreeDiff d(tr);
d.addV(u, v, w); // 路径上的点都加 w，包含两端点
d.addE(u, v, w); // 路径上的边都加 w
auto a = d.askV(), b = d.askE(); // 所有修改完成后统一回收
```

### 树链剖分

#### 路径与子树操作（内置线段树）

内置线段树，支持点权的路径加、路径和、子树加与子树和。建树与空间复杂度为 $\mathcal O(n)$；路径拆成 $\mathcal O(\log n)$ 个区间，因此路径操作为 $\mathcal O(\log^2 n)$，子树操作为 $\mathcal O(\log n)$。

```c++
struct HLD {
    struct node {
        int l, r, sum, lazy;
    };
    int n, idx = 0;
    vector<vector<int>> g;
    vector<int> fa, dep, siz, son, top, id;
    vector<node> t;

    HLD(int n) : n(n), g(n + 1), fa(n + 1), dep(n + 1), siz(n + 1),
                 son(n + 1), top(n + 1), id(n + 1), t(4 * n + 5) {}

    void add(int u, int v) {
        g[u].push_back(v);
        g[v].push_back(u);
    }

    void work(const vector<int> &a, int root = 1) {
        idx = 0;
        vector<int> b(n + 1);
        auto dfs1 = [&](auto &&self, int u, int p) -> void {
            fa[u] = p;
            dep[u] = dep[p] + 1;
            siz[u] = 1;
            son[u] = 0;
            for (int v : g[u]) {
                if (v == p) continue;
                self(self, v, u);
                siz[u] += siz[v];
                if (siz[v] > siz[son[u]]) son[u] = v;
            }
        };
        auto dfs2 = [&](auto &&self, int u, int up) -> void {
            top[u] = up;
            id[u] = ++idx;
            b[idx] = a[u];
            if (son[u]) self(self, son[u], up);
            for (int v : g[u]) {
                if (v == fa[u] || v == son[u]) continue;
                self(self, v, v);
            }
        };
        auto build = [&](auto &&self, int p, int l, int r) -> void {
            t[p] = {l, r, 0, 0};
            if (l == r) return t[p].sum = b[l], void();
            int m = (l + r) >> 1;
            self(self, p << 1, l, m);
            self(self, p << 1 | 1, m + 1, r);
            pull(p);
        };
        dfs1(dfs1, root, 0);
        dfs2(dfs2, root, root);
        build(build, 1, 1, n);
    }

    void work(int root = 1) { work(vector<int>(n + 1), root); }

    void pull(int p) { t[p].sum = t[p << 1].sum + t[p << 1 | 1].sum; }

    void apply(int p, int x) {
        t[p].sum += (t[p].r - t[p].l + 1) * x;
        t[p].lazy += x;
    }

    void push(int p) {
        if (!t[p].lazy) return;
        apply(p << 1, t[p].lazy);
        apply(p << 1 | 1, t[p].lazy);
        t[p].lazy = 0;
    }

    void update(int l, int r, int x, int p = 1) {
        if (l <= t[p].l && t[p].r <= r) return apply(p, x);
        push(p);
        int m = (t[p].l + t[p].r) >> 1;
        if (l <= m) update(l, r, x, p << 1);
        if (r > m) update(l, r, x, p << 1 | 1);
        pull(p);
    }

    int query(int l, int r, int p = 1) {
        if (l <= t[p].l && t[p].r <= r) return t[p].sum;
        push(p);
        int m = (t[p].l + t[p].r) >> 1, ans = 0;
        if (l <= m) ans += query(l, r, p << 1);
        if (r > m) ans += query(l, r, p << 1 | 1);
        return ans;
    }

    void modify(int u, int v, int x) { // 路径加
        while (top[u] != top[v]) {
            if (dep[top[u]] < dep[top[v]]) swap(u, v);
            update(id[top[u]], id[u], x);
            u = fa[top[u]];
        }
        if (dep[u] > dep[v]) swap(u, v);
        update(id[u], id[v], x);
    }

    int ask(int u, int v) { // 路径和
        int ans = 0;
        while (top[u] != top[v]) {
            if (dep[top[u]] < dep[top[v]]) swap(u, v);
            ans += query(id[top[u]], id[u]);
            u = fa[top[u]];
        }
        if (dep[u] > dep[v]) swap(u, v);
        return ans + query(id[u], id[v]);
    }

    void modify(int u, int x) { update(id[u], id[u] + siz[u] - 1, x); }
    int ask(int u) { return query(id[u], id[u] + siz[u] - 1); }
};

HLD tr(n);
// tr.add(u, v); 添加树边
tr.work(a, root);       // a 为点权；tr.work(root) 则全部初始化为 0

tr.modify(u, v, x);     // u 到 v 的路径加 x
tr.ask(u, v);           // 路径和
tr.modify(u, x);        // u 的子树加 x
tr.ask(u);              // 子树和
```

#### 独立路径分解与边权

标准重链剖分，按子树大小选择重儿子。空间与初始化复杂度 $\mathcal O(n)$；一条路径被拆为 $\mathcal O(\log n)$ 个 DFS 序区间。配合树状数组或线段树时，路径操作复杂度通常为 $\mathcal O(\log^2 n)$，子树操作仅对应一个区间。

将一条路径拆为若干 DFS 序区间。若区间结果可按任意顺序合并，则可直接接入（`exBIT`、`SegTree`）；若答案依赖整个连续区间的全局分布（小波矩阵、动态开点权值线段树），或合并顺序有意义（字符串拼接、矩阵连乘），则不能直接套用。

##### 统一封装

结点 $x$ 子树内的点权区间为 `[in[x], out[x]]`，边权区间为 `[in[x] + 1, out[x]]`。`forEdgePath(x, y, f)` 用于处理这个差值，约定边 `(fa[v], v)` 的权值存于 `in[v]`，并跳过 LCA。

```c++
struct HLDPath {
    int n, cur = 0;
    vector<vector<int>> g;
    vector<int> fa, dep, siz, son, top, in, out;

    HLDPath(int n) : n(n), g(n + 1), fa(n + 1), dep(n + 1),
        siz(n + 1), son(n + 1), top(n + 1), in(n + 1), out(n + 1) {}

    void addEdge(int x, int y) {
        g[x].push_back(y);
        g[y].push_back(x);
    }
    void build(int root = 1) {
        cur = 0;
        dep[root] = 0;
        auto dfs1 = [&](auto &&self, int x, int p) -> void {
            fa[x] = p, siz[x] = 1, son[x] = 0;
            for (int y : g[x]) {
                if (y == p) continue;
                dep[y] = dep[x] + 1;
                self(self, y, x);
                siz[x] += siz[y];
                if (!son[x] || siz[y] > siz[son[x]]) son[x] = y;
            }
        };
        dfs1(dfs1, root, 0);
        auto dfs2 = [&](auto &&self, int x, int h) -> void {
            top[x] = h, in[x] = ++cur;
            if (son[x]) self(self, son[x], h);
            for (int y : g[x]) {
                if (y != fa[x] && y != son[x]) self(self, y, y);
            }
            out[x] = cur;
        };
        dfs2(dfs2, root, root);
    }
    int lca(int x, int y) const {
        while (top[x] != top[y]) {
            if (dep[top[x]] < dep[top[y]]) swap(x, y);
            x = fa[top[x]];
        }
        return dep[x] < dep[y] ? x : y;
    }
    pair<int, int> subtree(int x) const {
        return {in[x], out[x]};
    }

    template<class F> void splitPath(int x, int y, F &f, bool edge) const {
        while (top[x] != top[y]) {
            if (dep[top[x]] < dep[top[y]]) swap(x, y);
            f(in[top[x]], in[x]);
            x = fa[top[x]];
        }
        if (dep[x] > dep[y]) swap(x, y);
        int l = in[x] + (edge ? 1 : 0);
        if (l <= in[y]) f(l, in[y]);
    }
    template<class F> void forPath(int x, int y, F &&f) const { // 点权路径
        splitPath(x, y, f, false);
    }
    template<class F> void forEdgePath(int x, int y, F &&f) const { // 边权路径
        splitPath(x, y, f, true);
    }
};
```

### 树上路径交

```c++
int intersection(int x, int y, int X, int Y) {
    vector<int> t = {lca(x, X), lca(x, Y), lca(y, X), lca(y, Y)};
    sort(t.begin(), t.end(), [&](int a, int b) { return dep[a] < dep[b]; });
    int r = lca(x, y), R = lca(X, Y);
    if (dep[t[0]] < min(dep[r], dep[R]) || dep[t[2]] < max(dep[r], dep[R])) {
        return 0;
    }
```

### Kruskal 重构树

用于无向图，先 `build(edges)`（每条边为 `{u,v,w}`），`ask(u,v)` 返回两点路径上最大边权的最小可能值，不连通返回 `nullopt`，同点返回 `0`。`count(u,w)` 返回只走边权不超过 `w` 的边时可达的点数（含自身），支持负权、重边和重复构建；建树 $O(m\log m+n\log n)$，查询 $O(\log n)$，空间 $O(m+n\log n)$。

```c++
struct KRT {
    struct Edge {
        int u, v, w;
    };
    int n, tot, lg;
    vector<int> val, siz, fa, dep, bel;
    vector<pii> son;
    vector<vector<int>> up;

    KRT(int n) : n(n) {}
    void build(vector<Edge> e) {
        tot = n;
        int cap = 2 * n + 1;
        vector<int> f(cap);
        iota(f.begin(), f.end(), 0LL);
        val.assign(cap, 0);
        siz.assign(cap, 0);
        fa.assign(cap, 0);
        dep.assign(cap, 0);
        son.assign(cap, {0, 0});
        bel.assign(n + 1, 0);
        for (int u = 1; u <= n; u++) siz[u] = 1;
        auto find = [&](int u) {
            while (f[u] != u) u = f[u] = f[f[u]];
            return u;
        };
        sort(e.begin(), e.end(), [](const Edge &a, const Edge &b) { return a.w < b.w; });
        for (auto [u, v, w] : e) {
            int x = find(u), y = find(v);
            if (x == y) continue;
            ++tot;
            val[tot] = w;
            siz[tot] = siz[x] + siz[y];
            son[tot] = {x, y};
            fa[x] = fa[y] = tot;
            f[x] = f[y] = tot;
        }
        for (int u = 1; u <= n; u++) bel[u] = find(u);

        lg = 1;
        while ((1LL << lg) <= tot) lg++;
        up.assign(tot + 1, vector<int>(lg));
        for (int u = tot; u >= 1; u--) { // 父节点编号总大于子节点
            dep[u] = dep[fa[u]] + 1;
            up[u][0] = fa[u];
            for (int k = 1; k < lg; k++) up[u][k] = up[up[u][k - 1]][k - 1];
        }
    }
    int lca(int u, int v) const {
        if (bel[u] != bel[v]) return 0;
        if (dep[u] < dep[v]) swap(u, v);
        int d = dep[u] - dep[v];
        for (int k = 0; k < lg; k++) {
            if (d >> k & 1) u = up[u][k];
        }
        if (u == v) return u;
        for (int k = lg - 1; k >= 0; k--) {
            if (up[u][k] != up[v][k]) u = up[u][k], v = up[v][k];
        }
        return fa[u];
    }
    optional<int> ask(int u, int v) const {
        int p = lca(u, v);
        if (!p) return nullopt;
        if (u == v) return 0;
        return val[p];
    }
    int count(int u, int w) const {
        for (int k = lg - 1; k >= 0; k--) {
            int p = up[u][k];
            if (p && val[p] <= w) u = p;
        }
        return siz[u];
    }
};
```

### 虚树

查询点数远少于树的节点数，且查询点之间的 $\rm LCA$ 不会改变时，可以考虑使用虚树，将有效点按照 $dfn$ 序排序去重后放入，再将 $\rm lca$ 也放入，随后再排序去重，找到每个节点最近的被保留下来的 $\rm lca$ 连边。

```c++
struct VTree {
    int rt = 0;
    vector<int> u;
    vector<pii> e;

    template <class F, class G> void build(vector<int> p, F lca, G dfn) {
        u.clear(), e.clear();
        rt = 0;
        if (p.empty()) return;
        auto cmp = [&](int u, int v) { return dfn(u) < dfn(v); };
        ranges::sort(p, cmp);
        p.erase(unique(p.begin(), p.end()), p.end());

        int n0 = p.size();
        for (int i = 1; i < n0; ++i) p.push_back(lca(p[i - 1], p[i]));
        ranges::sort(p, cmp);
        p.erase(unique(p.begin(), p.end()), p.end());
        u = p;

        vector<int> st;
        for (int u : u) {
            while (!st.empty() && lca(st.back(), u) != st.back()) st.pop_back();
            if (!st.empty()) e.push_back({st.back(), u});
            st.push_back(u);
        }
        rt = u.front();
    }
    auto build(int n) {
        vector<vector<int>> g(n + 1);
        for (auto [u, v] : e) g[u].push_back(v);
        return g;
    }
};
```

### 换根 DP

求连通树上每个点作为根的 DP。传入 `id`、`merge`、`trans`，设置点自身的状态 `bas[u]`、添加边，调用 `work()` 后读取 `ans[u]`；状态与两种操作均为 $O(1)$ 时，总时间、空间均为 $O(n)$。

`merge(a,b)`：合并同一节点处的两份状态并返回结果，须满足结合律，`id` 为合并时不影响结果的空状态。例如距离和问题中，将点数与距离和分别相加。

`trans(a,u,v,eid)`：将状态 `a` 从 `u` 沿边 `eid` 换算到 `v`，不加入 `v` 自身的信息；`eid` 是 `add` 返回的边号，从 `0` 开始。例如距离和增加 `a.cnt * w[eid]`。

```c++
template <class T, class F, class G> struct Rerooting {
    int n, m = 0;
    T id;
    F merge;
    G trans;
    vector<vector<pii>> g;
    vector<int> fa, ord;
    vector<T> bas, dn, up, ans;

    Rerooting(int n, T id, F merge, G trans)
        : n(n), id(id), merge(merge), trans(trans), g(n + 1), fa(n + 1),
          bas(n + 1, id), dn(n + 1, id), up(n + 1, id), ans(n + 1, id) {}

    int add(int u, int v) {
        int eid = m++;
        g[u].push_back({v, eid});
        g[v].push_back({u, eid});
        return eid;
    }
    void work(int root = 1) {
        fill(fa.begin(), fa.end(), 0);
        ord = {root};
        for (int i = 0; i < ord.size(); i++) {
            int u = ord[i];
            for (auto [v, eid] : g[u]) {
                if (v == fa[u]) continue;
                fa[v] = u;
                ord.push_back(v);
            }
        }

        for (int i = n - 1; i >= 0; i--) {
            int u = ord[i];
            dn[u] = bas[u];
            for (auto [v, eid] : g[u]) {
                if (v != fa[u]) dn[u] = merge(dn[u], trans(dn[v], v, u, eid));
            }
        }

        fill(up.begin(), up.end(), id);
        for (int u : ord) {
            int deg = g[u].size();
            vector<T> val(deg, id), pre(deg + 1, id), suf(deg + 1, id);
            for (int i = 0; i < deg; i++) {
                auto [v, eid] = g[u][i];
                val[i] = v == fa[u] ? up[u] : trans(dn[v], v, u, eid);
                pre[i + 1] = merge(pre[i], val[i]);
            }
            for (int i = deg - 1; i >= 0; i--) suf[i] = merge(val[i], suf[i + 1]);
            ans[u] = merge(bas[u], pre[deg]);
            for (int i = 0; i < deg; i++) {
                auto [v, eid] = g[u][i];
                if (v == fa[u]) continue;
                T a = merge(bas[u], merge(pre[i], suf[i + 1]));
                up[v] = trans(a, u, v, eid);
            }
        }
    }
};
```

### 树上启发式合并 (DSU on tree)

$\mathcal O(N\log N)$ 。

```c++
struct HLD {
    vector<vector<int>> e;
    vector<int> siz, son, cnt;
    vector<LL> ans;
    LL sum, Max;
    int hson;
    HLD(int n) {
        e.resize(n + 1);
        siz.resize(n + 1);
        son.resize(n + 1);
        ans.resize(n + 1);
        cnt.resize(n + 1);
        hson = 0;
        sum = 0;
        Max = 0;
    }
    void add(int u, int v) {
        e[u].push_back(v);
        e[v].push_back(u);
    }
    void dfs1(int u, int fa) {
        siz[u] = 1;
        for (auto v : e[u]) {
            if (v == fa) continue;
            dfs1(v, u);
            siz[u] += siz[v];
            if (siz[v] > siz[son[u]]) son[u] = v;
        }
    }
    void calc(int u, int fa, int val) {
        cnt[color[u]] += val;
        if (cnt[color[u]] > Max) {
            Max = cnt[color[u]];
            sum = color[u];
        } else if (cnt[color[u]] == Max) {
            sum += color[u];
        }
        for (auto v : e[u]) {
            if (v == fa || v == hson) continue;
            calc(v, u, val);
        }
    }
    void dfs2(int u, int fa, int opt) {
        for (auto v : e[u]) {
            if (v == fa || v == son[u]) continue;
            dfs2(v, u, 0);
        }
        if (son[u]) {
            dfs2(son[u], u, 1);
            hson = son[u]; //记录重链编号，计算的时候跳过
        }
        calc(u, fa, 1);
        hson = 0; //消除的时候所有儿子都清除
        ans[u] = sum;
        if (!opt) {
            calc(u, fa, -1);
            sum = 0;
            Max = 0;
        }
    }
};
```

### 点分治 / 树的重心

重心的定义：删除树上的某一个点，会得到若干棵子树；删除某点后，得到的最大子树最小，这个点称为重心。我们假设某个点是重心，记录此时最大子树的最小值，遍历完所有点后取最大值即可。

> 重心的性质：重心最多可能会有两个，且此时两个重心相邻。

点分治的一般过程是：取重心为新树的根，随后使用 $\tt dfs$ 处理当前这棵树，灵活运用 `child` 和 `pre` 两个数组分别计算通过根节点、不通过根节点的路径信息，根据需要进行答案的更新；再对子树分治，寻找子树的重心，……。时间复杂度降至 $\mathcal O(N\log N)$ 。

```c++
constexpr int inf = 1LL << 60;
int root = 0, MaxTree = inf; //分别代表重心下标、最大子树大小
vector<int> vis(n + 1), siz(n + 1);
auto get = [&](auto self, int x, int fa, int n) -> void { // 获取树的重心
    siz[x] = 1;
    int val = 0;
    for (auto [y, w] : ver[x]) {
        if (y == fa || vis[y]) continue;
        self(self, y, x, n);
        siz[x] += siz[y];
        val = max(val, siz[y]);
    }
    val = max(val, n - siz[x]);
    if (val < MaxTree) {
        MaxTree = val;
        root = x;
    }
};

auto clac = [&](int x) -> void { // 以 x 为新的根，维护询问
    set<int> pre = {0}; // 记录到根节点 x 距离为 i 的路径是否存在
    vector<int> dis(n + 1);
    for (auto [y, w] : ver[x]) {
        if (vis[y]) continue;
        vector<int> child; // 记录 x 的子树节点的深度信息
        auto dfs = [&](auto self, int x, int fa) -> void {
            child.push_back(dis[x]);
            for (auto [y, w] : ver[x]) {
                if (y == fa || vis[y]) continue;
                dis[y] = dis[x] + w;
                self(self, y, x);
            }
        };
        dis[y] = w;
        dfs(dfs, y, x);

        for (auto it : child) {
            for (int i = 1; i <= m; i++) { // 根据询问更新值
                if (q[i] < it || !pre.count(q[i] - it)) continue;
                ans[i] = 1;
            }
        }
        pre.insert(child.begin(), child.end());
    }
};

auto dfz = [&](auto self, int x, int fa) -> void { // 点分治
    vis[x] = 1; // 标记已经被更新过的旧重心，确保只对子树分治
    clac(x);
    for (auto [y, w] : ver[x]) {
        if (y == fa || vis[y]) continue;
        MaxTree = inf;
        get(get, y, x, siz[y]);
        self(self, root, x);
    }
};

get(get, 1, 0, n);
dfz(dfz, root, 0);
```

### 点分树：动态最近点

维护非负边权树上的动态标记点集，`add(u,v,w)` 加边后调用 `build()`；`add(u)` 标记、`del(u)` 取消，`ask(u)` 返回最近标记点的距离，无标记时返回 `-1`。`build()` 与 `clear()` 会清空标记，要求任意两点距离小于 `inf`；预处理 $O(n\log n)$，修改 $O(\log^2 n)$，查询 $O(\log n)$，空间 $O(n\log n)$。

```c++
struct CenTree {
    static constexpr int inf = 1LL << 60;
    int n;
    vector<vector<pii>> g, p;
    vector<int> fa; // 点分树上的父节点
    vector<bool> on;
    vector<multiset<int>> q;

    CenTree(int n) : n(n), g(n + 1), p(n + 1), fa(n + 1), on(n + 1), q(n + 1) {}
    void add(int u, int v, int w = 1) {
        g[u].push_back({v, w});
        g[v].push_back({u, w});
    }
    void clear() {
        fill(on.begin(), on.end(), false);
        for (auto &s : q) s.clear();
    }
    void build(int rt = 1) {
        clear();
        fill(fa.begin(), fa.end(), 0);
        for (auto &v : p) v.clear();
        vector<int> siz(n + 1), pre(n + 1), ord;
        vector<bool> ban(n + 1);
        vector<array<int, 3>> st;
        auto dfs = [&](auto &&self, int rt, int f) -> void {
            ord = {rt};
            pre[rt] = 0;
            for (int i = 0; i < ord.size(); i++) {
                int u = ord[i];
                for (auto [v, w] : g[u]) {
                    if (v == pre[u] || ban[v]) continue;
                    pre[v] = u;
                    ord.push_back(v);
                }
            }
            int tot = ord.size(), c = rt;
            for (int i = tot - 1; i >= 0; i--) {
                int u = ord[i], mx = 0;
                siz[u] = 1;
                for (auto [v, w] : g[u]) {
                    if (v == pre[u] || ban[v]) continue;
                    siz[u] += siz[v];
                    mx = max(mx, siz[v]);
                }
                if (max(mx, tot - siz[u]) <= tot / 2) c = u;
            }
            fa[c] = f;
            ban[c] = true;
            st = {{c, 0, 0}};
            while (st.size()) {
                auto [u, pr, d] = st.back();
                st.pop_back();
                p[u].push_back({c, d});
                for (auto [v, w] : g[u]) {
                    if (v != pr && !ban[v]) st.push_back({v, u, d + w});
                }
            }
            for (auto [v, w] : g[c]) {
                if (!ban[v]) self(self, v, c);
            }
        };
        if (n) dfs(dfs, rt, 0);
    }
    bool set(int u, bool v) {
        if (on[u] == v) return false;
        on[u] = v;
        for (auto [c, d] : p[u]) {
            if (v) q[c].insert(d);
            else q[c].erase(q[c].find(d));
        }
        return true;
    }
    bool add(int u) {
        return set(u, true);
    }
    bool del(int u) {
        return set(u, false);
    }
    int ask(int u) const {
        int ans = inf;
        for (auto [c, d] : p[u]) {
            if (q[c].size()) ans = min(ans, d + *q[c].begin());
        }
        return ans == inf ? -1 : ans;
    }
};
```

### 动态树 LCT

维护森林的动态连边、断边和点权路径统计，`set(u,x)` 修改点权，`link(u,v)` 连边，`cut(u,v)` 删边；连边将成环或待删边不存在时返回 `false`。`same(u,v)` 判断连通，连通时 `ask(u,v)` 返回 `{路径点权和, 路径点权最大值}`；点权默认为 `0`，点权及所有路径和须在 `int` 范围内，单次操作均摊 $O(\log n)$。

```c++
struct LCT {
    vector<pii> ch;
    vector<int> fa, val, sum, mx, st;
    vector<bool> tag;

    LCT(int n, int v = 0) : ch(n + 1), fa(n + 1), val(n + 1, v),
        sum(n + 1, v), mx(n + 1, v), tag(n + 1) {
        sum[0] = 0;
    }
    bool top(int x) const {
        return ch[fa[x]][0] != x && ch[fa[x]][1] != x;
    }
    void pull(int x) {
        sum[x] = sum[ch[x][0]] + val[x] + sum[ch[x][1]];
        mx[x] = val[x];
        for (int v : ch[x]) if (v) mx[x] = max(mx[x], mx[v]);
    }
    void rev(int x) {
        if (!x) return;
        swap(ch[x][0], ch[x][1]);
        tag[x] = !tag[x];
    }
    void push(int x) {
        if (!tag[x]) return;
        rev(ch[x][0]);
        rev(ch[x][1]);
        tag[x] = false;
    }
    void rot(int x) {
        int y = fa[x], z = fa[y], k = ch[y][1] == x, w = ch[x][k ^ 1];
        if (!top(y)) ch[z][ch[z][1] == y] = x;
        fa[x] = z;
        ch[x][k ^ 1] = y;
        fa[y] = x;
        ch[y][k] = w;
        if (w) fa[w] = y;
        pull(y);
        pull(x);
    }
    void splay(int x) {
        st = {x};
        for (int u = x; !top(u); u = fa[u]) st.push_back(fa[u]);
        for (int i = st.size(); i--;) push(st[i]);
        while (!top(x)) {
            int y = fa[x], z = fa[y];
            if (!top(y)) rot((ch[y][0] == x) == (ch[z][0] == y) ? y : x);
            rot(x);
        }
        pull(x);
    }
    void access(int x) {
        int y = 0;
        for (int u = x; u; u = fa[u]) {
            splay(u);
            ch[u][1] = y;
            pull(u);
            y = u;
        }
        splay(x);
    }
    void make(int x) {
        access(x);
        rev(x);
    }
    int root(int x) {
        access(x);
        while (ch[x][0]) {
            push(x);
            x = ch[x][0];
        }
        splay(x);
        return x;
    }
    bool same(int u, int v) {
        return u == v || root(u) == root(v);
    }
    bool link(int u, int v) {
        make(u);
        if (root(v) == u) return false;
        fa[u] = v;
        return true;
    }
    bool cut(int u, int v) {
        make(u);
        access(v);
        if (ch[v][0] != u || ch[u][1]) return false;
        ch[v][0] = fa[u] = 0;
        pull(v);
        return true;
    }
    void split(int u, int v) {
        make(u);
        access(v);
    }
    void set(int u, int v) {
        access(u);
        val[u] = v;
        pull(u);
    }
    pii ask(int u, int v) {
        split(u, v);
        return {sum[v], mx[v]};
    }
};
```
