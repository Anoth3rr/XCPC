## 第八章：数据结构

约定：
- 数组下标、元素排名等均从 $1$ 开始；
- 查询区间为闭区间 $[l, r]$；
- 相同名称的封装需独立选择，不能同时复制。

### 并查集

普通版本使用路径压缩与按大小合并，单次操作均摊复杂度为 $\mathcal O(\alpha(n))$，根编号不固定；可撤销、可删除版本的复杂度见对应小节。

#### 常用封装

```c++
struct DSU {
    vector<int> f, siz;
    int cnt;

    DSU(int n) : f(n + 1), siz(n + 1, 1), cnt(n) {
        iota(f.begin(), f.end(), 0);
    }
    int find(int x) {
        while (x != f[x]) x = f[x] = f[f[x]];
        return x;
    }
    bool merge(int x, int y) {
        x = find(x), y = find(y);
        if (x == y) return false;
        if (siz[x] < siz[y]) swap(x, y);
        f[y] = x;
        siz[x] += siz[y];
        cnt--;
        return true;
    }
    bool same(int x, int y) {
        return find(x) == find(y);
    }
    int size(int x) {
        return siz[find(x)];
    }
};
```

#### 扩展封装（维护无向图连通块信息）

仅用于无向图，维护无向图连通块数量、块内点数和边数。

```c++
struct exDSU {
    vector<int> f, siz, e;
    int cnt;

    exDSU(int n) : f(n + 1), siz(n + 1, 1), e(n + 1), cnt(n) {
        iota(f.begin(), f.end(), 0);
    }
    int find(int x) {
        while (x != f[x]) x = f[x] = f[f[x]];
        return x;
    }
    bool add(int x, int y) {
        x = find(x), y = find(y);
        if (x == y) {
            e[x]++;
            return false;
        }
        if (siz[x] < siz[y]) swap(x, y);
        f[y] = x;
        siz[x] += siz[y];
        e[x] += e[y] + 1;
        cnt--;
        return true;
    }
    bool same(int x, int y) { return find(x) == find(y); }
    int size(int x) { return siz[find(x)]; }
    int getEdge(int x) { return e[find(x)]; }
    bool isTree(int x) {
        x = find(x);
        return e[x] == siz[x] - 1;
    }
};
```

#### 带权并查集

当题目逐条给出两点间的差值或关系，并要求判断新陈述能否与已有陈述同时成立时使用。重载 `+` 表示正向运算，重载 `-` 为逆运算。

```c++
template <class T> struct wDSU {
    vector<int> f, siz;
    vector<T> dis;

    wDSU(int n) : f(n + 1), siz(n + 1, 1), dis(n + 1) {
        iota(f.begin(), f.end(), 0);
    }
    int find(int x) {
        if (x == f[x]) return x;
        int p = f[x];
        f[x] = find(p);
        dis[x] = dis[x] + dis[p];
        return f[x];
    }
    bool add(int x, int y, T w) {
        int rx = find(x), ry = find(y);
        if (rx == ry) return dis[x] - dis[y] == w;
        T d = w - dis[x] + dis[y];
        if (siz[rx] < siz[ry]) {
            f[rx] = ry;
            dis[rx] = d;
            siz[ry] += siz[rx];
        } else {
            f[ry] = rx;
            dis[ry] = T{} - d;
            siz[rx] += siz[ry];
        }
        return true;
    }
    bool same(int x, int y) {
        return find(x) == find(y);
    }
    T diff(int x, int y) {
        find(x), find(y);
        return dis[x] - dis[y];
    }
};
```

#### 可删除并查集

`erase(x)` 将元素 $x$ 从原集合分离为单元素集合，其他元素仍保持连通；不支持删除图中的一条边。为 $x$ 新建替身节点，旧节点保留，按秩合并并单独维护有效元素数；进行 $q$ 次操作的总时间为 $\mathcal O((n+q)\alpha(n+q))$，空间为 $\mathcal O(n+d)$，其中 $d$ 为实际新建替身的次数。

```cpp
struct DSU {
    vector<int> f, id, siz, rk;
    DSU(int n) : f(n + 1), id(n + 1), siz(n + 1, 1), rk(n + 1) {
        iota(f.begin(), f.end(), 0);
        iota(id.begin(), id.end(), 0);
    }
    int find(int x) {
        while (x != f[x]) x = f[x] = f[f[x]];
        return x;
    }
    bool merge(int x, int y) {
        x = find(id[x]), y = find(id[y]);
        if (x == y) return false;
        if (rk[x] < rk[y]) swap(x, y);
        f[y] = x, siz[x] += siz[y];
        if (rk[x] == rk[y]) ++rk[x];
        return true;
    }
    void erase(int x) {
        int p = find(id[x]);
        if (siz[p] == 1) return;
        --siz[p];
        id[x] = f.size();
        f.push_back(id[x]);
        siz.push_back(1), rk.push_back(0);
    }
    bool same(int x, int y) {
        return find(id[x]) == find(id[y]);
    }
    int size(int x) {
        return siz[find(id[x])];
    }
};
```

#### 可撤销并查集

不能路径压缩。仅成功合并会写入历史，`get`、`same`、`merge` 的复杂度为 $\mathcal O(\log n)$，一次 `revert(t)` 的复杂度与撤销的成功合并次数成正比。

```c++
struct rDSU {
    vector<int> f, siz;
    vector<pii> his;
    int cnt;

    rDSU(int n) : f(n + 1), siz(n + 1, 1), cnt(n) {
        iota(f.begin(), f.end(), 0);
    }
    int find(int x) {
        while (x != f[x]) x = f[x];
        return x;
    }
    bool merge(int x, int y) {
        x = find(x), y = find(y);
        if (x == y) return false;
        if (siz[x] < siz[y]) swap(x, y);
        his.push_back({x, y});
        f[y] = x;
        siz[x] += siz[y];
        cnt--;
        return true;
    }
    int time() const { return his.size(); }
    void revert(int t) {
        while (time() > t) {
            auto [x, y] = his.back();
            his.pop_back();
            f[y] = y;
            siz[x] -= siz[y];
            cnt++;
        }
    }
    bool same(int x, int y) { return find(x) == find(y); }
    int size(int x) { return siz[find(x)]; }
};
```

### 单调栈

求每个位置左侧最近的严格更小元素、右侧最近的小于等于元素，不存在时分别返回 $0$、$n+1$。用于统计最小值贡献时，一侧严格、一侧非严格可避免重复值重复计数；求最大值时反转比较方向。

```cpp
auto monoStack(const vector<int> &a) {
    int n = a.size() - 1;
    vector<int> l(n + 1), r(n + 1, n + 1), st;
    for (int i = 1; i <= n; ++i) {
        while (st.size() && a[st.back()] >= a[i]) st.pop_back();
        l[i] = st.empty() ? 0 : st.back();
        st.push_back(i);
    }
    st.clear();
    for (int i = n; i >= 1; --i) {
        while (st.size() && a[st.back()] > a[i]) st.pop_back();
        r[i] = st.empty() ? n + 1 : st.back();
        st.push_back(i);
    }
    return pair{l, r};
}
```

### 动态 Bitset

长度在运行时确定，位编号为 $0\ldots n-1$；`resize(n,true)` 将新增位设为 $1$，按位二元运算要求长度相同，移位数非负。`next(p)`、`prev(p)` 分别找不小于、不大于 $p$ 的置位位置，不存在返回 $n$；底层用 `u64` 存储完整的 $64$ 位，末块多余位始终清零。

单点操作为 $\mathcal O(1)$，整体位运算、移位及前后查找最坏为 $\mathcal O(\lceil n/64\rceil)$，`each` 枚举 $k$ 个置位的时间为 $\mathcal O(\lceil n/64\rceil+k)$。

```cpp
struct Bitset {
    int n = 0;
    vector<u64> a;
    Bitset(int n = 0, bool v = false) { resize(n, v); }
    void trim() {
        if (n % 64) a.back() &= (1ULL << (n % 64)) - 1;
    }
    void resize(int m, bool v = false) {
        a.resize((m + 63) / 64, v ? ~0ULL : 0);
        if (v && m > n && n % 64) a[n / 64] |= ~0ULL << (n % 64);
        n = m;
        trim();
    }
    bool test(int p) const { return a[p / 64] >> (p % 64) & 1; }
    void set(int p, bool v = true) {
        u64 bit = 1ULL << (p % 64);
        if (v) a[p / 64] |= bit;
        else a[p / 64] &= ~bit;
    }
    void set() { fill(a.begin(), a.end(), ~0ULL); trim(); }
    void reset() { fill(a.begin(), a.end(), 0); }
    void flip(int p) { a[p / 64] ^= 1ULL << (p % 64); }
    void flip() {
        for (auto &x : a) x = ~x;
        trim();
    }
    int count() const {
        int ans = 0;
        for (u64 x : a) ans += __builtin_popcountll(x);
        return ans;
    }
    bool any() const {
        for (u64 x : a) if (x) return true;
        return false;
    }
    bool all() const { return count() == n; }
    Bitset &operator&=(const Bitset &b) {
        for (int i = 0; i < a.size(); ++i) a[i] &= b.a[i];
        return *this;
    }
    Bitset &operator|=(const Bitset &b) {
        for (int i = 0; i < a.size(); ++i) a[i] |= b.a[i];
        return *this;
    }
    Bitset &operator^=(const Bitset &b) {
        for (int i = 0; i < a.size(); ++i) a[i] ^= b.a[i];
        return *this;
    }
    Bitset &operator<<=(int s) {
        if (s >= n) return reset(), *this;
        int w = s / 64, b = s % 64, m = a.size();
        for (int i = m - 1; i >= 0; --i) {
            u64 x = i >= w ? a[i - w] << b : 0;
            if (b && i > w) x |= a[i - w - 1] >> (64 - b);
            a[i] = x;
        }
        trim();
        return *this;
    }
    Bitset &operator>>=(int s) {
        if (s >= n) return reset(), *this;
        int w = s / 64, b = s % 64, m = a.size();
        for (int i = 0; i < m; ++i) {
            u64 x = i + w < m ? a[i + w] >> b : 0;
            if (b && i + w + 1 < m) x |= a[i + w + 1] << (64 - b);
            a[i] = x;
        }
        return *this;
    }
    friend Bitset operator&(Bitset a, const Bitset &b) { return a &= b; }
    friend Bitset operator|(Bitset a, const Bitset &b) { return a |= b; }
    friend Bitset operator^(Bitset a, const Bitset &b) { return a ^= b; }
    friend Bitset operator<<(Bitset a, int s) { return a <<= s; }
    friend Bitset operator>>(Bitset a, int s) { return a >>= s; }
    Bitset operator~() const {
        Bitset b = *this;
        b.flip();
        return b;
    }
    int next(int p = 0) const {
        p = max(p, 0LL);
        if (p >= n) return n;
        int k = p / 64;
        u64 x = a[k] & (~0ULL << (p % 64));
        while (!x && ++k < a.size()) x = a[k];
        return x ? k * 64 + __builtin_ctzll(x) : n;
    }
    int prev(int p) const {
        p = min(p, n - 1);
        if (p < 0) return n;
        int k = p / 64;
        u64 x = a[k] & (~0ULL >> (63 - p % 64));
        while (!x && --k >= 0) x = a[k];
        return x ? k * 64 + 63 - __builtin_clzll(x) : n;
    }
    template<class F> void each(F f) const {
        for (int i = 0; i < a.size(); ++i) {
            for (u64 x = a[i]; x; x &= x - 1) {
                f(i * 64 + __builtin_ctzll(x));
            }
        }
    }
};
```

### 静态区间查询

#### ST 表

用于静态数组的幂等区间查询（$x \operatorname{ Op } x=x$），**不支持修改操作**。空间与初始化复杂度 $\mathcal O(n \log n)$，单次操作复杂度 $\mathcal O(1)$。

```cpp
template <class Op> struct ST {
    int n;
    vector<vector<int>> st;

    ST(const vector<int> &a) : n(a.size() - 1) {
        st.push_back(a);
        for (int i = 1; (1 << i) <= n; ++i) {
            int l = 1 << i, h = l >> 1;
            st.emplace_back(n - l + 2);
            for (int j = 1; j + l - 1 <= n; ++j) {
                st[i][j] = Op{}(st[i - 1][j], st[i - 1][j + h]);
            }
        }
    }
    int ask(int l, int r) const {
        int k = __lg(r - l + 1);
        return Op{}(st[k][l], st[k][r - (1 << k) + 1]);
    }
};
```

#### 可结合区间查询（Disjoint Sparse Table）

静态区间运算只需满足结合律，不要求幂等、交换律或单位元；合并时保持原数组顺序。将每块中点两侧分别预处理后缀和前缀，查询由两端最高不同位确定分界；若一次合并为 $\mathcal O(1)$，则预处理与空间为 $\mathcal O(n\log n)$，查询为 $\mathcal O(1)$。

```cpp
template<class T, class Op> struct DST {
    vector<T> a;
    vector<vector<T>> st;

    DST(const vector<T> &v) : a(v.begin() + 1, v.end()) {
        int n = a.size(), lg = n <= 1 ? 0 : __lg(n - 1) + 1;
        st.assign(lg, a);
        for (int k = 0; k < lg; ++k) {
            int h = 1LL << k;
            for (int l = 0; l < n; l += h * 2) {
                int m = min(l + h, n), r = min(l + h * 2, n);
                for (int i = m - 2; i >= l; --i) {
                    st[k][i] = Op{}(a[i], st[k][i + 1]);
                }
                for (int i = m + 1; i < r; ++i) {
                    st[k][i] = Op{}(st[k][i - 1], a[i]);
                }
            }
        }
    }
    T ask(int l, int r) const {
        --l, --r;
        if (l == r) return a[l];
        int k = __lg(l ^ r);
        return Op{}(st[k][l], st[k][r]);
    }
};
```

#### 基于状压的线性 RMQ 算法

用于静态最小值或最大值查询，`Op{}(x,y)` 返回两者中更优的原值；本模板保留 $0$ 起始下标，查询为非空半开区间 $[l,r)$。每 $64$ 个元素压成一块，块内维护单调栈，块间使用 ST 表；预处理与空间为 $\mathcal O(n+\lceil n/64\rceil\log(\lceil n/64\rceil+1))$，单次查询为 $\mathcal O(1)$。

```c++
template<class Op> struct RMQ {
    static constexpr int B = 64;
    using u64 = unsigned long long;
    int n;
    vector<vector<int>> a;
    vector<int> pre, suf, ini;
    vector<u64> stk;
    RMQ() {}
    RMQ(const vector<int> &v) {
        init(v);
    }
    void init(const vector<int> &v) {
        n = v.size();
        pre = suf = ini = v;
        stk.resize(n);
        if (!n) {
            return;
        }
        const int M = (n - 1) / B + 1;
        const int lg = __lg(M);
        a.assign(lg + 1, vector<int>(M));
        for (int i = 0; i < M; ++i) {
            a[0][i] = v[i * B];
            for (int j = 1; j < B && i * B + j < n; ++j) {
                a[0][i] = Op{}(a[0][i], v[i * B + j]);
            }
        }
        for (int i = 1; i < n; ++i) {
            if (i % B) {
                pre[i] = Op{}(pre[i - 1], pre[i]);
            }
        }
        for (int i = n - 2; i >= 0; i--) {
            if (i % B != B - 1) {
                suf[i] = Op{}(suf[i], suf[i + 1]);
            }
        }
        for (int j = 0; j < lg; ++j) {
            for (int i = 0; i + (2 << j) <= M; ++i) {
                a[j + 1][i] = Op{}(a[j][i], a[j][i + (1 << j)]);
            }
        }
        for (int i = 0; i < M; ++i) {
            const int l = i * B;
            const int r = min(n, l + B);
            u64 s = 0;
            for (int j = l; j < r; ++j) {
                while (s && Op{}(v[j], v[__lg(s) + l]) == v[j]) {
                    s ^= 1ULL << __lg(s);
                }
                s |= 1ULL << (j - l);
                stk[j] = s;
            }
        }
    }
    int operator()(int l, int r) const {
        if (l / B != (r - 1) / B) {
            int ans = Op{}(suf[l], pre[r - 1]);
            l = l / B + 1;
            r = r / B;
            if (l < r) {
                int k = __lg(r - l);
                ans = Op{}(ans, Op{}(a[k][l], a[k][r - (1 << k)]));
            }
            return ans;
        } else {
            int x = B * (l / B);
            return ini[__builtin_ctzll(stk[r - 1] >> (l - x)) + l];
        }
    }
};
```

### 树状数组

#### 一维树状数组

##### 常用封装（单点加、单点/区间和）

空间复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$。

```c++
struct BIT {
    int n;
    vector<int> w;
    BIT(int n) : n(n), w(n + 1) {}

    void add(int x, int v) {
        for (; x <= n; x += x & -x) w[x] += v;
    }

    int ask(int l, int r) {
        auto sum = [&](int x) {
            int ans = 0;
            for (; x; x -= x & -x) ans += w[x];
            return ans;
        };
        return sum(r) - sum(l - 1);
    }
};
```

##### 扩展封装（单点/区间加、单点/区间和）

常数比上一个版本略微大一些。

```c++
struct exBIT {
    int n;
    vector<int> a, b;
    exBIT(int n) : n(n), a(n + 1), b(n + 1) {}

    void add(int l, int r, int v) {
        auto Add = [&](int x, int v) {
            for (int i = x; i <= n; i += i & -i) {
                a[i] += v;
                b[i] += x * v;
            }
        };
        Add(l, v), Add(r + 1, -v);
    }

    int ask(int l, int r) {
        auto sum = [&](int x) {
            int ans = 0;
            for (int i = x; i; i -= i & -i) ans += (x + 1) * a[i] - b[i];
            return ans;
        };
        return sum(r) - sum(l - 1);
    }
};
```

##### 值域封装（排名、第 k 小与前后继）

适用于值域有界或可离线离散化，且需要排名或第 $k$ 小；仅需插入、删除、前后继时，直接使用 `multiset`。

构造函数传入待维护值的闭区间 $[\text{lo}, \text{hi}]$，所有操作值必须落在其中；**离线时优先离散化**；值域过大且无法离线离散化时，使用平衡树替代。

注意排名的下标是从 $1$ 开始的。删除元素、查找前驱与后继时，应当自行保证目标元素存在，否则会出错。

```c++
struct rBIT {
    int n, lo;
    vector<int> w;
    rBIT(int lo, int hi) : lo(lo), n(hi - lo + 1), w(n + 1) {}

    int id(int x) const {
        return x - lo + 1;
    }
    void add(int x, int v) {
        for (int p = id(x); p <= n; p += p & -p) w[p] += v;
    }
    int sum(int p) const {
        int ans = 0;
        for (; p; p -= p & -p) ans += w[p];
        return ans;
    }

    int kth(int k) const {
        int ans = 0;
        for (int i = 1 << __lg(n); i; i >>= 1) {
            int nxt = ans + i;
            if (nxt <= n && w[nxt] < k) {
                k -= w[nxt];
                ans = nxt;
            }
        }
        return lo + ans;
    }
    int rank(int x) const { return sum(id(x) - 1) + 1; }
    int prec(int x) const { return kth(rank(x) - 1); }
    int sufc(int x) const { return kth(sum(id(x)) + 1); }
};
```

```c++
                for (int j = v; j <= m; j += j & -j) {
                    b1[i][j] += p;
                    b2[i][j] += p * (u - 1);
                    b3[i][j] += p * (v - 1);

                }

        Add(x, y, p), Add(u, y, -p);

                for (int j = v; j; j -= j & -j) {

                    ans -= b2[i][j] * v;
                    ans -= b3[i][j] * u;
                    ans += b4[i][j];
                }
            }
            return ans;
        };
```

```cpp
struct SegTree {
    int n;
    vector<Info> tr;

    SegTree(int n) : n(n), tr(4 * n + 5) {}

    template <class A> void build(int p, int l, int r, const A &a) {
        if (l == r) return tr[p] = a[l], void();

        int m = (l + r) >> 1;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        tr[p] = tr[p << 1] + tr[p << 1 | 1];
    }

    template <class A> void build(const A &a) {
        build(1, 1, n, a);
    }

    void modify(int p, int l, int r, int pos, const Info &v) {
        if (l == r) return tr[p] = v, void();

        int m = (l + r) >> 1;
        if (pos <= m)
            modify(p << 1, l, m, pos, v);
        else
            modify(p << 1 | 1, m + 1, r, pos, v);
        tr[p] = tr[p << 1] + tr[p << 1 | 1];
    }

    void modify(int pos, const Info &v) {
        modify(1, 1, n, pos, v);
    }

    Info ask(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return tr[p];
        int m = (l + r) >> 1;
        if (qr <= m) return ask(p << 1, l, m, ql, qr);
        if (ql > m) return ask(p << 1 | 1, m + 1, r, ql, qr);
        return ask(p << 1, l, m, ql, qr) + ask(p << 1 | 1, m + 1, r, ql, qr);
    }

    Info ask(int l, int r) {
        if (l > r) return Info{};
        return ask(1, 1, n, l, r);
    }
};
```

#### 迭代线段树

适用于区间信息能由左右子区间合并、修改只落在单个位置的题目。空间与初始化复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$。

使用 zkw 风格，自底向上迭代封装，常数更优。

重载 `Info + Info` 区间合并。

```c++
struct SegTree {
    int n = 0;
    vector<Info> t;

    SegTree(const vector<Info> &a) {
        n = a.size() - 1;
        t.assign(2 * n, Info());
        for (int i = 1; i <= n; ++i) {
            t[n + i - 1] = a[i];
        }
        for (int p = n - 1; p; p--) {
            t[p] = t[p << 1] + t[p << 1 | 1];
        }
    }
    void modify(int x, const Info &v) {
        t[x += n - 1] = v;
        for (x >>= 1; x; x >>= 1) {
            t[x] = t[x << 1] + t[x << 1 | 1];
        }
    }
    Info ask(int l, int r) const {
        Info u, v;
        for (l += n - 1, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) u = u + t[l++];
            if (r & 1) v = t[--r] + v;
        }
        return u + v;
    }
};
```

#### 懒标记线段树

重载 `Tag + Tag`（懒标记传递） 、 `Info + Info`（信息合并） 、 `Info += Tag`（懒标记应用） 。

```c++
struct SegTree {
    int n;
    vector<Info> tr;
    vector<Tag> tag;
    vector<bool> has;

    SegTree(int n) : n(n), tr(4 * n + 5), tag(4 * n + 5), has(4 * n + 5) {}

    void apply(int p, const Tag &v) {
        tr[p] += v;

        if (has[p])
            tag[p] += v;
        else
            tag[p] = v, has[p] = 1;
    }

    void push(int p) {
        if (!has[p]) return;

        apply(p << 1, tag[p]);
        apply(p << 1 | 1, tag[p]);

        tag[p] = Tag{};
        has[p] = 0;
    }

    template <class A> void build(int p, int l, int r, const A &a) {
        tag[p] = Tag{};
        has[p] = 0;

        if (l == r) return tr[p] = a[l], void();

        int m = (l + r) >> 1;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        tr[p] = tr[p << 1] + tr[p << 1 | 1];
    }

    template <class A> void build(const A &a) {
        build(1, 1, n, a);
    }

    void update(int p, int l, int r, int ql, int qr, const Tag &v) {
        if (ql <= l && r <= qr) return apply(p, v), void();

        push(p);
        int m = (l + r) >> 1;

        if (ql <= m) update(p << 1, l, m, ql, qr, v);
        if (qr > m) update(p << 1 | 1, m + 1, r, ql, qr, v);

        tr[p] = tr[p << 1] + tr[p << 1 | 1];
    }

    void update(int l, int r, const Tag &v) {
        if (l > r) return;
        update(1, 1, n, l, r, v);
    }

    void modify(int p, int l, int r, int pos, const Info &v) {
        if (l == r) return tr[p] = v, tag[p] = Tag{}, has[p] = 0, void();

        push(p);
        int m = (l + r) >> 1;

        if (pos <= m)
            modify(p << 1, l, m, pos, v);
        else
            modify(p << 1 | 1, m + 1, r, pos, v);

        tr[p] = tr[p << 1] + tr[p << 1 | 1];
    }

    Info ask(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return tr[p];

        push(p);
        int m = (l + r) >> 1;

        if (qr <= m) return ask(p << 1, l, m, ql, qr);
        if (ql > m) return ask(p << 1 | 1, m + 1, r, ql, qr);

        return ask(p << 1, l, m, ql, qr) + ask(p << 1 | 1, m + 1, r, ql, qr);
    }

    void modify(int pos, const Info &v) {
        modify(1, 1, n, pos, v);
    }

    Info ask(int l, int r) {
        if (l > r) return Info{};
        return ask(1, 1, n, l, r);
    }
};
```

```c++
            w[i] = x >> i & 1;
```

#### 动态开点线段树

在整数坐标范围 $[\text{lo},\text{hi}]$ 上维护区间加、区间和，初值全为 $0$，坐标跨度及区间和须能用 `int` 表示。标记留在本节点，查询携带祖先标记，因而查询不开新点；设 $V=\text{hi}-\text{lo}+1$，单次操作为 $\mathcal O(\log(V+1))$，$q$ 次修改至多占用 $\mathcal O(q\log(V+1))$ 空间。

```cpp
struct SegTree {
    struct node {
        int l = 0, r = 0, sum = 0, tag = 0;
    };
    int lo, hi, rt = 0;
    vector<node> t = {{}};
    SegTree(int lo, int hi) : lo(lo), hi(hi) {}

    int update(int p, int l, int r, int x, int y, int v) {
        if (!p) {
            p = t.size();
            t.push_back({});
        }
        if (x <= l && r <= y) {
            t[p].sum += (r - l + 1) * v;
            t[p].tag += v;
            return p;
        }
        int m = l + (r - l) / 2;
        if (x <= m) t[p].l = update(t[p].l, l, m, x, y, v);
        if (m < y) t[p].r = update(t[p].r, m + 1, r, x, y, v);
        t[p].sum = t[t[p].l].sum + t[t[p].r].sum + (r - l + 1) * t[p].tag;
        return p;
    }
    int ask(int p, int l, int r, int x, int y, int v) const {
        if (!p) return (min(r, y) - max(l, x) + 1) * v;
        if (x <= l && r <= y) return t[p].sum + (r - l + 1) * v;
        int m = l + (r - l) / 2, ans = 0;
        v += t[p].tag;
        if (x <= m) ans += ask(t[p].l, l, m, x, y, v);
        if (m < y) ans += ask(t[p].r, m + 1, r, x, y, v);
        return ans;
    }
    void update(int l, int r, int v) { rt = update(rt, lo, hi, l, r, v); }
    int ask(int l, int r) const { return ask(rt, lo, hi, l, r, 0); }
};
```

#### 区间取模/取整开方、单点赋值、区间和（完整封装）

本质是势能剪枝，无法仅由区间和直接计算取模后的区间和。额外维护区间最大值：
- 取模时，若区间最大值小于模数，整段不变；两次赋值之间，单点真正取模至多 $\mathcal O(\log(A+1))$ 次。
- 开方时，若区间最大值不超过 $1$，整段不变；两次赋值之间，单点真正开方至多 $\mathcal O(\log\log(A+2))$ 次。

其中，$A$ 为初值及所有赋入值的最大值。要求所有元素始终非负，且每次模数 $v > 0$。

总均摊复杂度为 $\mathcal O((n+q)\log n\log(A+2))$。

```c++
template<class T> T isqrt(T n) { // n >= 0
    T x = sqrt(n); // 兼容 __int128 需要更换为 sqrtl
    while (x + 1 <= n / (x + 1)) ++x;
    while (x && x > n / x) --x;
    return x;
}

struct SegTree {
    int n;
    vector<int> sum, mx;

    SegTree(const vector<int> &a) : n(a.size() - 1), sum(4 * n + 4), mx(4 * n + 4) {
        build(1, 1, n, a);
    }

    void modify(int x, int v) { modify(1, 1, n, x, v); }
    void mod(int l, int r, int v) { mod(1, 1, n, l, r, v); }
    void sqrt(int l, int r) { sqrt(1, 1, n, l, r); }
    int ask(int l, int r) { return ask(1, 1, n, l, r); }

private:
    void build(int p, int l, int r, const vector<int> &a) {
        if (l == r) return sum[p] = mx[p] = a[l], void();
        int m = (l + r) / 2;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        pull(p);
    }
    void pull(int p) {
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
        mx[p] = max(mx[p << 1], mx[p << 1 | 1]);
    }
    void modify(int p, int l, int r, int x, int v) {
        if (l == r) return sum[p] = mx[p] = v, void();
        int m = (l + r) / 2;
        if (x <= m) modify(p << 1, l, m, x, v);
        else modify(p << 1 | 1, m + 1, r, x, v);
        pull(p);
    }
    void mod(int p, int l, int r, int x, int y, int v) {
        if (mx[p] < v) return; // 整段取模后不变
        if (l == r) return sum[p] %= v, mx[p] = sum[p], void();
        int m = (l + r) / 2;
        if (x <= m) mod(p << 1, l, m, x, y, v);
        if (m < y) mod(p << 1 | 1, m + 1, r, x, y, v);
        pull(p);
    }
    void sqrt(int p, int l, int r, int x, int y) {
        if (mx[p] <= 1) return; // 0、1 开方后不变
        if (l == r) return sum[p] = mx[p] = isqrt(sum[p]), void();
        int m = (l + r) / 2;
        if (x <= m) rangeSqrt(p << 1, l, m, x, y);
        if (m < y) rangeSqrt(p << 1 | 1, m + 1, r, x, y);
        pull(p);
    }
    int rangeSum(int p, int l, int r, int x, int y) {
        if (x <= l && r <= y) return sum[p];
        int m = (l + r) / 2;
        int ans = 0;
        if (x <= m) ans += rangeSum(p << 1, l, m, x, y);
        if (m < y) ans += rangeSum(p << 1 | 1, m + 1, r, x, y);
        return ans;
    }
};
```

#### 区间加、单点赋值、区间 gcd 查询

本质是维护差分前缀和，令 $d_1=a_1$，$d_i=a_i-a_{i-1}$。区间加 $l,r$ 加上 $v$ 时，只会改变两个位置：$d_l \to d_l+v$、$d_{r+1} \to d_{r+1}-v \left(r<n \right)$。由定义：

$$
\gcd(a_l,a_{l+1},\ldots,a_r)
=\gcd(a_l,d_{l+1},d_{l+2},\ldots,d_r)
$$

故树状数组维护差分前缀和，得到 $a_l$；线段树维护差分绝对值的 $\gcd$。数值、差分及其绝对值须能用 `int` 表示。

空间与初始化复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$。

```c++
struct RangeAddGcd {
    int n;
    vector<int> d, bit, seg;

    RangeAddGcd(const vector<int> &a) : n(a.size() - 1), d(n + 1), bit(n + 1), seg(2 * n) {
        for (int i = 1; i <= n; ++i) {
            d[i] = a[i] - (i == 1 ? 0 : a[i - 1]);
            bit[i] += d[i];
            int j = i + (i & -i);
            if (j <= n) bit[j] += bit[i];
            seg[n + i - 1] = gcd(d[i], 0LL);
        }
        for (int p = n - 1; p; p--) {
            seg[p] = gcd(seg[p << 1], seg[p << 1 | 1]);
        }
    }

    void rangeAdd(int l, int r, int v) { // a[l..r] += v
        changeD(l, v);
        changeD(r + 1, -v);
    }
    void pointSet(int x, int v) { // a[x] = v
        rangeAdd(x, x, v - pointValue(x));
    }
    int pointValue(int x) const { return prefixSum(x); }
    int rangeGcd(int l, int r) const {
        return gcd(pointValue(l), rangeGcdDiff(l + 1, r));
    }

private:
    void bitAdd(int x, int v) {
        for (; x <= n; x += x & -x) bit[x] += v;
    }
    int prefixSum(int x) const {
        int ans = 0;
        for (; x; x -= x & -x) ans += bit[x];
        return ans;
    }
    void changeD(int x, int v) { // d[x] += v
        if (x > n) return;
        d[x] += v;
        bitAdd(x, v);

        int p = n + x - 1;
        seg[p] = gcd(d[x], 0LL);
        for (p >>= 1; p; p >>= 1) {
            seg[p] = gcd(seg[p << 1], seg[p << 1 | 1]);
        }
    }
    int rangeGcdDiff(int l, int r) const {
        if (l > r) return 0;
        int ans = 0;
        for (l += n - 1, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) ans = gcd(ans, seg[l++]);
            if (r & 1) ans = gcd(ans, seg[--r]);
        }
        return ans;
    }
};
```

#### Segment Tree Beats

支持区间加、对每个元素取 $\min(a_i,v)$ 或 $\max(a_i,v)$，以及区间和、最小值、最大值。以取 $\min$ 为例：若最大值不超过 $v$ 则跳过；只有严格次大值小于 $v$ 时，才能仅降低最大值并按其个数更新区间和，否则继续递归。

下传时先处理加法，再用父节点的最值约束子节点；所有元素与阈值须严格落在 $(-\text{inf},\text{inf})$ 内，区间和与中间运算不能溢出。初始化与空间为 $\mathcal O(n)$，$q$ 次混合操作的总复杂度为 $\mathcal O((n+q)\log^2 n)$，不能将每次修改都视为最坏 $\mathcal O(\log n)$。

```cpp
struct SegTree {
    static constexpr int inf = (1LL << 60);
    struct node {
        int sum = 0, mx = -inf, se = -inf, cmx = 0;
        int mn = inf, ne = inf, cmn = 0, tag = 0;
    };
    int n;
    vector<node> t;
    SegTree(const vector<int> &a) : n(a.size() - 1), t(n * 4 + 4) {
        auto build = [&](auto &&self, int p, int l, int r) -> void {
            if (l == r) {
                t[p] = {a[l], a[l], -inf, 1, a[l], inf, 1, 0};
                return;
            }
            int m = (l + r) / 2;
            self(self, p << 1, l, m);
            self(self, p << 1 | 1, m + 1, r);
            pull(p);
        };
        build(build, 1, 1, n);
    }
    void pull(int p) {
        auto &u = t[p];
        auto &a = t[p << 1], &b = t[p << 1 | 1];
        u.sum = a.sum + b.sum;
        u.mx = max(a.mx, b.mx), u.mn = min(a.mn, b.mn);
        u.se = -inf, u.ne = inf, u.cmx = u.cmn = 0;
        for (int k : {p << 1, p << 1 | 1}) {
            auto &v = t[k];
            if (v.mx == u.mx) u.cmx += v.cmx, u.se = max(u.se, v.se);
            else u.se = max(u.se, v.mx);
            if (v.mn == u.mn) u.cmn += v.cmn, u.ne = min(u.ne, v.ne);
            else u.ne = min(u.ne, v.mn);
        }
    }
    void add(int p, int len, int v) {
        auto &u = t[p];
        u.sum += len * v, u.tag += v;
        u.mx += v, u.mn += v;
        if (u.se != -inf) u.se += v;
        if (u.ne != inf) u.ne += v;
    }
    void cap(int p, int v) { // se < v < mx
        auto &u = t[p];
        u.sum += (v - u.mx) * u.cmx;
        if (u.mn == u.mx) u.mn = v;
        else if (u.ne == u.mx) u.ne = v;
        u.mx = v;
    }
    void floor(int p, int v) { // mn < v < ne
        auto &u = t[p];
        u.sum += (v - u.mn) * u.cmn;
        if (u.mx == u.mn) u.mx = v;
        else if (u.se == u.mn) u.se = v;
        u.mn = v;
    }
    void push(int p, int l, int r) {
        int m = (l + r) / 2;
        if (t[p].tag) {
            add(p << 1, m - l + 1, t[p].tag);
            add(p << 1 | 1, r - m, t[p].tag);
            t[p].tag = 0;
        }
        for (int k : {p << 1, p << 1 | 1}) {
            if (t[k].mx > t[p].mx) cap(k, t[p].mx);
            if (t[k].mn < t[p].mn) floor(k, t[p].mn);
        }
    }
    void update(int p, int l, int r, int x, int y, int v, int op) {
        if (op == 1 && t[p].mx <= v) return;
        if (op == 2 && t[p].mn >= v) return;
        if (x <= l && r <= y) {
            if (op == 0) return add(p, r - l + 1, v);
            if (op == 1 && t[p].se < v) return cap(p, v);
            if (op == 2 && t[p].ne > v) return floor(p, v);
        }
        push(p, l, r);
        int m = (l + r) / 2;
        if (x <= m) update(p << 1, l, m, x, y, v, op);
        if (m < y) update(p << 1 | 1, m + 1, r, x, y, v, op);
        pull(p);
    }
    int ask(int p, int l, int r, int x, int y, int op) {
        if (x <= l && r <= y) {
            return op == 0 ? t[p].sum : (op == 1 ? t[p].mn : t[p].mx);
        }
        push(p, l, r);
        int m = (l + r) / 2;
        if (y <= m) return ask(p << 1, l, m, x, y, op);
        if (m < x) return ask(p << 1 | 1, m + 1, r, x, y, op);
        int a = ask(p << 1, l, m, x, y, op);
        int b = ask(p << 1 | 1, m + 1, r, x, y, op);
        return op == 0 ? a + b : (op == 1 ? min(a, b) : max(a, b));
    }
    void update(int l, int r, int v) { update(1, 1, n, l, r, v, 0); }
    void chmin(int l, int r, int v) { update(1, 1, n, l, r, v, 1); }
    void chmax(int l, int r, int v) { update(1, 1, n, l, r, v, 2); }
    int askSum(int l, int r) { return ask(1, 1, n, l, r, 0); }
    int askMin(int l, int r) { return ask(1, 1, n, l, r, 1); }
    int askMax(int l, int r) { return ask(1, 1, n, l, r, 2); }
};
```

### 李超线段树

在整数横坐标范围 $[\text{lo},\text{hi}]$ 上插入直线 $y=kx+b$ 或仅在子区间有效的线段，查询给定横坐标处的最小值，无有效直线时返回 `inf`。每个节点保留中点处更优的直线，另一条至多在一侧更优；直线插入、查询为 $\mathcal O(\log(V+1))$，线段插入为 $\mathcal O(\log^2(V+1))$，$q$ 次插入的空间上界为 $\mathcal O(q\log(V+1))$。

这里 $V=\text{hi}-\text{lo}+1$，坐标跨度须能用 `int` 表示，线段区间和查询横坐标须在定义域内；函数值用 `i128` 计算和返回且须在 $(-\text{inf},\text{inf})$ 内。取最大值可将 $k,b$ 取反后查询并取反，注意先判断空结果。

```cpp
struct LiChao {
    static constexpr i128 inf = i128(1) << 120;
    struct Line {
        int k = 0, b = 0;
        i128 at(int x) const { return i128(k) * x + b; }
    };
    struct node {
        Line f;
        int l = 0, r = 0;
        bool has = false;
    };
    int lo, hi, rt = 0;
    vector<node> t = {{}};
    LiChao(int lo, int hi) : lo(lo), hi(hi) {}
    int make() {
        t.push_back({});
        return t.size() - 1;
    }
    int insert(int p, int l, int r, Line f) {
        if (!p) p = make();
        if (!t[p].has) {
            t[p].f = f, t[p].has = true;
            return p;
        }
        int m = l + (r - l) / 2;
        if (f.at(m) < t[p].f.at(m)) swap(f, t[p].f);
        if (l == r) return p;
        if (f.at(l) < t[p].f.at(l)) t[p].l = insert(t[p].l, l, m, f);
        else if (f.at(r) < t[p].f.at(r)) t[p].r = insert(t[p].r, m + 1, r, f);
        return p;
    }
    int insert(int p, int l, int r, int x, int y, Line f) {
        if (x <= l && r <= y) return insert(p, l, r, f);
        if (!p) p = make();
        int m = l + (r - l) / 2;
        if (x <= m) t[p].l = insert(t[p].l, l, m, x, y, f);
        if (m < y) t[p].r = insert(t[p].r, m + 1, r, x, y, f);
        return p;
    }
    i128 ask(int p, int l, int r, int x) const {
        if (!p) return inf;
        i128 ans = t[p].has ? t[p].f.at(x) : inf;
        if (l == r) return ans;
        int m = l + (r - l) / 2;
        if (x <= m) return min(ans, ask(t[p].l, l, m, x));
        return min(ans, ask(t[p].r, m + 1, r, x));
    }
    void add(int k, int b) { rt = insert(rt, lo, hi, {k, b}); }
    void addSeg(int k, int b, int l, int r) { rt = insert(rt, lo, hi, l, r, {k, b}); }
    i128 ask(int x) const { return ask(rt, lo, hi, x); }
};
```

### 主席树（可持久化线段树）

维护前缀版本，在第 $l\ldots r$ 个插入元素中查询值域计数、权值和及第 $k$ 小；`add(x,v)` 以 $x$ 为坐标、$v$ 为权重，`add(x)` 的权重就是 $x$。相减顺序是 `root[r] - root[l-1]`，第 $k$ 小要求 $1\le k\le r-l+1$；离散化时传排名作坐标、原值作权重，`kth` 返回的坐标要映射回原值。

构造时指定坐标范围 $[\text{lo},\text{hi}]$，跨度须能用 `int` 表示；每次插入仅复制根到叶的路径，设 $V=\text{hi}-\text{lo}+1$，插入和查询为 $\mathcal O(\log(V+1))$，$n$ 个版本的空间为 $\mathcal O(n\log(V+1))$。底层 `modify` 返回新根，不改变旧根；任意两版本做第 $k$ 小时，必须保证相减后的每个坐标计数非负。

```cpp
struct PresidentTree {
    struct Info {
        int cnt = 0, sum = 0;
        Info operator+(const Info &b) const { return {cnt + b.cnt, sum + b.sum}; }
    };
    struct node {
        int l = 0, r = 0, cnt = 0, sum = 0;
    };
    int lo, hi;
    vector<node> t = {{}};
    vector<int> root = {0};
    PresidentTree(int lo, int hi) : lo(lo), hi(hi) {}

    int modify(int p, int l, int r, int x, int v) {
        int k = t.size();
        node u = t[p];
        t.push_back(u);
        ++t[k].cnt, t[k].sum += v;
        if (l == r) return k;
        int m = l + (r - l) / 2;
        if (x <= m) t[k].l = modify(t[k].l, l, m, x, v);
        else t[k].r = modify(t[k].r, m + 1, r, x, v);
        return k;
    }
    void add(int x, int v) {
        int p = modify(root.back(), lo, hi, x, v);
        root.push_back(p);
    }
    void add(int x) { add(x, x); }
    Info ask(int u, int v, int l, int r, int x, int y) const {
        if (y < l || r < x) return {};
        if (x <= l && r <= y) return {t[v].cnt - t[u].cnt, t[v].sum - t[u].sum};
        int m = l + (r - l) / 2;
        return ask(t[u].l, t[v].l, l, m, x, y)
             + ask(t[u].r, t[v].r, m + 1, r, x, y);
    }
    int kth(int u, int v, int l, int r, int k) const {
        if (l == r) return l;
        int cnt = t[t[v].l].cnt - t[t[u].l].cnt;
        int m = l + (r - l) / 2;
        if (k <= cnt) return kth(t[u].l, t[v].l, l, m, k);
        return kth(t[u].r, t[v].r, m + 1, r, k - cnt);
    }
    Info ask(int l, int r, int x, int y) const {
        if (l > r || x > y) return {};
        return ask(root[l - 1], root[r], lo, hi, x, y);
    }
    int kth(int l, int r, int k) const {
        return kth(root[l - 1], root[r], lo, hi, k);
    }
};
```

### 小波矩阵

#### 常用版（静态区间顺序统计）

将离散化后的值按二进制逐层稳定划分，支持第 $k$ 小/大、阈值计数、值域计数和单值频次。`countLess(l,r,x)` 统计小于 $x$ 的个数，`countLE` 统计小于等于，`count(l,r,x,y)` 统计值域 $[x,y]$ 内的个数，`freq(l,r,x)` 统计 $x$ 出现次数；第 $k$ 小/大要求 $1\le k\le r-l+1$。

设不同值个数为 $\sigma$，初始化为 $\mathcal O(n\log n)$，空间为 $\mathcal O(n\log(\sigma+1))$，单次查询为 $\mathcal O(\log(\sigma+1))$。仅需顺序统计且空间紧张时，可选后面的压位版本。

```c++
struct WaveletMatrix {
    int lg;
    vector<int> val;
    vector<int> mid;
    vector<vector<int>> z; // 每层的 0 前缀和

    WaveletMatrix(const vector<int> &a) {
        int n = a.size() - 1;
        val.assign(a.begin() + 1, a.end());
        sort(val.begin(), val.end());
        val.erase(unique(val.begin(), val.end()), val.end());

        int m = val.size();
        lg = m <= 1 ? 1 : __lg(m - 1) + 1;

        vector<int> cur(n + 1), nxt(n + 1);
        for (int i = 1; i <= n; ++i) {
            cur[i] = lower_bound(val.begin(), val.end(), a[i]) - val.begin();
        }

        mid.resize(lg);
        z.assign(lg, vector<int>(n + 1));
        for (int d = 0; d < lg; ++d) {
            int pos = lg - 1 - d;
            for (int i = 1; i <= n; ++i) {
                z[d][i] = z[d][i - 1] + ((cur[i] >> pos & 1) == 0);
            }

            mid[d] = z[d][n];
            int p0 = 0, p1 = mid[d];
            for (int i = 1; i <= n; ++i) {
                if (cur[i] >> pos & 1) nxt[++p1] = cur[i];
                else nxt[++p0] = cur[i];
            }
            cur.swap(nxt);
        }
    }

    int kth(int L, int R, int k) const {
        int ans = 0;
        for (int d = 0; d < lg; ++d) {
            int zL = z[d][L - 1], zR = z[d][R];
            if (k <= zR - zL) {
                L = zL + 1, R = zR;
            } else {
                k -= zR - zL, ans |= 1LL << (lg - 1 - d);
                L = mid[d] + L - zL, R = mid[d] + R - zR;
            }
        }
        return val[ans];
    }

    int kthLarge(int L, int R, int k) const {
        return kth(L, R, R - L + 2 - k);
    }

    int lessRank(int l, int r, int k) const { // 小于离散化编号 k 的个数
        if (l > r || k <= 0) return 0;
        if (k >= val.size()) return r - l + 1;
        int ans = 0;
        for (int d = 0; d < lg; ++d) {
            int x = z[d][l - 1], y = z[d][r];
            if (k >> (lg - 1 - d) & 1) {
                ans += y - x;
                l = mid[d] + l - x, r = mid[d] + r - y;
            } else {
                l = x + 1, r = y;
            }
        }
        return ans;
    }
    int countLess(int l, int r, int x) const {
        int k = lower_bound(val.begin(), val.end(), x) - val.begin();
        return lessRank(l, r, k);
    }
    int countLE(int l, int r, int x) const {
        int k = upper_bound(val.begin(), val.end(), x) - val.begin();
        return lessRank(l, r, k);
    }
    int count(int l, int r, int x, int y) const {
        if (x > y) return 0;
        return countLE(l, r, y) - countLess(l, r, x);
    }
    int freq(int l, int r, int x) const {
        return countLE(l, r, x) - countLess(l, r, x);
    }
};
```

#### 压位版（静态区间第 k 小/大）

手写 `bitset` 压位，以 $\mathcal O(N \log N)$ 的时间复杂度和 $\mathcal O(N + \frac{N \log N}{64})$ 的空间建树后，实现单次 $\mathcal O(\log N)$ 复杂度的区间第 $k$ 大值询问。已经过偏移，请使用 $\texttt{1-idx}$ 。

```c++
#define __count(x) __builtin_popcountll(x)
struct Wavelet {
    vector<int> val, sum;
    vector<u64> bit;
    int t, n;

    int getSum(int i) {
        return sum[i >> 6] + __count(bit[i >> 6] & ((1ULL << (i & 63)) - 1));
    }

    Wavelet(vector<int> v) : val(v), n(v.size()) {
        sort(val.begin(), val.end());
        val.erase(unique(val.begin(), val.end()), val.end());

        int n_ = val.size();
        t = __lg(2 * n_ - 1);
        bit.resize((t * n + 64) >> 6);
        sum.resize(bit.size());
        vector<int> cnt(n_ + 1);

        for (int &x : v) {
            x = lower_bound(val.begin(), val.end(), x) - val.begin();
            cnt[x + 1]++;
        }
        for (int i = 1; i < n_; ++i) {
            cnt[i] += cnt[i - 1];
        }
        for (int j = 0; j < t; ++j) {
            for (int i : v) {
                int tmp = i >> (t - 1 - j);
                int pos = (tmp >> 1) << (t - j);
                auto setBit = [&](int i, u64 v) {
                    bit[i >> 6] |= (v << (i & 63));
                };
                setBit(j * n + cnt[pos], tmp & 1);
                cnt[pos]++;
            }
            for (int i : v) {
                cnt[(i >> (t - j)) << (t - j)]--;
            }
        }
        for (int i = 1; i < sum.size(); ++i) {
            sum[i] = sum[i - 1] + __count(bit[i - 1]);
        }
    }

    int small(int l, int r, int k) {
        r++, k--;
        for (int j = 0, x = 0, y = n, res = 0;; ++j) {
            if (j == t) return val[res];
            int A = getSum(n * j + x), B = getSum(n * j + l);
            int C = getSum(n * j + r), D = getSum(n * j + y);
            int ab_zeros = r - l - C + B;
            if (ab_zeros > k) {
                res = res << 1;
                y -= D - A;
                l -= B - A;
                r -= C - A;
            } else {
                res = (res << 1) | 1;
                k -= ab_zeros;
                x += y - x - D + A;
                l += y - l - D + B;
                r += y - r - D + C;
            }
        }
    }
    int large(int l, int r, int k) {
        return small(l, r, r - l + 2 - k);
    }
};
```

#### FHQ Treap（有序多重集）

按值分裂、按随机优先级合并，支持重复值、删除、排名、第 $k$ 小和严格前后继；`rank(x)` 等于小于 $x$ 的元素个数加一，`insert/erase` 的数量 $c>0$，删除时至多删除 $c$ 个。不存在的第 $k$ 小或前后继返回 `nullopt`；各操作期望 $\mathcal O(\log(n+1))$，节点不回收，空间随累计新建节点数增长。

```cpp
struct FHQ {
    struct node {
        int l = 0, r = 0, siz = 0, cnt = 0, v = 0, pri = 0;
    };
    vector<node> t = {{}};
    int rt = 0;
    mt19937 rng{random_device{}()};

    void pull(int p) {
        t[p].siz = t[t[p].l].siz + t[t[p].r].siz + t[p].cnt;
    }
    void split(int p, int v, int &x, int &y, bool eq = false) {
        if (!p) return x = y = 0, void();
        if (t[p].v < v || (eq && t[p].v == v)) {
            x = p;
            split(t[p].r, v, t[p].r, y, eq);
        } else {
            y = p;
            split(t[p].l, v, x, t[p].l, eq);
        }
        pull(p);
    }
    int merge(int x, int y) {
        if (!x || !y) return x | y;
        if (t[x].pri < t[y].pri) {
            t[x].r = merge(t[x].r, y);
            pull(x);
            return x;
        }
        t[y].l = merge(x, t[y].l);
        pull(y);
        return y;
    }
    void insert(int v, int c = 1) {
        int x, y, z;
        split(rt, v, x, y);
        split(y, v, y, z, true);
        if (y) t[y].cnt += c, pull(y);
        else {
            y = t.size();
            int pri = rng();
            t.push_back({0, 0, c, c, v, pri});
        }
        rt = merge(merge(x, y), z);
    }
    void erase(int v, int c = 1) {
        int x, y, z;
        split(rt, v, x, y);
        split(y, v, y, z, true);
        if (y) {
            if (t[y].cnt > c) t[y].cnt -= c, pull(y);
            else y = 0;
        }
        rt = merge(merge(x, y), z);
    }
    int size() const { return t[rt].siz; }
    int count(int v) const {
        int p = rt;
        while (p && t[p].v != v) p = v < t[p].v ? t[p].l : t[p].r;
        return t[p].cnt;
    }
    int rank(int v) const {
        int p = rt, ans = 1;
        while (p) {
            if (t[p].v >= v) p = t[p].l;
            else ans += t[t[p].l].siz + t[p].cnt, p = t[p].r;
        }
        return ans;
    }
    optional<int> kth(int k) const {
        if (k < 1 || k > size()) return nullopt;
        int p = rt;
        while (p) {
            int s = t[t[p].l].siz;
            if (k <= s) p = t[p].l;
            else if (k <= s + t[p].cnt) return t[p].v;
            else k -= s + t[p].cnt, p = t[p].r;
        }
        return nullopt;
    }
    optional<int> prec(int v) const { return kth(rank(v) - 1); }
    optional<int> sufc(int v) const { return kth(rank(v) + count(v)); }
};
```

#### 隐式 Treap（序列）

按前 $k$ 个元素分裂，支持位置变化的序列插入、区间删除、区间加、翻转和求和；`insert(p,v)` 在第 $p$ 个位置前插入，$p=\text{size()}+1$ 表示尾插。分裂和合并前先下传标记，各操作期望 $\mathcal O(\log(n+1))$；此处逐个插入建树为期望 $\mathcal O(n\log(n+1))$，节点删除后不回收。

```cpp
struct FHQ {
    struct node {
        int l = 0, r = 0, siz = 0, pri = 0;
        int v = 0, sum = 0, tag = 0;
        bool rev = false;
    };
    int rt = 0;
    vector<node> t = {{}};
    mt19937 rng{random_device{}()};
    FHQ() {}
    FHQ(const vector<int> &a) {
        for (int i = 1; i < a.size(); ++i) insert(i, a[i]);
    }
    int size() const { return t[rt].siz; }
    void pull(int p) {
        t[p].siz = t[t[p].l].siz + t[t[p].r].siz + 1;
        t[p].sum = t[t[p].l].sum + t[t[p].r].sum + t[p].v;
    }
    void add(int p, int v) {
        if (!p) return;
        t[p].v += v, t[p].sum += t[p].siz * v, t[p].tag += v;
    }
    void flip(int p) {
        if (!p) return;
        swap(t[p].l, t[p].r);
        t[p].rev ^= 1;
    }
    void push(int p) {
        if (t[p].tag) {
            add(t[p].l, t[p].tag), add(t[p].r, t[p].tag);
            t[p].tag = 0;
        }
        if (t[p].rev) {
            flip(t[p].l), flip(t[p].r);
            t[p].rev = false;
        }
    }
    void split(int p, int k, int &x, int &y) {
        if (!p) return x = y = 0, void();
        push(p);
        if (t[t[p].l].siz < k) {
            x = p;
            split(t[p].r, k - t[t[p].l].siz - 1, t[p].r, y);
        } else {
            y = p;
            split(t[p].l, k, x, t[p].l);
        }
        pull(p);
    }
    int merge(int x, int y) {
        if (!x || !y) return x | y;
        if (t[x].pri < t[y].pri) {
            push(x);
            t[x].r = merge(t[x].r, y);
            pull(x);
            return x;
        }
        push(y);
        t[y].l = merge(x, t[y].l);
        pull(y);
        return y;
    }
    void insert(int p, int v) {
        int x, y, k = t.size(), pri = rng();
        split(rt, p - 1, x, y);
        t.push_back({0, 0, 1, pri, v, v, 0, false});
        rt = merge(merge(x, k), y);
    }
    void erase(int l, int r) {
        int x, y, z;
        split(rt, r, y, z), split(y, l - 1, x, y);
        rt = merge(x, z);
    }
    void update(int l, int r, int v) {
        int x, y, z;
        split(rt, r, y, z), split(y, l - 1, x, y);
        add(y, v);
        rt = merge(merge(x, y), z);
    }
    void reverse(int l, int r) {
        int x, y, z;
        split(rt, r, y, z), split(y, l - 1, x, y);
        flip(y);
        rt = merge(merge(x, y), z);
    }
    int ask(int l, int r) {
        int x, y, z;
        split(rt, r, y, z), split(y, l - 1, x, y);
        int ans = t[y].sum;
        rt = merge(merge(x, y), z);
        return ans;
    }
};
```

### 可并堆（左偏树）

`Heap<T> h(n)` 维护 $n$ 个大根堆，改用 `Heap<T, greater<T>>` 得到小根堆；`merge(x,y)` 把第 $y$ 个堆并入第 $x$ 个堆并清空 $y$，`ask/pop` 要求堆非空。空节点距离为 $0$，维持左子树距离不小于右子树，保证沿右链合并只需 $\mathcal O(\log n)$；插入、弹出同阶，已弹出的节点不回收，空间随累计插入数增长。

```cpp
template<class T, class Cmp = less<T>> struct Heap {
    struct node {
        T v{};
        int l = 0, r = 0, dis = 0;
    };
    vector<node> t = {{}};
    vector<int> rt, siz;
    Heap(int n) : rt(n + 1), siz(n + 1) {}

    int join(int x, int y) {
        if (!x || !y) return x | y;
        if (Cmp{}(t[x].v, t[y].v)) swap(x, y);
        t[x].r = join(t[x].r, y);
        if (t[t[x].l].dis < t[t[x].r].dis) swap(t[x].l, t[x].r);
        t[x].dis = t[t[x].r].dis + 1;
        return x;
    }
    void merge(int x, int y) {
        if (x == y) return;
        rt[x] = join(rt[x], rt[y]);
        siz[x] += siz[y];
        rt[y] = siz[y] = 0;
    }
    void push(int x, T v) {
        t.push_back({v, 0, 0, 1});
        rt[x] = join(rt[x], t.size() - 1);
        ++siz[x];
    }
    T ask(int x) const { return t[rt[x]].v; }
    T pop(int x) {
        int p = rt[x];
        rt[x] = join(t[p].l, t[p].r);
        --siz[x];
        return t[p].v;
    }
    void clear(int x) { rt[x] = siz[x] = 0; }
};
```

### 分块与莫队

#### 基础分块（区间加、区间和）

整块维护懒加标记，零散部分直接修改；`a[i]` 不含标记，`sum[k]` 始终是整块的实际和，因此单点赋值要扣除所属块的标记。块长取 $\sqrt n$，初始化与空间为 $\mathcal O(n)$，区间操作为 $\mathcal O(\sqrt n)$，单点赋值为 $\mathcal O(1)$。

```cpp
struct Block {
    int n, len;
    vector<int> a, sum, tag;
    Block(const vector<int> &v) : n(v.size() - 1), len(sqrt(n) + 1), a(v),
        sum((n + len - 1) / len), tag(sum.size()) {
        for (int i = 1; i <= n; ++i) sum[id(i)] += a[i];
    }
    int id(int x) const { return (x - 1) / len; }
    int left(int k) const { return k * len + 1; }
    int right(int k) const { return min(n, (k + 1) * len); }

    void update(int l, int r, int v) {
        for (int k = id(l); k <= id(r); ++k) {
            int x = max(l, left(k)), y = min(r, right(k));
            sum[k] += (y - x + 1) * v;
            if (x == left(k) && y == right(k)) tag[k] += v;
            else for (int i = x; i <= y; ++i) a[i] += v;
        }
    }
    void modify(int x, int v) {
        int k = id(x);
        sum[k] += v - (a[x] + tag[k]);
        a[x] = v - tag[k];
    }
    int ask(int l, int r) const {
        int ans = 0;
        for (int k = id(l); k <= id(r); ++k) {
            int x = max(l, left(k)), y = min(r, right(k));
            if (x == left(k) && y == right(k)) ans += sum[k];
            else for (int i = x; i <= y; ++i) ans += a[i] + tag[k];
        }
        return ans;
    }
};
```

#### 普通莫队

以 $\mathcal O(N \sqrt N)$ 的复杂度完成 $Q$ 次询问的离线查询，其中每个分块的大小取 $\sqrt N=\sqrt {10^5} = 317$ ，也可以使用 `n / min<int>(n, sqrt(q))` 、 `ceil((double)n / (int)sqrt(n))` 或者 `sqrt(n)` 划分。

```c++
signed main() {
    int n;
    cin >> n;
    vector<int> w(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> w[i];
    }

    int q;
    cin >> q;
    vector<array<int, 3>> query(q + 1);
    for (int i = 1; i <= q; ++i) {
        int l, r;
        cin >> l >> r;
        query[i] = {l, r, i};
    }

    int Knum = n / min<int>(n, sqrt(q)); // 计算块长
    vector<int> K(n + 1);
    for (int i = 1; i <= n; ++i) { // 固定块长
        K[i] = (i - 1) / Knum + 1;
    }
    sort(query.begin() + 1, query.end(), [&](auto x, auto y) {
        if (K[x[0]] != K[y[0]]) return x[0] < y[0];
        if (K[x[0]] & 1) return x[1] < y[1];
        return x[1] > y[1];
    });

    int l = 1, r = 0, val = 0;
    vector<int> ans(q + 1);
    for (int i = 1; i <= q; ++i) {
        auto [ql, qr, id] = query[i];
        auto add = [&](int x) -> void {};
        auto del = [&](int x) -> void {};
        while (l > ql) add(w[--l]);
        while (r < qr) add(w[++r]);
        while (l < ql) del(w[l++]);
        while (r > qr) del(w[r--]);
        ans[id] = val;
    }
    for (int i = 1; i <= q; ++i) {
        cout << ans[i] << endl;
    }
}
```

```c++
    vector<int> ans(query.size()), cnt(N);
    for (int i = 1; i < query.size(); ++i) {
        auto [ql, qr, qt, id] = query[i];
        auto add = [&](int x) -> void {
            if (cnt[x] == 0) ++ val;
            ++ cnt[x];
        };
        auto del = [&](int x) -> void {
            -- cnt[x];
            if (cnt[x] == 0) -- val;
        };
        auto time = [&](int x, int l, int r) -> void {
            if (l <= modify[x][1] && modify[x][1] <= r) { //当修改的位置在询问期间内部时才会改变num的值
                del(w[modify[x][1]]);
                add(modify[x][0]);
            }
            swap(w[modify[x][1]], modify[x][0]); //直接交换修改数组的值与原始值，减少额外的数组开销，且方便复原
        };
        while (l > ql) add(w[--l]);
        while (r < qr) add(w[++r]);
        while (l < ql) del(w[l++]);
        while (r > qr) del(w[r--]);
        while (t < qt) time(++t, ql, qr);
        while (t > qt) time(t--, ql, qr);
        ans[id] = val;
    }
    for (int i = 1; i < ans.size(); ++i) {
        cout << ans[i] << endl;
    }
}
```

### 珂朵莉树（ODT）

区间赋值的数据结构都可以骗分，在数据随机的情况下，复杂度可以保证，时间复杂度：$\mathcal O(N\log\log N)$ 。

```c++
struct ODT {
    struct node {
        int l, r;
        mutable LL v;
        node(int l, int r = -1, LL v = 0) : l(l), r(r), v(v) {}
        bool operator<(const node &o) const {
            return l < o.l;
        }
    };
    set<node> s;
    ODT() {
        s.clear();
    }
    auto split(int pos) {
        auto it = s.lower_bound(node(pos));
        if (it != s.end() && it->l == pos) return it;
        it--;
        int l = it->l, r = it->r;
        LL v = it->v;
        s.erase(it);
        s.insert(node(l, pos - 1, v));
        return s.insert(node(pos, r, v)).first;
    }
    void assign(int l, int r, LL x) {
        auto itr = split(r + 1), itl = split(l);
        s.erase(itl, itr);
        s.insert(node(l, r, x));
    }
    void add(int l, int r, LL x) {
        auto itr = split(r + 1), itl = split(l);
        for (auto it = itl; it != itr; it++) {
            it->v += x;
        }
    }
    LL kth(int l, int r, int k) {
        vector<pair<LL, int>> a;
        auto itr = split(r + 1), itl = split(l);
        for (auto it = itl; it != itr; it++) {
            a.push_back(pair<LL, int>(it->v, it->r - it->l + 1));
        }
        sort(a.begin(), a.end());
        for (auto [val, len] : a) {
            k -= len;
            if (k <= 0) return val;
        }
    }
    LL power(LL a, int b, int mod) {
        a %= mod;
        LL res = 1;
        for (; b; b /= 2, a = a * a % mod) {
            if (b % 2) {
                res = res * a % mod;
            }
        }
        return res;
    }
    LL powersum(int l, int r, int x, int mod) {
        auto itr = split(r + 1), itl = split(l);
        LL ans = 0;
        for (auto it = itl; it != itr; it++) {
            ans = (ans + power(it->v, x, mod) * (it->r - it->l + 1) % mod) % mod;
        }
        return ans;
    }
};
```

### KD Tree

#### 静态二维查询（最近点、矩形计数）

用包围盒剪枝，支持矩形内点数和到最近点的距离平方；`askNear(x,y,ban)` 可排除原数组编号 `ban`，默认不排除，无候选点时返回 `nullopt`，重合但编号不同的点仍是合法候选。坐标为整数，距离平方用 `i128` 且须小于 `inf`；交替按两维中位数建树的期望时间为 $\mathcal O(n\log n)$，空间为 $\mathcal O(n)$，矩形计数为 $\mathcal O(\sqrt n)$，最近点查询最坏为 $\mathcal O(n)$。

```cpp
struct KDT {
    static constexpr i128 inf = i128(1) << 120;
    struct node {
        array<int, 2> p{}, mn{}, mx{};
        int l = 0, r = 0, siz = 0, id = 0;
    };
    vector<node> t;
    int rt;
    KDT(const vector<array<int, 2>> &a) : t(a.size()) {
        int n = a.size() - 1;
        for (int i = 1; i <= n; ++i) t[i].p = a[i], t[i].id = i;
        auto build = [&](auto &&self, int l, int r, int d) -> int {
            if (l > r) return 0;
            int m = (l + r) / 2;
            nth_element(t.begin() + l, t.begin() + m, t.begin() + r + 1,
                [&](const node &a, const node &b) { return a.p[d] < b.p[d]; });
            t[m].l = self(self, l, m - 1, d ^ 1);
            t[m].r = self(self, m + 1, r, d ^ 1);
            t[m].mn = t[m].mx = t[m].p;
            t[m].siz = r - l + 1;
            for (int k : {t[m].l, t[m].r}) if (k) {
                for (int j = 0; j < 2; ++j) {
                    t[m].mn[j] = min(t[m].mn[j], t[k].mn[j]);
                    t[m].mx[j] = max(t[m].mx[j], t[k].mx[j]);
                }
            }
            return m;
        };
        rt = build(build, 1, n, 0);
    }
    i128 bound(int p, const array<int, 2> &a) const {
        if (!p) return inf;
        i128 ans = 0;
        for (int j = 0; j < 2; ++j) {
            i128 d = 0;
            if (a[j] < t[p].mn[j]) d = i128(t[p].mn[j]) - a[j];
            if (a[j] > t[p].mx[j]) d = i128(a[j]) - t[p].mx[j];
            ans += d * d;
        }
        return ans;
    }
    optional<i128> askNear(int x, int y, int ban = 0) const {
        array<int, 2> a = {x, y};
        i128 ans = inf;
        auto dfs = [&](auto &&self, int p) -> void {
            if (!p || bound(p, a) >= ans) return;
            if (t[p].id != ban) {
                i128 dx = i128(x) - t[p].p[0], dy = i128(y) - t[p].p[1];
                ans = min(ans, dx * dx + dy * dy);
            }
            int l = t[p].l, r = t[p].r;
            if (bound(l, a) > bound(r, a)) swap(l, r);
            self(self, l), self(self, r);
        };
        dfs(dfs, rt);
        if (ans == inf) return nullopt;
        return ans;
    }
    int ask(int x1, int y1, int x2, int y2) const {
        if (x1 > x2 || y1 > y2) return 0;
        auto dfs = [&](auto &&self, int p) -> int {
            if (!p) return 0;
            auto &u = t[p];
            if (u.mx[0] < x1 || x2 < u.mn[0] || u.mx[1] < y1 || y2 < u.mn[1]) return 0;
            if (x1 <= u.mn[0] && u.mx[0] <= x2 && y1 <= u.mn[1] && u.mx[1] <= y2) return u.siz;
            int ans = x1 <= u.p[0] && u.p[0] <= x2 && y1 <= u.p[1] && u.p[1] <= y2;
            return ans + self(self, u.l) + self(self, u.r);
        };
        return dfs(dfs, rt);
    }
};
```

#### 动态插入与二维偏序计数

在线插入二维点，`query(a)` 统计每一维都不超过 `a` 的点数，重复点按插入次数计。此实现按方差选择划分维、按子树大小失衡重建，查询用包围盒剪枝；单次查询最坏为 $\mathcal O(n)$，不能套用静态交替划分的矩形查询复杂度。

```c++
struct KDT {
    constexpr static int N = 1e5 + 10, K = 2;
    double alpha = 0.725;
    struct node {
        int info[K];
        int mn[K], mx[K];
    } tr[N];
    int ls[N], rs[N], siz[N], id[N], d[N];
    int idx, rt, cur;
    int ans;
    KDT() {
        rt = 0;
        cur = 0;
        memset(ls, 0, sizeof ls);
        memset(rs, 0, sizeof rs);
        memset(d, 0, sizeof d);
    }
    void apply(int p, int son) {
        if (son) {
            for (int i = 0; i < K; ++i) {
                tr[p].mn[i] = min(tr[p].mn[i], tr[son].mn[i]);
                tr[p].mx[i] = max(tr[p].mx[i], tr[son].mx[i]);
            }
            siz[p] += siz[son];
        }
    }
    void maintain(int p) {
        for (int i = 0; i < K; ++i) {
            tr[p].mn[i] = tr[p].info[i];
            tr[p].mx[i] = tr[p].info[i];
        }
        siz[p] = 1;
        apply(p, ls[p]);
        apply(p, rs[p]);
    }
    int build(int l, int r) {
        if (l > r) return 0;
        vector<double> avg(K);
        for (int i = 0; i < K; ++i) {
            for (int j = l; j <= r; ++j) {
                avg[i] += tr[id[j]].info[i];
            }
            avg[i] /= (r - l + 1);
        }
        vector<double> var(K);
        for (int i = 0; i < K; ++i) {
            for (int j = l; j <= r; ++j) {
                var[i] += (tr[id[j]].info[i] - avg[i]) * (tr[id[j]].info[i] - avg[i]);
            }
        }
        int mid = (l + r) / 2;
        int x = max_element(var.begin(), var.end()) - var.begin();
        nth_element(id + l, id + mid, id + r + 1, [&](int a, int b) {
            return tr[a].info[x] < tr[b].info[x];
        });
        d[id[mid]] = x;
        ls[id[mid]] = build(l, mid - 1);
        rs[id[mid]] = build(mid + 1, r);
        maintain(id[mid]);
        return id[mid];
    }
    void print(int p) {
        if (!p) return;
        print(ls[p]);
        id[++idx] = p;
        print(rs[p]);
    }
    void rebuild(int &p) {
        idx = 0;
        print(p);
        p = build(1, idx);
    }
    bool bad(int p) {
        return alpha * siz[p] <= max(siz[ls[p]], siz[rs[p]]);
    }
    void insert(int &p, int cur) {
        if (!p) {
            p = cur;
            maintain(p);
            return;
        }
        if (tr[p].info[d[p]] > tr[cur].info[d[p]]) insert(ls[p], cur);
        else insert(rs[p], cur);
        maintain(p);
        if (bad(p)) rebuild(p);
    }
    void insert(vector<int> &a) {
        cur++;
        for (int i = 0; i < K; ++i) {
            tr[cur].info[i] = a[i];
        }
        insert(rt, cur);
    }
    bool out(int p, vector<int> &a) {
        for (int i = 0; i < K; ++i) {
            if (a[i] < tr[p].mn[i]) {
                return true;
            }
        }
        return false;
    }
    bool in(int p, vector<int> &a) {
        for (int i = 0; i < K; ++i) {
            if (a[i] < tr[p].info[i]) {
                return false;
            }
        }
        return true;
    }
    bool all(int p, vector<int> &a) {
        for (int i = 0; i < K; ++i) {
            if (a[i] < tr[p].mx[i]) {
                return false;
            }
        }
        return true;
    }
    void query(int p, vector<int> &a) {
        if (!p) return;
        if (out(p, a)) return;
        if (all(p, a)) {
            ans += siz[p];
            return;
        }
        if (in(p, a)) ans++;
        query(ls[p], a);
        query(rs[p], a);
    }
    int query(vector<int> &a) {
        ans = 0;
        query(rt, a);
        return ans;
    }
};
```

### 常用技巧

#### 循环移位与下标映射

题意：（区间移位问题）要求将整个序列左移/右移若干个位置，例如，原序列为 $A=(a_1, a_2, \dots, a_n)$ ，右移 $x$ 位后变为 $A=(a_{x+1}, a_{x+2}, \dots, a_n,a_1,a_2,\dots, a_x)$ 。

区间的端点只是一个数字，即使被改变了，通过一定的转换也能够还原，所以我们可以 $\mathcal O(1)$ 解决这一问题。为了方便计算，我们规定下标从 $0$ 开始，即整个线段的区间为 $[0, n)$ ，随后，使用一个偏移量 `shift` 记录。使用 `shift = (shift + x) % n;` 更新偏移量；此后的区间查询/修改前，再将坐标偏移回去即可，下方代码使用区间修改作为示例。

```c++
cin >> l >> r >> x;
l--; // 坐标修改为 0 开始
r--;
l = (l + shift) % n; // 偏移
r = (r + shift) % n;
if (l > r) { // 区间分离则分别操作
    segt.modify(l, n - 1, x);
    segt.modify(0, r, x);
} else {
    segt.modify(l, r, x);
}
```
