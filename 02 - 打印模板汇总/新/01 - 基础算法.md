### 快速断点输出

可以输出重载了 `<<` 的类型，以及 `vector` 和 `array` 。

```cpp
template<class... Args> void _(Args... args) {
    auto _ = [&](auto&&self, auto x) -> void { 
        using T = remove_cvref_t<decltype(x)>;
        if (requires {x.begin(); x.end();} && !is_same_v<T, string>) && !is_same_v<T, string_view> {
            cout << '[';
            bool f = 1;
            for (auto v : x) {
                if (!f) cout << ", ";
                self(self, v);
                f = 0;
            } 
            cout << ']';
        } else {
            cout << x;
        }
    };
    cout << "--->";
    ((_(args), cout << " ", ...));
    cout << "\n";
}
```

### 常用函数

#### 开方

对于 `__i128` 类型，需要将 `sqrt` 替换为 `sqrtl` 。

```cpp
template<class T> T mysqrt(T n) {
    T r = sqrt(n);
    while (r + 1 <= n / (r + 1)) ++r;
    while (r && r > n / r) --r;
    return r;
}
```

#### 对数

```cpp
template<class T> int log2f(T n) {
    int r = -1;
    while (n) ++r, n >>= 1;
    return r;
}
template<class T> int log2c(T n) {
    return log2f(n) + ((n & (n - 1)) != 0);
}
```

#### 取整

```cpp
template <class T> T sign(const T &a) {
    return a == 0 ? 0 : (a < 0 ? -1 : 1);
}
template <class T> T floor(const T &a, const T &b) {
    T q = a / b, r = a % b;
    return q - (r != 0 && ((r > 0) != (b > 0)));
}
template <class T> T ceil(const T &a, const T &b) {
    T q = a / b, r = a % b;
    return q + (r != 0 && ((r > 0) == (b > 0)));
}
```

#### 二次方程

ld delta(ld a, ld b, ld c) {
    return b * b - 4 * a * c;

ld mysqrt(ld x) {
    ld l = 0, r = 1E12;
    for (int i = 0; i < 80; ++i) {
        ld m = (l + r) / 2;
        if (m * m <= x) l = m;
        else r = m;
    }
    return l;
}
void ssolve(int a,int b,int c) {
    int d = delta(a, b, c);
    assert(d >= 0);
    ld x1 = (- b + mysqrt(d)) / (2 * a);
    ld x2 = (- b - mysqrt(d)) / (2 * a);
}

### 整数域二分

#### 库函数

在使用前需要先进行排序，默认升序，降序需要 `lower_bound(<begin>, <end>, <value>, greater<>)`。

`lower_bound` 和 `upper_bound` 返回首个 $\geq x$ 和 $>x$ 的迭代器。

```cpp
vector<int> v = {1, 3, 3, 5};
int x = 3;
auto L = lower_bound(v.begin(), v.end(), x);
auto R = upper_bound(v.begin(), v.end(), x);
int c = R - L; // 元素出现次数

```

- $x$ 或 $x$ 的后继

```cpp
int ans = r;
while (l <= r) {
    int m = (l + r) >> 1;
    if (check(m)) ans = m, r = m - 1;
    else l = m + 1;
}
```

- $x$ 或 $x$ 的前驱

```cpp
int ans = l;
while (l <= r) {
    int m = (l + r) >> 1;
    if (check(m)) ans = m, l = m + 1;
    else r = m - 1;
}
```

### 实数域三分

限制次数实现。

```cpp
ld l = -1E9, r = 1E9;
for (int t = 1; t <= 100; t++) {
    ld m1 = (l * 2 + r) / 3;
    ld m2 = (l + r * 2) / 3;
    if (judge(m1) < judge(m2)) {
        r = m2;
    } else {
        l = m1;
    }
}
cout << l << endl;
```
