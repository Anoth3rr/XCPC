### 常用函数

#### 开方

对非负整数返回精确的向下取整平方根。`__int128` 需要使用 `sqrtl` 。浮点估值后用除法校正，避免平方溢出。

```cpp
template<class T> T mysqrt(T n) {
    T r = sqrt(n);
    while (r + 1 <= n / (r + 1)) ++r;
    while (r && r > n / r) --r;
    return r;
}
```

#### 对数

```cpp
// 等价于 63 - __builtin_clzll(n)
template<class T> int log2f(T n) {
    int r = -1;
    while (n) ++r, n >>= 1;
    return r;
}
template<class T> int log2c(T n) {
    return log2f(n) + ((n & (n - 1)) != 0);
}
```

#### 取整

```cpp
template <class T> T sign(const T &x) {
    return (x > 0) - (x < 0);
}
template<class T> T floor(const T &a, const T &b) {
    T A = a / b, B = a % b;
    return A - (B && ((B < 0) != (b < 0)));
}
template<class T> T ceil(const T &a, const T &b) {
    T A = a / b, B = a % b;
    return A + (B && ((B < 0) == (b < 0)));
}
```

### 整数域二分

#### 库函数

在使用前需要**先进行排序**，默认升序，降序需要 `lower_bound(<begin>, <end>, <value>, <cmp>)`，没有满足条件的位置时返回 `end()`。

升序下，`lower_bound` 和 `upper_bound` 返回首个 $\geq x$ 和 $ >x $ 的迭代器。

```cpp
vector<int> v = {1, 3, 3, 5};
int x = 3;
auto L = lower_bound(v.begin(), v.end(), x);
auto R = upper_bound(v.begin(), v.end(), x);

auto l = L - v.begin(); // 下标：1
auto r = R - v.begin(); // 下标：3

int c = R - L; // 元素出现次数
```

#### 目标或目标后继

`check(x)` 随 $x$ 增大从 `false` 变为 `true`，**且存在 `true`**。返回首个 `true` 的位置。

```cpp
int ans = r;
// assert(check(ans));
while (l <= r) {
    int m = l + (r - l) / 2;
    if (check(m)) ans = m, r = m - 1;
    else l = m + 1;
}
return ans;
```

#### 目标或目标的前驱

`check(x)` 随 $x$ 增大从 `true` 变为 `false`，**且存在 `true`**。返回最后一个 `true` 的位置。

```cpp
int ans = l;
// assert(check(ans));
while (l <= r) {
    int m = l + (r - l) / 2;
    if (check(m)) ans = m, l = m + 1;
    else r = m - 1;
}
return ans;
```

### 实数域二分

限制二分次数实现，`check(l) = false`、`check(r) = true`、`check(x)` 随 $x$ 增大从 `false` 变为 `true`。

for (int i = 1; i <= 100; ++i) {

### 整数域三分

针对单谷函数，单峰应改变比较方向，返回最小值所在位置，多个最小值返回平台最左侧位置。

```cpp
while (l < r) {
    int m = l + (r - l) / 2;
    if (check(m) <= check(m + 1)) r = m;
    else l = m + 1;
}
return l;
```

### 实数域三分

针对单谷函数，单峰应改变比较方向，返回最小值所在近似位置，**处理不了形如“平台-谷-平台”的函数，如 $\min((x - 1) ^ 2, 1)$。**

```cpp
for (int i = 1; i <= 100; ++i) {
    ld m1 = l + (r - l) / 3;
    ld m2 = l + (r - l) * 2 / 3;
    if (check(m1) < check(m2)) r = m2;
    else l = m1;
}
return l;
```

### 前缀和与差分：离线区间修改、静态区间查询

先完成全部区间加，再还原数组并建立前缀和；修改与查询不能交错。下标从 1 开始，数组第 0 项为 0。

```c++
vector<int> d(n + 2);
int p = 0;
for (int i = 1; i <= n; ++i) d[i] = a[i] - a[i - 1];
auto add = [&](int l, int r, int v) { d[l] += v, d[r + 1] -= v; };
// 所有修改完成后还原 a
for (int i = 1; i <= n; ++i) d[i] += d[i - 1], a[i] = d[i];
vector<int> pre(n + 1);
for (int i = 1; i <= n; ++i) pre[i] = pre[i - 1] + a[i];
auto ask = [&](int l, int r) { return pre[r] - pre[l - 1]; };
```

### 二维前缀和与差分：离线矩形修改、静态矩形查询

先完成全部矩形加，再还原矩阵并建立二维前缀和。第 0 行、第 0 列须补零，修改与查询不能交错。

```c++
vector<vector<int>> d(n + 2, vector<int>(m + 2));
for (int x = 1; x <= n; ++x)
    for (int y = 1; y <= m; ++y)
        d[x][y] = a[x][y] - a[x - 1][y] - a[x][y - 1] + a[x - 1][y - 1];
auto add = [&](int u, int v, int x, int y, int p) {
    x++, y++;
    d[u][v] += p, d[x][v] -= p;
    d[x][y] += p, d[u][y] -= p;
};

for (int x = 1; x <= n; ++x)
    for (int y = 1; y <= m; ++y)
        d[x][y] += d[x - 1][y] + d[x][y - 1] - d[x - 1][y - 1], a[x][y] = d[x][y];
vector<vector<int>> pre(n + 1, vector<int>(m + 1));
for (int x = 1; x <= n; ++x)
    for (int y = 1; y <= m; ++y)
        pre[x][y] = a[x][y] + pre[x - 1][y] + pre[x][y - 1] - pre[x - 1][y - 1];
auto ask = [&](int u, int v, int x, int y) {
    return pre[x][y] - pre[u - 1][y] - pre[x][v - 1] + pre[u - 1][v - 1];
};
```

#### 封装

```c++
template <typename T> struct Compress_ {
    static constexpr T inf = numeric_limits<T>::max();
    int n, shift = 0; // shift 用于标记下标偏移量
    vector<T> alls;

    Compress_() {}
    Compress_(auto in) : alls(in) {
        init();
    }
    void add(T x) {
        alls.emplace_back(x);
    }
    template <typename... Args> void add(T x, Args... args) {
        add(x), add(args...);
    }
    void init() {
        alls.emplace_back(inf);
        sort(alls.begin(), alls.end());
        alls.erase(unique(alls.begin(), alls.end()), alls.end());
        this->n = alls.size();
    }
    int size() {
        return n;
    }
    int operator[](T x) { // 返回 x 元素的新下标
        return upper_bound(alls.begin(), alls.end(), x) - alls.begin() + shift;
    }
    T Get(int x) { // 根据新下标返回原来元素
        assert(x - shift < n);
        return x - shift < n ? alls[x - shift] : -1;
    }
    bool count(T x) { // 查找元素 x 是否存在
        return binary_search(alls.begin(), alls.end(), x);
    }
    friend auto &operator<< (ostream &o, const auto &j) {
        cout << "{";
        for (auto it : j.alls) {
            o << it << " ";
        }
        return o << "}";
    }
};
using Compress = Compress_<int>;
```
