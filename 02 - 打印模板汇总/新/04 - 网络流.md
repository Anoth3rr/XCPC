## 第四章：网络流

### 最大流

使用 $\tt Dinic$ 算法，理论最坏复杂度为 $\mathcal O(N^2M)$ ，例题范围：$N=1200,\ m=5\times 10^3$ 。一般步骤：$\tt BFS$ 建立分层图，无回溯 $\tt DFS$ 寻找所有可行的增广路径。封装：求从点 $S$ 到点 $T$ 的最大流。预流推进见==常数优化章节==。

```c++
template <typename T> struct Flow {
    const int n;
    static constexpr T inf = numeric_limits<T>::max();
    struct Edge {
        int to;
        T w;
        Edge(int to, T w) : to(to), w(w) {}
    };
    vector<Edge> ver;
    vector<vector<int>> h;
    vector<int> cur, d;

    Flow(int n) : n(n + 1), h(n + 1) {}
    void add(int u, int v, T c) {
        h[u].push_back(ver.size());
        ver.emplace_back(v, c);
        h[v].push_back(ver.size());
        ver.emplace_back(u, 0);
    }
    bool bfs(int s, int t) {
        d.assign(n, -1);
        d[s] = 0;
        queue<int> q;
        q.push(s);
        while (!q.empty()) {
            auto x = q.front();
            q.pop();
            for (auto it : h[x]) {
                auto [y, w] = ver[it];
                if (w && d[y] == -1) {
                    d[y] = d[x] + 1;
                    if (y == t) return true;
                    q.push(y);
                }
            }
        }
        return false;
    }
    T dfs(int u, int t, T f) {
        if (u == t) return f;
        auto r = f;
        for (int &i = cur[u]; i < h[u].size(); i++) {
            auto j = h[u][i];
            auto &[v, c] = ver[j];
            auto &[u, rc] = ver[j ^ 1];
            if (c && d[v] == d[u] + 1) {
                auto a = dfs(v, t, std::min(r, c));
                c -= a;
                rc += a;
                r -= a;
                if (!r) return f;
            }
        }
        return f - r;
    }
    T work(int s, int t) {
        T ans = 0;
        while (bfs(s, t)) {
            cur.assign(n, 0);
            ans += dfs(s, t, inf);
        }
        return ans;
    }
};
```

### 最小割

最小割值等于最大流值；选择带依赖关系的点使总权最大，可用最大权闭合图建模。

#### 最大权闭合图

设置 `w[u]`，`add(u,v)` 表示选 `u` 必须选 `v`；`work()` 返回最大权值，`ans` 保存所选点。使用 Dinic `Flow<int>` 模板，正权和加一及负权的相反数须能用 `int` 表示。

```c++
struct Closure {
    int n;
    vector<int> w, ans;
    vector<pii> e;
    Closure(int n) : n(n), w(n + 1) {}
    void add(int u, int v) { e.push_back({u, v}); }
    int work() {
        int s = n + 1, t = n + 2, sum = 0;
        Flow<int> f(t);
        for (int u = 1; u <= n; u++) {
            if (w[u] > 0) f.add(s, u, w[u]), sum += w[u];
            if (w[u] < 0) f.add(u, t, -w[u]);
        }
        for (auto [u, v] : e) f.add(u, v, sum + 1);
        int r = sum - f.work(s, t);
        f.bfs(s, t);
        ans.clear();
        for (int u = 1; u <= n; u++) if (f.d[u] != -1) ans.push_back(u);
        return r;
    }
};
```

项目收益与共享材料成本：读入 `n` 个收益，再读入 `m` 组 `x,y,w`，表示项目 `x,y` 均依赖成本为 `w` 的材料。

```c++
signed main() {
    int n, m;
    cin >> n >> m;
    Closure g(n + m);
    for (int i = 1; i <= n; i++) cin >> g.w[i];
    for (int i = 1; i <= m; i++) {
        int x, y, w;
        cin >> x >> y >> w;
        g.w[n + i] = -w;
        g.add(x, n + i);
        g.add(y, n + i);
    }
    cout << g.work() << '\n';
}
```

### 最小割树 Gomory-Hu Tree

求无向非负容量图的任意两点最小割，等于最小割树路径上的最小边权；原图可以不连通，不同连通块之间答案为 $0$，约定 `ask(u,u)=0`。加边后调用 `build()`，支持重复构建。

每轮先退流，再利用残量图划分点集，共调用 $n-1$ 次最大流。此版预处理全部点对答案，额外时间、空间均为 $O(n^2)$，单次查询 $O(1)$；使用本章 Dinic 时，建树与预处理的总时间上界为 $O(n^3m+n^2)$。

```c++
struct GHTree {
    int n;
    Flow<int> flow;
    vector<vector<pii>> g;
    vector<vector<int>> ans;

    GHTree(int n) : n(n), flow(n) {}
    void add(int u, int v, int w) {
        flow.add(u, v, w);
        flow.add(v, u, w);
    }
    void build() {
        vector<int> fa(n + 1, 1), cut(n + 1);
        fa[1] = 0;
        for (int s = 2; s <= n; s++) {
            for (int i = 0; i < flow.ver.size(); i += 2) {
                flow.ver[i].w += flow.ver[i ^ 1].w;
                flow.ver[i ^ 1].w = 0;
            }
            int t = fa[s], w = flow.work(s, t);
            auto &d = flow.d; // 最后一轮失败 BFS：d[u] != -1 表示残量可达
            cut[s] = w;
            for (int u = s + 1; u <= n; u++) {
                if (fa[u] == t && d[u] != -1) fa[u] = s;
            }
            if (fa[t] && d[fa[t]] != -1) {
                fa[s] = fa[t], fa[t] = s;
                cut[s] = cut[t], cut[t] = w;
            }
        }

        g.assign(n + 1, {});
        for (int u = 2; u <= n; u++) {
            g[u].push_back({fa[u], cut[u]});
            g[fa[u]].push_back({u, cut[u]});
        }
        ans.assign(n + 1, vector<int>(n + 1));
        for (int s = 1; s <= n; s++) {
            auto dfs = [&](auto &&self, int u, int fa, int w) -> void {
                ans[s][u] = w;
                for (auto [v, c] : g[u]) {
                    if (v != fa) self(self, v, u, min(w, c));
                }
            };
            dfs(dfs, s, 0, flow.inf);
            ans[s][s] = 0;
        }
    }
    int ask(int u, int v) const { return ans[u][v]; }
};
```

```cpp
GHTree tr(n);
for (auto [u, v, w] : edges) tr.add(u, v, w); // 每条无向边调用一次
tr.build();
cout << tr.ask(u, v) << "\n";
```

### 上下界网络流

每条有向边满足 `lo <= f <= hi`，要求 `0 <= lo <= hi`，且所有上界之和能用 `int` 表示。`add(u, v, lo, hi)` 返回边编号；`solve()` 求可行循环流，`solve(s, t)` 求非负最大流（`s != t`），返回是否可行；有源汇版本若仅存在负流值的可行流，也返回 `false`。

成功后，`ans[id]` 为该边的实际流量，`val` 为最大流值（循环流为 `0`）；失败时清空 `ans` 并将 `val` 置 `0`，支持重复调用。使用 Dinic `Flow<int>` 模板，增广图增加两个点及 $O(n)$ 条边，最多调用两次最大流，时间 $O(n^2(m+n))$，空间 $O(n+m)$。

```c++
struct BoundFlow {
    int n, val = 0;
    vector<array<int, 4>> e;
    vector<int> ans;

    BoundFlow(int n) : n(n) {}
    int add(int u, int v, int lo, int hi) {
        e.push_back({u, v, lo, hi});
        return e.size() - 1;
    }
    bool solve(int s = 0, int t = 0) {
        ans.clear();
        val = 0;
        Flow<int> flow(n + 2);
        vector<int> bal(n + 1);
        int cap = 0;
        for (auto [u, v, lo, hi] : e) {
            flow.add(u, v, hi - lo);
            bal[u] -= lo;
            bal[v] += lo;
            cap += hi;
        }
        int m = e.size(), id = flow.ver.size();
        if (s) flow.add(t, s, cap);
        int ss = n + 1, tt = n + 2, need = 0;
        for (int u = 1; u <= n; u++) {
            if (bal[u] > 0) {
                flow.add(ss, u, bal[u]);
                need += bal[u];
            } else if (bal[u] < 0) {
                flow.add(u, tt, -bal[u]);
            }
        }
        if (flow.work(ss, tt) != need) return false;
        if (s) {
            val = flow.ver[id + 1].w;
            for (int i = id; i < flow.ver.size(); i++) flow.ver[i].w = 0;
            val += flow.work(s, t);
        }
        for (int i = 0; i < m; i++) ans.push_back(e[i][2] + flow.ver[2 * i + 1].w);
        return true;
    }
};
```

```c++
BoundFlow tr(3);
int id = tr.add(1, 2, 1, 3);
tr.add(2, 3, 1, 2);
bool ok = tr.solve(1, 3); // true，tr.val = 2，tr.ans[id] = 2
```

### 费用流

给定一个带费用的网络，规定 $(u,v)$ 间的费用为 $f(u,v) \times w(u,v)$ ，求解该网络中总花费最小的最大流称之为**最小费用最大流**。总时间复杂度为 $\mathcal O(NMf)$ ，其中 $f$ 代表最大流。

```c++
struct MinCostFlow {
    static constexpr int inf = numeric_limits<int>::max();
    struct Edge {
        int v, c, f;
        Edge(int v, int c, int f) : v(v), c(c), f(f) {}
    };
    const int n;
    vector<Edge> e;
    vector<vector<int>> g;
    vector<int> h, dis;
    vector<int> pre;
    
    MinCostFlow(int n) : n(n), g(n) {}
    void add(int u, int v, int c, int f) { // c 流量, f 费用
        // if (f < 0) {
        //     g[u].push_back(e.size());
        //     e.emplace_back(v, 0, f);
        //     g[v].push_back(e.size());
        //     e.emplace_back(u, c, -f);
        // } else {
            g[u].push_back(e.size());
            e.emplace_back(v, c, f);
            g[v].push_back(e.size());
            e.emplace_back(u, 0, -f);
        // }
    }
    bool dijkstra(int s, int t) {
        dis.assign(n, inf);
        pre.assign(n, -1);
        priority_queue<pii, vector<pii>, greater<pii>> que;
        dis[s] = 0;
        que.push({0, s});
        while (!que.empty()) {
            auto [d, u] = que.top();
            que.pop();
            if (dis[u] < d) continue;
            for (int i : g[u]) {
                auto [v, c, f] = e[i];
                if (c > 0 && dis[v] > d + h[u] - h[v] + f) {
                    dis[v] = d + h[u] - h[v] + f;
                    pre[v] = i;
                    que.push({dis[v], v});
                }
            }
        }
        return dis[t] != inf;
    }
    pii flow(int s, int t) {
        int flow = 0;
        int cost = 0;
        h.assign(n, 0);
        while (dijkstra(s, t)) {
            for (int i = 0; i < n; ++i) {
                if (dis[i] != inf) h[i] += dis[i];
            }
            int aug = inf;
            for (int i = t; i != s; i = e[pre[i] ^ 1].v) aug = min(aug, e[pre[i]].c);
            for (int i = t; i != s; i = e[pre[i] ^ 1].v) {
                e[pre[i]].c -= aug;
                e[pre[i] ^ 1].c += aug;
            }
            flow += aug;
            cost += aug * h[t];
        }
        return {flow, cost};
    }
};  
```
