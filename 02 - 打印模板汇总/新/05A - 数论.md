#### 调和级数

### 最大公约数 `gcd` 

$\tt{GCC} 环境下 $`std::gcd` 和 `__gcd` 使用的都是 $\tt{Stein}$ 算法，单次时间与欧几里得算法几乎没有差别，大量运算时常数略优，相比于欧几里得算法不容易被卡掉。但是在极端环境，如 $\gcd(10^{18}, 1)$ ，表现略逊于欧几里得算法。

#### 位运算优化

手写 $\tt {Stein}$ 算法。**略快于内置函数，用于卡常。**

```cpp
int gcd(int a, int b) {
    if (!a || !b) return a | b;
    int t = __builtin_ctzll(a | b);
    a >>= __builtin_ctzll(a);
    while (b) {
        b >>= __builtin_ctzll(b);
        if (a > b) swap(a, b);
        b -= a;
    }
    return a << t;
}
```

#### 最小公倍数

```cpp
int lcm(int x, int y) { return x / gcd(x, y) * y; }
```

### 大整数类（高精度计算）

保留十进制读写、四则运算、余数、gcd/lcm 与 Karatsuba 乘法；零统一为正零。小整数乘除要求其绝对值不超过 $2^{31}-1$，除数非零；高精 gcd 的 Stein 备选见第十三章。

```c++
const int base = 1000000000;
const int base_digits = 9; // 分解为九个数位一个数字
struct bigint {
    vector<int> a;
    int sign;

    bigint() : sign(1) {}
    bigint operator-() const {
        bigint res = *this;
        if (!res.a.empty()) res.sign = -sign;
        return res;
    }
    bigint(long long v) {
        *this = v;
    }
    bigint(const string &s) {
        read(s);
    }
    void operator=(const bigint &v) {
        sign = v.sign;
        a = v.a;
    }
    void operator=(long long v) {
        a.clear();
        sign = v < 0 ? -1 : 1;
        u64 x = v < 0 ? -u64(v) : u64(v);
        for (; x; x /= base) a.push_back(x % base);
    }

    // 基础加减乘除
    bigint operator+(const bigint &v) const {
        if (sign == v.sign) {
            bigint res = v;
            for (int i = 0, carry = 0; i < max(a.size(), v.a.size()) || carry; ++i) {
                if (i == res.a.size()) {
                    res.a.push_back(0);
                }
                res.a[i] += carry + (i < a.size() ? a[i] : 0);
                carry = res.a[i] >= base;
                if (carry) {
                    res.a[i] -= base;
                }
            }
            return res;
        }
        return *this - (-v);
    }
    bigint operator-(const bigint &v) const {
        if (sign == v.sign) {
            if (abs() >= v.abs()) {
                bigint res = *this;
                for (int i = 0, carry = 0; i < v.a.size() || carry; ++i) {
                    res.a[i] -= carry + (i < v.a.size() ? v.a[i] : 0);
                    carry = res.a[i] < 0;
                    if (carry) {
                        res.a[i] += base;
                    }
                }
                res.trim();
                return res;
            }
            return -(v - *this);
        }
        return *this + (-v);
    }
    void operator*=(int v) {
        check(v);
        for (int i = 0, carry = 0; i < a.size() || carry; ++i) {
            if (i == a.size()) {
                a.push_back(0);
            }
            long long cur = a[i] * (long long)v + carry;
            carry = (int)(cur / base);
            a[i] = (int)(cur % base);
        }
        trim();
    }
    void operator/=(int v) {
        check(v);
        for (int n = a.size(), i = n - 1, rem = 0; i >= 0; --i) {
            long long cur = a[i] + rem * (long long)base;
            a[i] = (int)(cur / v);
            rem = (int)(cur % v);
        }
        trim();
    }
    int operator%(int v) const {
        if (v < 0) {
            v = -v;
        }
        int m = 0;
        for (int n = a.size(), i = n - 1; i >= 0; --i) {
            m = (a[i] + m * (long long)base) % v;
        }
        return m * sign;
    }

    void operator+=(const bigint &v) {
        *this = *this + v;
    }
    void operator-=(const bigint &v) {
        *this = *this - v;
    }
    bigint operator*(int v) const {
        bigint res = *this;
        res *= v;
        return res;
    }
    bigint operator/(int v) const {
        bigint res = *this;
        res /= v;
        return res;
    }
    void operator%=(const int &v) {
        *this = *this % v;
    }

    bool operator<(const bigint &v) const {
        if (sign != v.sign) return sign < v.sign;
        int n = a.size(), m = v.a.size();
        if (n != m) return sign > 0 ? n < m : n > m;
        for (int n = a.size(), i = n - 1; i >= 0; i--)
            if (a[i] != v.a[i]) return a[i] * sign < v.a[i] * sign;
        return false;
    }
    bool operator>(const bigint &v) const {
        return v < *this;
    }
    bool operator<=(const bigint &v) const {
        return !(v < *this);
    }
    bool operator>=(const bigint &v) const {
        return !(*this < v);
    }
    bool operator==(const bigint &v) const {
        return !(*this < v) && !(v < *this);
    }
    bool operator!=(const bigint &v) const {
        return *this < v || v < *this;
    }

    bigint abs() const {
        bigint res = *this;
        res.sign *= res.sign;
        return res;
    }
    void check(int &v) { // 检查输入的是否为负数
        if (v < 0) {
            sign = -sign;
            v = -v;
        }
    }
    void trim() { // 去除前导零
        while (!a.empty() && !a.back()) a.pop_back();
        if (a.empty()) sign = 1;
    }
    bool isZero() const { // 判断是否等于零
        return a.empty() || (a.size() == 1 && !a[0]);
    }
    friend bigint gcd(const bigint &a, const bigint &b) {
        return b.isZero() ? a.abs() : gcd(b, a % b);
    }
    friend bigint lcm(const bigint &a, const bigint &b) {
        if (a.isZero() || b.isZero()) return 0;
        return (a / gcd(a, b) * b).abs();
    }
    void read(const string &s) {
        sign = 1;
        a.clear();
        int pos = 0;
        while (pos < s.size() && (s[pos] == '-' || s[pos] == '+')) {
            if (s[pos] == '-') sign = -sign;
            ++pos;
        }
        for (int n = s.size(), i = n - 1; i >= pos; i -= base_digits) {
            int x = 0;
            for (int j = max(pos, i - base_digits + 1); j <= i; ++j) x = x * 10 + s[j] - '0';
            a.push_back(x);
        }
        trim();
    }
    friend istream &operator>>(istream &stream, bigint &v) {
        string s;
        stream >> s;
        v.read(s);
        return stream;
    }
    friend ostream &operator<<(ostream &stream, const bigint &v) {
        if (v.sign == -1) stream << '-';
        stream << (v.a.empty() ? 0 : v.a.back());
        for (int n = v.a.size(), i = n - 2; i >= 0; --i)
            stream << setw(base_digits) << setfill('0') << v.a[i];
        return stream;
    }

    /* 大整数乘除大整数部分 */
    typedef vector<long long> vll;
    bigint operator*(const bigint &v) const { // 大整数乘大整数
        vector<int> a6 = convert_base(this->a, base_digits, 6);
        vector<int> b6 = convert_base(v.a, base_digits, 6);
        vll a(a6.begin(), a6.end());
        vll b(b6.begin(), b6.end());
        while (a.size() < b.size()) a.push_back(0);
        while (b.size() < a.size()) b.push_back(0);
        while (a.size() & (a.size() - 1)) a.push_back(0), b.push_back(0);
        vll c = karatsubaMultiply(a, b);
        bigint res;
        res.sign = sign * v.sign;
        for (int i = 0, carry = 0; i < c.size(); ++i) {
            long long cur = c[i] + carry;
            res.a.push_back((int)(cur % 1000000));
            carry = (int)(cur / 1000000);
        }
        res.a = convert_base(res.a, 6, base_digits);
        res.trim();
        return res;
    }
    friend pair<bigint, bigint> divmod(const bigint &a1,
                                       const bigint &b1) { // 大整数除大整数，同时返回答案与余数
        assert(!b1.isZero());
        int norm = base / (b1.a.back() + 1);
        bigint a = a1.abs() * norm;
        bigint b = b1.abs() * norm;
        bigint q, r;
        q.a.resize(a.a.size());
        for (int n = a.a.size(), i = n - 1; i >= 0; i--) {
            r *= base;
            r += a.a[i];
            int s1 = r.a.size() <= b.a.size() ? 0 : r.a[b.a.size()];
            int s2 = r.a.size() <= b.a.size() - 1 ? 0 : r.a[b.a.size() - 1];
            int d = ((long long)base * s1 + s2) / b.a.back();
            r -= b * d;
            while (r < 0) r += b, --d;
            q.a[i] = d;
        }
        q.sign = a1.sign * b1.sign;
        r.sign = a1.sign;
        q.trim();
        r.trim();
        return make_pair(q, r / norm);
    }
    static vector<int> convert_base(const vector<int> &a, int old_digits, int new_digits) {
        vector<long long> p(max(old_digits, new_digits) + 1);
        p[0] = 1;
        for (int i = 1; i < p.size(); ++i) p[i] = p[i - 1] * 10;
        vector<int> res;
        long long cur = 0;
        int cur_digits = 0;
        for (int i = 0; i < a.size(); ++i) {
            cur += a[i] * p[cur_digits];
            cur_digits += old_digits;
            while (cur_digits >= new_digits) {
                res.push_back((int)(cur % p[new_digits]));
                cur /= p[new_digits];
                cur_digits -= new_digits;
            }
        }
        res.push_back((int)cur);
        while (!res.empty() && !res.back()) res.pop_back();
        return res;
    }
    static vll karatsubaMultiply(const vll &a, const vll &b) {
        int n = a.size();
        vll res(n + n);
        if (n <= 32) {
            for (int i = 0; i < n; ++i) {
                for (int j = 0; j < n; ++j) {
                    res[i + j] += a[i] * b[j];
                }
            }
            return res;
        }

        int k = n >> 1;
        vll a1(a.begin(), a.begin() + k);
        vll a2(a.begin() + k, a.end());
        vll b1(b.begin(), b.begin() + k);
        vll b2(b.begin() + k, b.end());

        vll a1b1 = karatsubaMultiply(a1, b1);
        vll a2b2 = karatsubaMultiply(a2, b2);

        for (int i = 0; i < k; ++i) a2[i] += a1[i];
        for (int i = 0; i < k; ++i) b2[i] += b1[i];

        vll r = karatsubaMultiply(a2, b2);
        for (int i = 0; i < a1b1.size(); ++i) r[i] -= a1b1[i];
        for (int i = 0; i < a2b2.size(); ++i) r[i] -= a2b2[i];

        for (int i = 0; i < r.size(); ++i) res[i + k] += r[i];
        for (int i = 0; i < a1b1.size(); ++i) res[i] += a1b1[i];
        for (int i = 0; i < a2b2.size(); ++i) res[i + n] += a2b2[i];
        return res;
    }

    void operator*=(const bigint &v) {
        *this = *this * v;
    }
    bigint operator/(const bigint &v) const {
        return divmod(*this, v).first;
    }
    void operator/=(const bigint &v) {
        *this = *this / v;
    }
    bigint operator%(const bigint &v) const {
        return divmod(*this, v).second;
    }
    void operator%=(const bigint &v) {
        *this = *this % v;
    }
};
```

#### 朴素高精乘法备选

两数较短或长度悬殊时可直接调用 `mulSimple`。设块数为 n、m，时间 $O(nm)$、额外空间 $O(n+m)$。

```c++
bigint mulSimple(const bigint &x, const bigint &y) {
    if (x.isZero() || y.isZero()) return 0;
    bigint r;
    r.sign = x.sign * y.sign;
    r.a.assign(x.a.size() + y.a.size(), 0);
    for (int i = 0; i < x.a.size(); ++i) {
        int c = 0;
        for (int j = 0; j < y.a.size() || c; ++j) {
            int v = r.a[i + j] + c;
            if (j < y.a.size()) v += x.a[i] * y.a[j];
            r.a[i + j] = v % base, c = v / base;
        }
    }
    r.trim();
    return r;
}
```

### 分数运算类

分母恒正、每次运算后既约，乘除先交叉约分；保留 `x/y`、`val()`、`norm()` 接口。整数中间结果仍须不溢出；可用本章 `bigint` 作底层类型，此时不调用转浮点的 `val()`。

```c++
template<class T = int> struct Frac {
    T x, y;
    Frac(T x = 0, T y = 1) : x(x), y(y) { reduce(); }
    static T ab(T v) { return v < 0 ? -v : v; }
    void reduce() {
        assert(y != T(0));
        if (y < 0) x = -x, y = -y;
        if (x == T(0)) { y = 1; return; }
        T g = gcd(ab(x), y);
        x = x / g, y = y / g;
    }
    Frac norm() const { return Frac(x, y); }
    double val() const { return 1. * x / y; }
    Frac &operator+=(const Frac &v) {
        T g = gcd(y, v.y);
        x = x * (v.y / g) + v.x * (y / g), y = y / g * v.y;
        reduce();
        return *this;
    }
    Frac &operator-=(const Frac &v) { return *this += -v; }
    Frac &operator*=(const Frac &v) {
        T g1 = gcd(ab(x), v.y), g2 = gcd(ab(v.x), y);
        x = x / g1 * (v.x / g2), y = y / g2 * (v.y / g1);
        return *this;
    }
    Frac &operator/=(const Frac &v) {
        assert(v.x != T(0));
        T g1 = gcd(ab(x), ab(v.x)), g2 = gcd(y, v.y), vx = v.x / g1;
        x = x / g1 * (v.y / g2), y = y / g2 * vx;
        if (y < 0) x = -x, y = -y;
        return *this;
    }
    friend Frac operator+(Frac a, const Frac &b) { return a += b; }
    friend Frac operator-(Frac a, const Frac &b) { return a -= b; }
    friend Frac operator*(Frac a, const Frac &b) { return a *= b; }
    friend Frac operator/(Frac a, const Frac &b) { return a /= b; }
    friend Frac operator-(Frac a) { return Frac(-a.x, a.y); }
    friend bool operator<(const Frac &a, const Frac &b) {
        using W = conditional_t<is_integral_v<T>, i128, T>;
        return W(a.x) * W(b.y) < W(b.x) * W(a.y);
    }
    friend bool operator>(const Frac &a, const Frac &b) { return b < a; }
    friend bool operator<=(const Frac &a, const Frac &b) { return !(b < a); }
    friend bool operator>=(const Frac &a, const Frac &b) { return !(a < b); }
    friend bool operator==(const Frac &a, const Frac &b) { return a.x == b.x && a.y == b.y; }
    friend bool operator!=(const Frac &a, const Frac &b) { return !(a == b); }
    friend ostream &operator<<(ostream &out, const Frac &a) {
        return a.y == T(1) ? out << a.x : out << a.x << '/' << a.y;
    }
};
```

### 快速幂

#### 常规

时间复杂度 $\mathcal O(\log N)$ 。

```c++
int power(int a, int b, int m) {
    int r = 1 % m;
    a %= m;
    for (; b; b >>= 1, a = a * a % m) 
        if (b & 1) r = r * a % m;
    return r;
}
```

#### 长整型 with 防爆乘法

使用 `i128` 精确计算，支持 $1\le m\le 2^{63}-1$；底数允许为负，指数非负。浮点估商法依赖浮点精度且容易触发有符号溢出，此处统一用整数版本。

```c++
int mul(int a, int b, int m) {
    int r = i128(a) * b % m;
    return r < 0 ? r + m : r;
}

int power(int a, int b, int m) {

    a %= m;
    for (; b; b >>= 1, a = mul(a, a, m))

    return r;
}
```

### 取模运算类

集成了常见的取模四则运算，运算速度与手动取模相差无几，效率极高。

```c++
template <class T> constexpr T power(T n, int k) {
    T r = 1;
    for (; k; k >>= 1, n *= n)
        if (k & 1) r *= n;
    return r;
}

template <const int &MOD> struct MInt {
    int x;
    MInt(signed x = 0) : x(norm(x % MOD)) {}
    MInt(int x) : x(norm(x % MOD)) {}

    constexpr int norm(int x) const noexcept {
        if (x < 0) [[unlikely]] {
            x += MOD;
        }
        if (x >= MOD) [[unlikely]] {
            x -= MOD;
        }
        return x;
    }
    explicit operator int() const { return x; }
    constexpr int val() const { return x; }
    constexpr MInt operator-() const {
        MInt val = norm(MOD - x);
        return val;
    }
    constexpr MInt inv() const {
        assert(x != 0);
        return power(*this, MOD - 2);
    }
    friend constexpr auto &operator>>(istream &in, MInt &j) {
        int v;
        in >> v;
        j = MInt(v);
        return in;
    }
    friend constexpr auto &operator<<(ostream &o, const MInt &j) { return o << j.val(); }
    constexpr MInt &operator++() {
        x = norm(x + 1);
        return *this;
    }
    constexpr MInt &operator--() {
        x = norm(x - 1);
        return *this;
    }
    constexpr MInt operator++(signed) {
        MInt res = *this;
        ++*this;
        return res;
    }
    constexpr MInt operator--(signed) {
        MInt res = *this;
        --*this;
        return res;
    }
    constexpr MInt &operator+=(const MInt &i) {
        x = norm(x + i.x);
        return *this;
    }
    constexpr MInt &operator-=(const MInt &i) {
        x = norm(x - i.x);
        return *this;
    }
    constexpr MInt &operator*=(const MInt &i) {
        x = x * i.x % MOD;
        return *this;
    }
    constexpr MInt &operator/=(const MInt &i) { return *this *= i.inv(); }
    constexpr MInt &operator%=(const int &i) { return x %= i, *this; }
    friend constexpr MInt operator+(const MInt i, const MInt j) { return MInt(i) += j; }
    friend constexpr MInt operator-(const MInt i, const MInt j) { return MInt(i) -= j; }
    friend constexpr MInt operator*(const MInt i, const MInt j) { return MInt(i) *= j; }
    friend constexpr MInt operator/(const MInt i, const MInt j) { return MInt(i) /= j; }
    friend constexpr MInt operator%(const MInt i, const int j) { return MInt(i) %= j; }
    friend constexpr bool operator==(const MInt i, const MInt j) { return i.val() == j.val(); }
    friend constexpr bool operator!=(const MInt i, const MInt j) { return i.val() != j.val(); }
    friend constexpr bool operator<(const MInt i, const MInt j) { return i.val() < j.val(); }
    friend constexpr bool operator>(const MInt i, const MInt j) { return i.val() > j.val(); }
};
```

### 动态模数取模类（Barrett 约减）

运行期确定模数时选用。先调用 `MInt<id>::setMod(m)`，要求 $1<m<2^{31}$；同一 `id` 共享模数，换模后旧对象不可继续使用。加减乘可用合数模，除法／逆元要求质数模且除数非零；不同模数同时存在时用不同 `id`。

```c++
struct Barrett {
    uint32_t m;
    uint64_t im;
    Barrett(uint32_t m = 1) : m(m), im(uint64_t(-1) / m + 1) {}
    uint32_t mul(uint32_t a, uint32_t b) const {
        uint64_t z = uint64_t(a) * b;
        uint64_t q = __uint128_t(z) * im >> 64;
        uint32_t y = z - q * m;
        return y < m ? y : uint32_t(y + m);
    }
};

template <int id = 0> struct MInt {
    inline static Barrett B;
    uint32_t x;
    MInt(i64 v = 0) : x(norm(v)) {}
    static void setMod(int m) {
        assert(1 < m && m < (1LL << 31));
        B = Barrett(m);
    }
    static int getMod() {
        return B.m;
    }
    static uint32_t norm(int v) {
        v %= getMod();
        return v < 0 ? v + getMod() : v;
    }
    int val() const {
        return x;
    }
    MInt pow(int k) const {
        assert(k >= 0);
        MInt a = *this, r = 1;
        for (; k; k >>= 1, a *= a)
            if (k & 1) r *= a;
        return r;
    }
    MInt inv() const {
        assert(x);
        return pow(getMod() - 2);
    }
    MInt operator-() const { return x ? MInt(getMod() - x) : MInt(0); }
    MInt &operator+=(MInt v) {
        x += v.x;
        if (x >= B.m) x -= B.m;
        return *this;
    }
    MInt &operator-=(MInt v) {
        x = x >= v.x ? x - v.x : x + B.m - v.x;
        return *this;
    }
    MInt &operator*=(MInt v) {
        x = B.mul(x, v.x);
        return *this;
    }
    MInt &operator/=(MInt v) { return *this *= v.inv(); }
    friend MInt operator+(MInt a, MInt b) { return a += b; }
    friend MInt operator-(MInt a, MInt b) { return a -= b; }
    friend MInt operator*(MInt a, MInt b) { return a *= b; }
    friend MInt operator/(MInt a, MInt b) { return a /= b; }
    friend bool operator==(MInt a, MInt b) { return a.x == b.x; }
    friend bool operator!=(MInt a, MInt b) { return a.x != b.x; }
    friend istream &operator>>(istream &in, MInt &a) {
        i64 v;
        in >> v;
        a = v;
        return in;
    }
    friend ostream &operator<<(ostream &out, MInt a) { return out << a.x; }
};
```

### 扩展欧几里得 exgcd

求解形如 $a\cdot x + b\cdot y = \gcd(a,b)$ 的不定方程的任意一组解。


> 求解二元一次不定方程 $A\cdot x + B\cdot y = C$ 。

若 $d\mid c$，把特解乘 $c/d$ 得 $(x_0,y_0)$，通解为
$$
x=x_0+t\frac bd,\qquad y=y_0-t\frac ad,\qquad t\in\mathbb Z.
$$
系数为负时，先对绝对值求解，再改变对应解的符号。缩放特解可能需要 `i128`。

当 $a,b>0$ 时，正整数解对应
$$
\left\lceil\frac{1-x_0}{b/d}\right\rceil\le t\le
\left\lfloor\frac{y_0-1}{a/d}\right\rfloor.
$$
```cpp
auto calc = [&](int a, int b, int c) {
    int x, y, d = exgcd(a, b, x, y);
    if (c % d != 0) return {-1, -1};
    int x0 = x * c / d, y0 = y * c / d;
    return {x0, y0};
    auto norm = [%](int x, int m) {
        x %= m;
        return x < 0 ? x + m : x;
    };
    int X = norm(x0, b), Y = norm(y0, a);
    int l = (X - x0) / b, r = (y0 - Y) / a;
    // x 为最小正整数
    return {X, y - l * a};
    // y 为最小正整数
    return {x + r * b, Y};
    // x, y 均为正整数的组数
    int cnt = max(0LL, r - l + 1);
}
```

### 质因子分解

#### 筛法

`spf`、`maxp` 分别维护最小、最大质因子；`fact` 返回质因子与指数。

使用欧拉筛（线性筛），预处理时间复杂度 $\mathcal O(N)$ ，单次查询 $\mathcal O(\textrm{Prime Numer})$ 。

#### Pollard-Rho

`fact(x)` 要求 $x\ge1$。若最小质因子为 p，寻找一个非平凡因子的期望迭代数为 $O(\sqrt p)$，最坏规模约为 $O(x^{1/4})$，另计素性判定开销。第十三章保留 Montgomery 优化版。

```c++
int rho(int n) {
    if (n % 2 == 0) return 2;
    int x = 0, y = 0, c = 0, p = 1, d;
    auto f = [&](int x) {
        int v = mul(x, x, n) - c;
        return v < 0 ? v + n : v;
    };
    for (int l = 1;; l *= 2) {
        if (x == y) c = rng() % (n - 1) + 1, y = 0, l = 1;
        x = y;
        for (int i = 0; i < l; ++i) {
            y = f(y);
            if (x == y) break;
            if (int q = mul(p, abs(x - y), n)) p = q;
            if (i % 128 == 0 && (d = gcd(p, n)) > 1) return d;
        }
    }
}

vector<int> fact(int x) {
    vector<int> r;
    auto f = [&](auto f, int x) {
        if (x == 1) return;
        if (isprime(x)) return r.push_back(x);
        int y = rho(x);
        f(f, y), f(f, x / y);
    };
```

### 欧拉函数

#### 直接求解单个数的欧拉函数

$1$ 到 $N$ 中与 $N$ 互质数的个数称为欧拉函数，记作 $\varphi (N)$ 。求解欧拉函数的过程即为分解质因数的过程，复杂度 $\mathcal{O}(\sqrt{n})$ 。

```c++
int phi(int n) {
    int ans = n;
    for(int i = 2; i <= n / i; i ++) {
        if(n % i == 0) {
            ans = ans / i * (i - 1);
            while(n % i == 0) n /= i;
        }
    }
    if(n > 1) ans = ans / n * (n - 1);
    return ans;
} 
```

#### 求解 1 到 N 所有数的欧拉函数

利用上述第四条性质，我们可以快速递推出 $2-N$ 中每个数的欧拉函数，复杂度 $\mathcal{O}(N)$ ，而该算法**即是线性筛的算法**。

```c++
constexpr int N = 1E5;

vector<int> primes, spf(N + 1), phi(N + 1);

void init() {
    for (int i = 2; i <= N; ++i) {
        if (!spf[i]) spf[i] = i, primes.push_back(i), phi[i] = i - 1;
        for (auto p : primes) {
            if (p > N / i) break;
            int y = p * i;
            spf[y] = p;
            if (p == spf[i]) {
                phi[y] = phi[i] * p;
                break;
            }
            phi[y] = phi[i] * (p - 1);
        }
    }
}
```

#### 约数反演递推欧拉函数

由 $n=\sum_{d\mid n}\varphi(d)$ 扣除真因子的贡献，返回 $0$ 到 $n$ 的表。时间 $O(n\log n)$、空间 $O(n)$，线性筛仍是批量求值的首选。

```c++
vector<int> phiDivisors(int n) {

    iota(phi.begin(), phi.end(), 0LL);
    for (int i = 1; i <= n; i++)
        for (int j = i + i; j <= n; j += i) phi[j] -= phi[i];

}
```

### 离散对数 bsgs 与 exbsgs

以 $\mathcal O(\sqrt {P})$ 的复杂度求解 $a^x \equiv b(\bmod P)$  。其中标准 $\tt BSGS$ 算法不能计算 $a$ 与 $MOD$ 互质的情况，而 exbsgs 则可以。

```c++
namespace BSGS {
    int power(int a, int b, int m) {
        int r = 1 % m;
        a %= m;
        for (; b; b >>= 1, a = a * a % m)
            if (b & 1) r = r * a % m;
        return r;
    }

    int exgcd(int a, int b, int &x, int &y) {
        if (!b) {
            x = 1, y = 0;
            return a;
        }
        int d = exgcd(b, a % b, y, x);
        y -= a / b * x;
        return d;
    }
    int inv(int a, int m) {
        int x, y, d = exgcd(a, m, x, y);
        return d == 1 ? (x % m + m) % m : -1;
    }

    int bsgs(int a, int b, int p) {
        a %= p, b %= p;
        if (b == 1) return 0;

        int m = sqrtl(p) + 1, e = 1;
        map<int, int> mp;
        for (int j = 0; j < m; ++j) {
            mp.emplace(e, j);
            e = e * a % p;
        }

        int step = inv(power(a, m, p), p), cur = b;
        for (int i = 0; i <= m; ++i) {
            auto it = mp.find(cur);
            if (it != mp.end()) return i * m + it->second;
            cur = cur * step % p;
        }
        return -1;
    }

    int exbsgs(int a, int b, int p) {
        if (p == 1) return 0;
        a %= p, b %= p;
        if (a < 0) a += p;
        if (b < 0) b += p;
        if (b == 1) return 0;

        int k = 0, t = 1, g;
        while ((g = gcd(a, p)) > 1) {
            if (b % g) return -1;
            b /= g, p /= g, ++k;
            if (p == 1) return k;

            t = t * (a / g) % p % p;
            if (t == b) return k;
        }

        int it = inv(t, p);
        int x = bsgs(a, b * it % p, p);
        return x < 0 ? -1 : x + k;
    }
} // namespace BSGS

using namespace BSGS;
```

#### Pohlig–Hellman

`ph(g,h,p,fac)` 求最小非负离散对数，要求 `p` 为质数、`g` 为原根，`fac` 为 `p-1` 的完整 `{质因子,指数}` 分解；`h=0` 无解返回 `-1`。适合群阶光滑时，省略 `fac` 会试除分解，主体约 $O(\sum e(\sqrt q+\log p))$。

```c++
int ph(int g, int h, int p, const vector<pii> &fac) {
    g %= p;
    h %= p;
    if (g < 0) g += p;
    if (h < 0) h += p;
    if (!g || !h) return -1;
    int ord = p - 1, ans = 0, mod = 1;
    for (auto [q, e] : fac) {
        int a = power(g, ord / q, p), b = mysqrt(q) + 1, v = 1;
        unordered_map<int, int> mp;
        for (int j = 0; j < b; j++) {
            mp.emplace(v, j);
            v = mul(v, a, p);
        }
        int step = power(a, (q - b % q) % q, p);
        auto log = [&](int x) {
            for (int i = 0; i <= b; i++, x = mul(x, step, p)) {
                auto it = mp.find(x);
                if (it != mp.end() && i * b + it->second < q) return i * b + it->second;
            }
            return -1LL;
        };
        int x = 0, pw = 1;
        for (int i = 0; i < e; i++) {
            int v = mul(h, power(g, (ord - x) % ord, p), p);
            int d = log(power(v, ord / (pw * q), p));
            if (d < 0) return -1;
            x += d * pw;
            pw *= q;
        }
        int u;
        exgcd(mod, pw, u, v);
        int k = mul((x - ans) % pw, u, pw);
        ans += i128(mod) * k;
        mod *= pw;
    }
    return ans;
}
int ph(int g, int h, int p) {
    vector<pii> fac;
    int n = p - 1;
    for (int q = 2; q <= n / q; q++) if (n % q == 0) {
        int e = 0;
        do n /= q, e++; while (n % q == 0);
        fac.push_back({q, e});
    }
    if (n > 1) fac.push_back({n, 1});
    return ph(g, h, p, fac);
}
```

### 模平方根（Tonelli–Shanks）

`mysqrt(a, p)` 求 $x^2\equiv a\pmod p$，要求 `p` 为质数，`a` 允许为负数；无解返回 `-1`，否则返回较小的非负根，另一根为 `(p - x) % p`。

```c++
int mysqrt(int a, int p) {
    a %= p;
    if (a < 0) a += p;
    if (a == 0 || p == 2) return a;
    if (power(a, (p - 1) / 2, p) != 1) return -1;
    if (p % 4 == 3) {
        int x = power(a, p / 4 + 1, p);
        return min(x, p - x);
    }
    int q = p - 1, s = 0;
    while (q % 2 == 0) q /= 2, s++;
    int z = 2;
    while (power(z, (p - 1) / 2, p) == 1) z++;
    int x = power(a, (q + 1) / 2, p);
    int b = power(a, q, p), c = power(z, q, p);
    while (b != 1) {
        int i = 0;
        for (int v = b; v != 1; v = mul(v, v, p)) i++;
        int t = power(c, 1LL << (s - i - 1), p);
        x = mul(x, t, p);
        c = mul(t, t, p);
        b = mul(b, c, p);
        s = i;
    }
    return min(x, p - x);
}
```

#### 常规预处理

模数必须为质数，且 $n,m\leq MOD$。

```c++
struct Comb {
    int n;
    vector<Z> _fac, _ifac, _inv;

    Comb() : n{0}, _fac{1}, _ifac{1}, _inv{0} {}
    Comb(int n) : Comb() {
        init(n);
    }

    void init(int m) {
        if (m <= n) return;
        _fac.resize(m + 1);
        _ifac.resize(m + 1);
        _inv.resize(m + 1);
        for (int i = n + 1; i <= m; i++) {
            _fac[i] = _fac[i - 1] * i;
        }
        _ifac[m] = _fac[m].inv();
        for (int i = m; i > n; i--) {
            _ifac[i - 1] = _ifac[i] * i;
            _inv[i] = _ifac[i] * _fac[i - 1];
        }
        n = m;
    }
    Z fac(int m) {
        if (m > n) init(m);

        if (m > n) init(m);
        return _ifac[m];

        if (m > n) init(m);
        return _inv[m];

        return fac(n) * ifac(n - m);
```

#### ExLucas

求解组合数模任意非质数的情况，复杂度为 $\mathcal{O}(P\log P)$ 。

```cpp
int exL(int n, int m, int mod) {
    if (m < 0 || m > n || mod == 1) return 0;
    int ans = 0, x = mod;
    auto solve = [&](int p, int pk) {
        vector<int> f(pk + 1, 1);
        for (int i = 1; i <= pk; ++i) f[i] = mul(f[i - 1], i % p ? i : 1, pk);

        auto fac = [&](auto &&self, int x) -> int {
            if (!x) return 1;
            return mul(mul(power(f[pk], x / pk, pk), f[x % pk], pk), self(self, x / p), pk);
        };

        int e = 0, phi = pk / p * (p - 1);
        for (int y = n; y; y /= p) e += y / p;
        for (int y = m; y; y /= p) e -= y / p;
        for (int y = n - m; y; y /= p) e -= y / p;

        int den = mul(fac(fac, m), fac(fac, n - m), pk);
        int r = mul(mul(fac(fac, n), power(den, phi - 1, pk), pk), power(p, e, pk), pk);
        int M = mod / pk;
        int add = mul(mul(r, M, mod), power(M, phi - 1, pk), mod);
        ans = (ans + add) % mod;
    };
    for (int p = 2; p <= x / p; ++p) {
        if (x % p) continue;
        int pk = 1;
        while (x % p == 0) x /= p, pk *= p;
        solve(p, pk);
    }
    if (x > 1) solve(x, x);
    return ans;
}
```

#### 质因数指数法（任意正模数）

先筛出 $1$ 到上限 $n$ 的质数，再用阶乘的质因子指数计算组合数，无需求逆。适用于 $n$ 可预处理、模数可变或为合数的情况；预处理 $O(n)$，单次至多 $O(\pi(n)\log n)$。

```c++
struct FactorComb {
    int n;
    vector<int> primes, spf;
    FactorComb(int n) : n(n), spf(n + 1) {
        for (int i = 2; i <= n; i++) {
            if (!spf[i]) spf[i] = i, primes.push_back(i);
            for (auto p : primes) {
                if (p > spf[i] || p > n / i) break;
                spf[i * p] = p;
            }
        }
    }
    int C(int x, int y, int mod) const {
        assert(mod > 0 && x <= n);
        if (x < 0 || y < 0 || y > x || mod == 1) return 0;
        auto count = [](int x, int p) {
            int r = 0;
            for (; x; x /= p) r += x / p;
            return r;
        };
        auto power = [&](int a, int b) {
            int r = 1 % mod;
            a %= mod;
            for (; b; b >>= 1, a = i128(a) * a % mod)
                if (b & 1) r = i128(r) * a % mod;
            return r;
        };
        int r = 1 % mod;
        for (auto p : primes) {
            if (p > x) break;
            int e = count(x, p) - count(y, p) - count(x - y, p);
            r = i128(r) * power(p, e) % mod;
        }
        return r;
    }
};
```

#### 正向展开树状数组解

给定一个全排列，求出它是 1 ~ $n$ 所有全排列的第几个，答案对 $998244353$ 取模。

答案就是 $\sum_{i = 1}^{n} res_{a_i} (n - i)!$ 。$res_x$ 表示剩下的比 $x$ 小的数字的数量，通过**树状数组**处理。

```c++
#include <bits/stdc++.h>
using namespace std;
#define LL long long
const int mod = 998244353, N = 1e6 + 10;
LL fact[N];
struct fwt{
    LL n;
    vector <LL> a;
    fwt(LL n) : n(n), a(n + 1) {}
    LL sum(LL x){
        LL res = 0;
        for (; x; x -= x & -x)
            res += a[x];
        return res;
    }
    void add(LL x, LL k){
        for (; x <= n; x += x & -x)
            a[x] += k;
    }
    LL query(LL x, LL y){
        return sum(y) - sum(x - 1);
    }
};
int main(){
    ios::sync_with_stdio(false);cin.tie(0);
    LL n;
    cin >> n;
    fwt a(n);
    fact[0] = 1;
    for (int i = 1; i <= n; i ++ ){
        fact[i] = fact[i - 1] * i % mod;
        a.add(i, 1);
    }
    LL ans = 0;
    for (int i = 1; i <= n; i ++ ){
        LL x;
        cin >> x;
        ans = (ans + a.query(1, x - 1) * fact[n - i] % mod ) % mod;
        a.add(x, -1);
    }
    cout << (ans + 1) % mod << "\n";
    return 0;
}
```

### 莫比乌斯函数/反演

莫比乌斯函数定义：$\displaystyle {\mu(n) = \begin{cases} 1 &n = 1 \\ (-1)^k &n = \prod_{i = 1}^k p_i \text{ 且 } p_i \text{ 互质 } \\ 0 &else \end{cases}}$ 。

> 莫比乌斯函数性质：对于任意正整数 $n$ 满足 $\displaystyle {\sum_{d|n}\mu(d) = \begin{cases} 1 & n = 1 \\ 0 & n \neq 1\end{cases}}$ ；$\displaystyle {\sum_{d|n} \frac{\mu(d)}{d} = \frac{\varphi(n)}{n}}$ 。 

莫比乌斯反演定义：定义：$F(n)$ 和 $f(n)$ 是定义在非负整数集合上的两个函数，并且满足 $\displaystyle F(n) = \sum_{d|n}f(d)$ ，可得 $\displaystyle f(n) = \sum_{d|n}\mu(d)F(\left \lfloor \frac{n}{d} \right \rfloor)$ 。

```c++
const int N = 5e4 + 10;
bool st[N];
int mu[N], prime[N], cnt, sum[N];
void getMu() {
    mu[1] = 1;
    for (int i = 2; i <= N - 10; i++) {
        if (!st[i]) {
            prime[++cnt] = i;
            mu[i] = -1;
        }
        for (int j = 1; j <= cnt && i * prime[j] <= N - 10; j++) {
            st[i * prime[j]] = true;
            if (i % prime[j] == 0) {
                mu[i * prime[j]] = 0;
                break;
            }
            mu[i * prime[j]] = -mu[i];
        }
    }
    for (int i = 1; i <= N - 10; i++) {
        sum[i] = sum[i - 1] + mu[i];
    }
}
void solve() {
    int n, m, k; cin >> n >> m >> k;
    n = n / k, m = m / k;
    if (n < m) swap(n, m);
    LL ans = 0;
    for (int i = 1, j = 0; i <= m; i = j + 1) {
        j = min(n / (n / i), m / (m / i));
        ans += (LL)(sum[j] - sum[i - 1]) * (n / i) * (m / i);
    }
    cout << ans << "\n";
}
int main() {
    getMu();
    int T; cin >> T;
    while (T--) solve();
}
```

### 杜教筛

修改常量 `N` 设置预筛范围（至少为 `1`），先调用 `init()`，再用 `getMu(n)` 求 $\sum_{i=1}^n\mu(i)$，按 `Z` 的模数计算，要求 `n >= 0`；多次查询共用缓存。

对单次大查询 $n$，通常预筛至 $n^{2/3}$，连同预处理时间为 $O(n^{2/3})$，空间为 $O(N+n/N)$，也可按内存调整 `N`。

```c++
constexpr int N = 5E5;
vector<Z> pre(N + 1);
unordered_map<int, Z> sMu;

void init() {
    vector<int> mu(N + 1), prime;
    vector<bool> vis(N + 1);
    mu[1] = 1;
    for (int i = 2; i <= N; i++) {
        if (!vis[i]) {
            prime.push_back(i);
            mu[i] = -1;
        }
        for (int p : prime) {
            if (p > N / i) break;
            int j = i * p;
            vis[j] = true;
            if (i % p == 0) break;
            mu[j] = -mu[i];
        }
    }
    pre[0] = 0;
    for (int i = 1; i <= N; i++) pre[i] = pre[i - 1] + mu[i];
    sMu.clear();
}

Z getMu(int n) {
    if (n <= N) return pre[n];
    if (auto it = sMu.find(n); it != sMu.end()) return it->second;
    Z ans = 1;
    for (int l = 2, r; l <= n; l = r + 1) {
        r = n / (n / l);
        ans -= Z(r - l + 1) * getMu(n / l);
        if (r == n) break;
    }
    return sMu[n] = ans;
}
```

### Lehmer 质数计数

先 `init()`，`getPi(x)` 返回不超过 `x` 的质数个数，要求 $0\le x\le N^2$。预处理空间 $O(N+KM)$，另有记忆化缓存，适合筛表无法直接覆盖的大范围查询。

```c++
constexpr int N = 5000000, M = 32768, K = 64;
vector<int> pri, pc(N + 1);
vector pre(K, vector<int>(M));
unordered_map<int, int> mp;
void init() {
    vector<bool> vis(N + 1);
    pri.clear();
    mp.clear();
    for (int i = 2; i <= N; i++) {
        if (!vis[i]) pri.push_back(i);
        for (int p : pri) {
            if (p > N / i) break;
            vis[p * i] = true;
            if (i % p == 0) break;
        }
        pc[i] = pc[i - 1] + !vis[i];
    }
    iota(pre[0].begin(), pre[0].end(), 0);
    for (int a = 1; a < K; a++)
        for (int x = 0; x < M; x++) pre[a][x] = pre[a - 1][x] - pre[a - 1][x / pri[a - 1]];
}
int phi(int x, int a) {
    if (!a || !x) return x;
    if (a < K && x < M) return pre[a][x];
    if (x <= pri[a - 1]) return 1;
    if (x <= N && pri[a - 1] > x / pri[a - 1]) return pc[x] - a + 1;
    return phi(x, a - 1) - phi(x / pri[a - 1], a - 1);
}
int getPi(int x) {
    if (x <= N) return pc[x];
    if (mp.count(x)) return mp[x];
    int t = cbrtl(x);
    while (i128(t + 1) * (t + 1) * (t + 1) <= x) t++;
    while (i128(t) * t * t > x) t--;
    int a = getPi(mysqrt(mysqrt(x))), b = getPi(mysqrt(x)), c = getPi(t);
    int r = phi(x, a) + (b + a - 2) * (b - a + 1) / 2;
    for (int i = a; i < b; i++) {
        int y = x / pri[i];
        r -= getPi(y);
        if (i < c) {
            int z = getPi(mysqrt(y));
            for (int j = i; j < z; j++) r -= getPi(y / pri[j]) - j;
        }
    }
    return mp[x] = r;
}
```

### 类欧几里得求和

`floorSum(n, m, a, b)` 计算 $\displaystyle\sum_{i=0}^{n-1}\left\lfloor\frac{ai+b}{m}\right\rfloor$，要求 `n >= 0`、`m > 0`，允许 `a`、`b` 为负数。这里求和从 `0` 开始，共 `n` 项，负数按向下取整计算。

时间 $O(\log m)$，空间 $O(1)$；参数使用 `int`，返回值及必要的中间运算使用 `i128`，中间值与结果均须在 `i128` 范围内。

```c++
i128 floorSum(int n, int m, int a, int b) {
    i128 ans = 0;
    auto norm = [&](int &x) {
        int q = x / m;
        x %= m;
        if (x < 0) x += m, q--;
        return q;
    };
    while (true) {
        ans += i128(n) * (n - 1) / 2 * norm(a) + i128(n) * norm(b);
        i128 y = i128(a) * n + b;
        if (y < m) break;
        n = y / m;
        b = y % m;
        swap(a, m);
    }
    return ans;
}
```

#### 加权类欧几里得

`Euclid::ask(a,b,c,n)` 返回 $\sum y_i$、$\sum i y_i$、$\sum y_i^2$，其中 $y_i=\lfloor(ai+b)/c\rfloor$，$0\le i\le n$。要求 `a,b,n>=0,c>0`、中间商能用 `int` 表示、模数中 `2,6` 可逆，时间 $O(\log\max(a,c))$。

```c++
struct Euclid {
    struct Ans {
        Z s, t, u;
    };

    static Z sum1(int n) {
        return Z(n) * (Z(n) + 1) / Z(2);
    }

    static Z sum2(int n) {
        return Z(n) * (Z(n) + 1) * (Z(n) * 2 + 1) / Z(6);
    }

    static Ans ask(int a, int b, int c, int n) {
        assert(a >= 0 && b >= 0 && c > 0 && n >= 0);
        assert(Z(2) != Z(0) && Z(6) != Z(0));
        if (a >= c || b >= c) {
            int q = a / c, p = b / c;
            Ans x = ask(a % c, b % c, c, n);
            Z m = Z(n) + 1, s = sum1(n), t = sum2(n);
            Z qz = q, pz = p;
            return {
                qz * s + pz * m + x.s,
                qz * t + pz * s + x.t,
                qz * qz * t + pz * pz * m + Z(2) * qz * pz * s + Z(2) * qz * x.t + Z(2) * pz * x.s + x.u,
            };
        }
        if (!a) return {};
        int m = (i128(a) * n + b) / c;
        if (!m) return {};
        Ans x = ask(c, c - b - 1, a, m - 1);
        Z mz = m, nz = n;
        return {
            nz * mz - x.s,
            mz * sum1(n) - (x.u + x.s) / Z(2),
            nz * mz * mz - Z(2) * x.t - x.s,
        };
    }
};
```

### 线性基（高斯消元法）

设向量长度为 $N$（一般取 $63$），总数为 $M$，时间复杂度为 $\mathcal O(NM)$ 。

```c++
struct LB { // Linear Basis
    using int = long long;
    const int BASE = 63;
    vector<int> d, p;
    int cnt, flag;

    LB() {
        d.resize(BASE + 1);
        p.resize(BASE + 1);
        cnt = flag = 0;
    }
    bool insert(int val) {
        for (int i = BASE - 1; i >= 0; i--) {
            if (val & (1ll << i)) {
                if (!d[i]) {
                    d[i] = val;
                    return true;
                }
                val ^= d[i];
            }
        }
        flag = 1; //可以异或出0
        return false;
    }
    bool check(int val) { // 判断 val 是否能被异或得到
        for (int i = BASE - 1; i >= 0; i--) {
            if (val & (1ll << i)) {
                if (!d[i]) {
                    return false;
                }
                val ^= d[i];
            }
        }
        return true;
    }
    int ask_max() {
        int res = 0;
        for (int i = BASE - 1; i >= 0; i--) {
            if ((res ^ d[i]) > res) res ^= d[i];
        }
        return res;
    }
    int ask_min() {
        if (flag) return 0; // 特判 0
        for (int i = 0; i <= BASE - 1; ++i) {
            if (d[i]) return d[i];
        }
    }
    void rebuild() { // 第k小值独立预处理
        for (int i = BASE - 1; i >= 0; i--) {
            for (int j = i - 1; j >= 0; j--) {
                if (d[i] & (1ll << j)) d[i] ^= d[j];
            }
        }
        for (int i = 0; i <= BASE - 1; ++i) {
            if (d[i]) p[cnt++] = d[i];
        }
    }
    int kthquery(int k) { // 查询能被异或得到的第 k 小值, 如不存在则返回 -1
        if (flag) k--; // 特判 0, 如果不需要 0, 直接删去
        if (!k) return 0;
        int res = 0;
        if (k >= (1ll << cnt)) return -1;
        for (int i = BASE - 1; i >= 0; i--) {
            if (k & (1LL << i)) res ^= p[i];
        }
        return res;
    }
    void Merge(const LB &b) { // 合并两个线性基
        for (int i = BASE - 1; i >= 0; i--) {
            if (b.d[i]) {
                insert(b.d[i]);
            }
        }
    }
};
```

### 矩阵

#### 矩阵操作

`Matrix<T> a(n, m)` 创建零矩阵，`a[i][j]` 访问元素；支持矩阵加减乘、数乘、复合赋值，`trans()` 返回转置，`unit(n)` 创建单位矩阵。取模用 `Matrix<Z>`，普通整数用 `Matrix<int>` 并保证运算不溢出；原模板[参考提交](https://ac.nowcoder.com/acm/contest/view-submission?submissionId=48594258)。

方阵的 `pow(k)` 返回 $A^k$，`sum(k)` 返回 $I+A+\cdots+A^{k-1}$，要求 `k >= 0`；`k = 0` 时分别返回单位矩阵、零矩阵。两者时间均为 $O(n^3\log(k+2))$，空间 $O(n^2)$。

```c++
template<class T> struct Matrix {
    int n, m;
    vector<vector<T>> a;
    Matrix(int n, int m) : n(n), m(m), a(n + 1, vector<T>(m + 1)) {}
    vector<T> &operator[](int i) { return a[i]; }
    const vector<T> &operator[](int i) const { return a[i]; }

    static Matrix unit(int n) {
        Matrix r(n, n);
        for (int i = 1; i <= n; i++) r[i][i] = 1;
        return r;
    }
    Matrix trans() const {
        Matrix r(m, n);
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++) r[j][i] = a[i][j];
        return r;
    }
    Matrix &operator+=(const Matrix &b) {
        assert(n == b.n && m == b.m);
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++) a[i][j] += b[i][j];
        return *this;
    }
    Matrix &operator-=(const Matrix &b) {
        assert(n == b.n && m == b.m);
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++) a[i][j] -= b[i][j];
        return *this;
    }
    Matrix &operator*=(T x) {
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++) a[i][j] *= x;
        return *this;
    }
    friend Matrix operator+(Matrix a, const Matrix &b) { a += b; return a; }
    friend Matrix operator-(Matrix a, const Matrix &b) { a -= b; return a; }
    friend Matrix operator*(Matrix a, T x) { a *= x; return a; }
    friend Matrix operator*(T x, Matrix a) { a *= x; return a; }

    Matrix operator*(const Matrix &b) const {
        assert(m == b.n);
        Matrix c(n, b.m);
        for (int i = 1; i <= n; i++)
            for (int k = 1; k <= m; k++)
                for (int j = 1; j <= b.m; j++)
                    c[i][j] += a[i][k] * b[k][j];
        return c;
    }
    Matrix &operator*=(const Matrix &b) { return *this = *this * b; }
    Matrix pow(int k) const {
        assert(n == m && k >= 0);
        Matrix b = *this, r = unit(n);
        while (k) {
            if (k & 1) r *= b;
            k >>= 1;
            if (k) b *= b;
        }
        return r;
    }
    Matrix sum(int k) const {
        assert(n == m && k >= 0);
        Matrix b = *this, c = unit(n), p = unit(n), r(n, n);
        while (k) {
            if (k & 1) {
                r += p * c;
                if (k > 1) p *= b;
            }
            k >>= 1;
            if (k) {
                c += b * c;
                if (k > 1) b *= b;
            }
        }
        return r;
    }
};
```

#### Min-plus 矩阵

以 `min` 为加法、普通加法为乘法，邻接矩阵 `pow(k)[u][v]` 为恰走 `k` 条边的最小代价；缺边用 `inf`，有限路径和须在 $(-\mathrm{inf},\mathrm{inf})$ 内。时间 $O(n^3\log(k+2))$，空间 $O(n^2)$。

```c++
struct MinMatrix {
    static constexpr int inf = 1LL << 60;
    int n;
    vector<vector<int>> a;
    MinMatrix(int n) : n(n), a(n + 1, vector<int>(n + 1, inf)) {}
    vector<int> &operator[](int i) { return a[i]; }
    const vector<int> &operator[](int i) const { return a[i]; }
    static MinMatrix unit(int n) {
        MinMatrix r(n);
        for (int i = 1; i <= n; i++) r[i][i] = 0;
        return r;
    }
    MinMatrix operator+(const MinMatrix &b) const {
        MinMatrix r(n);
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= n; j++) r[i][j] = min(a[i][j], b[i][j]);
        return r;
    }
    MinMatrix operator*(const MinMatrix &b) const {
        MinMatrix r(n);
        for (int i = 1; i <= n; i++)
            for (int k = 1; k <= n; k++) if (a[i][k] != inf)
                for (int j = 1; j <= n; j++) if (b[k][j] != inf)
                    r[i][j] = min(r[i][j], a[i][k] + b[k][j]);
        return r;
    }
    MinMatrix pow(int k) const {
        MinMatrix b = *this, r = unit(n);
        while (k) {
            if (k & 1) r = r * b;
            k >>= 1;
            if (k) b = b * b;
        }
        return r;
    }
};
```

#### 高斯消元

以下函数独立使用，下标均从 `0` 开始；传入增广矩阵时，最后一列为右端常数。`Matrix<T>` 的第 `0` 行、列是占位，不能直接把整个 `a.a` 传入。

##### 实数方程组

`gauss(a, m, ans, eps)` 求解 `m` 个未知数的实数方程组，不修改输入；返回自由变量数，`-1` 表示无解并清空 `ans`，否则给出自由变量置 `0` 的一组解，返回 `0` 即唯一解。每行需有 `m + 1` 个元素，`eps` 按数据范围调整。

```c++
int gauss(vector<vector<ld>> a, int m, vector<ld> &ans, ld eps = 1E-9) {
    int n = a.size(), r = 0;
    vector<int> pos;
    ans.clear();
    for (int c = 0; c < m && r < n; c++) {
        int p = r;
        for (int i = r + 1; i < n; i++)
            if (abs(a[i][c]) > abs(a[p][c])) p = i;
        if (abs(a[p][c]) <= eps) continue;
        swap(a[p], a[r]);
        ld v = a[r][c];
        for (int j = c; j <= m; j++) a[r][j] /= v;
        for (int i = 0; i < n; i++) {
            if (i == r || abs(a[i][c]) <= eps) continue;
            ld x = a[i][c];
            for (int j = c; j <= m; j++) a[i][j] -= x * a[r][j];
        }
        pos.push_back(c);
        r++;
    }
    for (int i = r; i < n; i++) {
        if (abs(a[i][m]) > eps) return -1;
    }
    ans.assign(m, 0);
    for (int i = 0; i < r; i++) ans[pos[i]] = a[i][m];
    return m - r;
}
```

##### 异或方程组（GF(2)）

用于求解系数与未知数均为 `0/1` 的异或方程组，`a[i][0..m-1]` 为系数，`a[i][m]` 为右端常数，`bitset<M>` 需满足 `M > m`；这里下标从 `0` 开始。`xorGauss<M>(a, m, ans)` 返回自由变量数 `k`：`-1` 表示无解并清空 `ans`，否则共有 $2^k$ 个解，`ans` 给出自由变量置 `0` 的一组解，`k = 0` 时即为唯一解。

原矩阵 `a` 不会被修改。设方程数为 $n$、位集底层字长为 $w$，消元时间为 $O(nm\lceil M/w\rceil)$，`M` 尽量贴近所需位数。

```c++
template<int M> int xorGauss(vector<bitset<M>> a, int m, vector<int> &ans) {
    int n = a.size(), r = 0;
    vector<int> pos;
    ans.clear();
    for (int c = 0; c < m && r < n; c++) {
        int p = r;
        while (p < n && !a[p][c]) p++;
        if (p == n) continue;
        swap(a[p], a[r]);
        for (int i = 0; i < n; i++) {
            if (i != r && a[i][c]) a[i] ^= a[r];
        }
        pos.push_back(c);
        r++;
    }
    for (int i = r; i < n; i++) {
        if (a[i][m]) return -1;
    }
    ans.assign(m, 0);
    for (int i = 0; i < r; i++) ans[pos[i]] = a[i][m];
    return m - r;
}
```

##### 模质数方程组

求解模质数 `mod` 下的线性方程组，`a` 为从 `0` 开始、每行 `m + 1` 列的增广矩阵，输入允许负数。

`modGauss(a, m, mod, ans)` 不修改原矩阵，返回自由变量数 `k`：`-1` 表示无解并清空 `ans`，否则共有 $\mathrm{mod}^k$ 个解，`ans` 给出自由变量置 `0` 的一组解，`k = 0` 时即为唯一解。

```c++
int modGauss(vector<vector<int>> a, int m, int mod, vector<int> &ans) {
    int n = a.size(), r = 0;
    vector<int> pos;
    ans.clear();
    for (auto &v : a) {
        for (int &x : v) {
            x %= mod;
            if (x < 0) x += mod;
        }
    }
    for (int c = 0; c < m && r < n; c++) {
        int p = r;
        while (p < n && !a[p][c]) p++;
        if (p == n) continue;
        swap(a[p], a[r]);
        int iv = power(a[r][c], mod - 2, mod);
        for (int j = c; j <= m; j++) a[r][j] = mul(a[r][j], iv, mod);
        for (int i = 0; i < n; i++) {
            int z = a[i][c];
            if (i == r || !z) continue;
            for (int j = c; j <= m; j++) {
                a[i][j] -= mul(z, a[r][j], mod);
                if (a[i][j] < 0) a[i][j] += mod;
            }
        }
        pos.push_back(c);
        r++;
    }
    for (int i = r; i < n; i++) {
        if (a[i][m]) return -1;
    }
    ans.assign(m, 0);
    for (int i = 0; i < r; i++) ans[pos[i]] = a[i][m];
    return m - r;
}
```


##### 矩阵的秩（模质数）

`rk(a, m, mod)` 返回模质数 `mod` 下矩阵的秩，每行有 `m` 个元素。给齐次方程组补上全零常数列后，秩等于未知数个数减去自由变量数。

```c++
int rk(vector<vector<int>> a, int m, int mod) {
    for (auto &v : a) v.push_back(0);
    vector<int> ans;
    return m - modGauss(move(a), m, mod, ans);
}
```

##### 逆矩阵（模质数）

`inv(a, mod, b)` 求方阵 `a` 在模质数 `mod` 下的逆矩阵，输入允许负数且不会被修改；成功返回 `true` 并写入 `b`，不可逆返回 `false` 并清空 `b`。零阶矩阵返回 `true`，时间 $O(n^3+n\log\mathrm{mod})$，空间 $O(n^2)$。

```c++
bool inv(vector<vector<int>> a, int mod, vector<vector<int>> &b) {
    int n = a.size();
    b.assign(n, vector<int>(n));
    for (int i = 0; i < n; i++) {
        b[i][i] = 1;
        for (int &x : a[i]) {
            x %= mod;
            if (x < 0) x += mod;
        }
    }
    for (int c = 0; c < n; c++) {
        int p = c;
        while (p < n && !a[p][c]) p++;
        if (p == n) {
            b.clear();
            return false;
        }
        swap(a[p], a[c]);
        swap(b[p], b[c]);
        int v = power(a[c][c], mod - 2, mod);
        for (int j = 0; j < n; j++) {
            a[c][j] = mul(a[c][j], v, mod);
            b[c][j] = mul(b[c][j], v, mod);
        }
        for (int i = 0; i < n; i++) {
            int x = a[i][c];
            if (i == c || !x) continue;
            for (int j = 0; j < n; j++) {
                a[i][j] -= mul(x, a[c][j], mod);
                b[i][j] -= mul(x, b[c][j], mod);
                if (a[i][j] < 0) a[i][j] += mod;
                if (b[i][j] < 0) b[i][j] += mod;
            }
        }
    }
    return true;
}
```

##### 行列式（模质数）

`det(a, mod)` 返回方阵 `a` 的行列式模质数 `mod` 的值，不修改原矩阵；输入允许负数，零阶行列式为 `1`。

```c++
int det(vector<vector<int>> a, int mod) {
    int n = a.size(), ans = 1;
    for (auto &v : a) {
        for (int &x : v) {
            x %= mod;
            if (x < 0) x += mod;
        }
    }
    for (int c = 0; c < n; c++) {
        int p = c;
        while (p < n && !a[p][c]) p++;
        if (p == n) return 0;
        if (p != c) {
            swap(a[p], a[c]);
            ans = mod - ans;
        }
        ans = mul(ans, a[c][c], mod);
        int iv = power(a[c][c], mod - 2, mod);
        for (int i = c + 1; i < n; i++) {
            if (!a[i][c]) continue;
            int x = mul(a[i][c], iv, mod);
            for (int j = c; j < n; j++) {
                a[i][j] -= mul(x, a[c][j], mod);
                if (a[i][j] < 0) a[i][j] += mod;
            }
        }
    }
    return ans;
}
```

##### 矩阵树定理

`kirch(n, e, mod)` 求无向图所有生成树的边权乘积之和，模数 `mod` 必须为质数；点号为 `1..n`（$n\ge 1$），每条边为 `{u, v, w}`，边权全设为 `1` 即得到生成树数量。支持重边和负权，自环忽略；不连通图返回 `0`，单点图返回 `1`，答案为 `0` 也可能只是模意义下为零，不能据此判断连通性。

代码直接构造删去第 $n$ 行、第 $n$ 列的拉普拉斯余子式，并调用上面的 `det`；空间 $O(n^2)$，时间 $O(n^3+n\log\mathrm{mod}+|e|)$。

```c++
int kirch(int n, const vector<array<int, 3>> &e, int mod) {
    vector<vector<int>> a(n - 1, vector<int>(n - 1));
    auto add = [&](int u, int v, int w) {
        if (u == n || v == n) return;
        int &x = a[u - 1][v - 1];
        x = (i128(x) + w + mod) % mod;
    };
    for (auto [u, v, w] : e) {
        if (u == v) continue;
        w %= mod;
        if (w < 0) w += mod;
        add(u, u, w);
        add(v, v, w);
        add(u, v, -w);
        add(v, u, -w);
    }
    return det(move(a), mod);
}
```

### 线性规划：两阶段单纯形

`LP(a,b,c).work()` 求 $\max c\cdot x$，满足 $Ax\le b,x\ge0$；数组从 `0` 开始，返回 `{type,val,x}`，`type` 为 `0` 无解、`1` 最优、`2` 无界。浮点精度按数据量级调整 `eps`，最坏指数复杂度，不求整数解。

```c++
struct LP {
    struct Res { int type; ld val; vector<ld> x; };
    static constexpr ld eps = 1e-12L;
    int n, m;
    vector<int> bas, nb;
    vector<vector<ld>> a;

    LP(const vector<vector<ld>>& A, const vector<ld>& b,
                             const vector<ld>& c)
        : n(b.size()), m(c.size()), bas(n),
          nb(m + 1), a(n + 2, vector<ld>(m + 2, 0)) {
        assert(A.size() == n);
        for (int i = 0; i < n; ++i) {
            assert(A[i].size() == m);
            for (int j = 0; j < m; ++j) a[i][j] = A[i][j];
        }
        for (int i = 0; i < n; ++i) {
            bas[i] = m + i;
            a[i][m] = -1;
            a[i][m + 1] = b[i];
        }
        for (int j = 0; j < m; ++j) {
            nb[j] = j;
            a[n][j] = -c[j];
        }
        nb[m] = -1;
        a[n + 1][m] = 1;
    }

    void pivot(int row, int c) {
        ld inv = 1 / a[row][c];
        for (int i = 0; i < n + 2; ++i) {
            if (i == row) continue;
            for (int j = 0; j < m + 2; ++j) {
                if (j == c) continue;
                a[i][j] -= a[row][j] * a[i][c] * inv;
            }
        }
        for (int j = 0; j < m + 2; ++j) if (j != c) a[row][j] *= inv;
        for (int i = 0; i < n + 2; ++i) if (i != row) a[i][c] *= -inv;
        a[row][c] = inv;
        swap(bas[row], nb[c]);
    }

    bool simplex(int ph) {
        int obj = ph == 1 ? n + 1 : n;
        while (true) {
            int in = -1;
            for (int c = 0; c <= m; c++) {
                if ((ph == 2 && nb[c] == -1) || a[obj][c] >= -eps) continue;
                if (in == -1 || nb[c] < nb[in]) in = c;
            }
            if (in == -1) return true;

            int out = -1;
            for (int row = 0; row < n; ++row) {
                if (a[row][in] <= eps) continue;
                if (out == -1 ||
                    a[row][m + 1] / a[row][in] <
                        a[out][m + 1] / a[out][in] - eps ||
                    (fabsl(a[row][m + 1] / a[row][in] -
                           a[out][m + 1] / a[out][in]) <= eps &&
                     bas[row] < bas[out])) {
                    out = row;
                }
            }
            if (out == -1) return false;
            pivot(out, in);
        }
    }

    Res work() {
        if (n == 0) {
            for (int j = 0; j < m; ++j)
                if (a[n][j] < -eps) return {2, 0, {}};
            return {1, 0, vector<ld>(m, 0)};
        }

        int row = 0;
        for (int i = 1; i < n; ++i)
            if (a[i][m + 1] < a[row][m + 1]) row = i;
        if (a[row][m + 1] < -eps) {
            pivot(row, m);
            if (!simplex(1) || a[n + 1][m + 1] < -eps) {
                return {0, 0, {}};
            }
            if (fabsl(a[n + 1][m + 1]) > eps) {
                return {0, 0, {}};
            }
            for (int i = 0; i < n; ++i) {
                if (bas[i] != -1) continue;
                int c = -1;
                for (int j = 0; j <= m; j++)
                    if (fabsl(a[i][j]) > eps && (c == -1 || nb[j] < nb[c])) c = j;
                if (c != -1) pivot(i, c);
            }
        }
        if (!simplex(2)) return {2, 0, {}};
        vector<ld> sol(m, 0);
        for (int i = 0; i < n; ++i) {
            if (0 <= bas[i] && bas[i] < m) sol[bas[i]] = a[i][m + 1];
        }
        return {1, a[n][m + 1], sol};
    }
};
```

### 自适应 Simpson 积分

`simpson(f,l,r,eps)` 计算定积分，`eps>0` 是绝对误差目标；函数需足够光滑，奇点、间断与高频振荡须另行分段。递归深度耗尽时返回当前估值，不保证达到误差目标。

```c++
template<class F> ld simpson(F f, ld l, ld r, ld eps = 1e-10L, int dep = 30) {
    if (l == r) return 0;
    if (l > r) return -simpson(f, r, l, eps, dep);
    auto calc = [](ld a, ld b, ld x, ld y, ld z) {
        return (b - a) * (x + 4 * y + z) / 6;
    };
    auto dfs = [&](auto &&self, ld a, ld b, ld x, ld y, ld z, ld s, ld e, int d) -> ld {
        ld m = (a + b) / 2, u = f((a + m) / 2), v = f((m + b) / 2);
        ld p = calc(a, m, x, u, y), q = calc(m, b, y, v, z);
        ld err = p + q - s;
        if (!d || fabsl(err) <= 15 * e) return p + q + err / 15;
        return self(self, a, m, x, u, y, p, e / 2, d - 1)
             + self(self, m, b, y, v, z, q, e / 2, d - 1);
    };
    ld x = f(l), y = f((l + r) / 2), z = f(r);
    return dfs(dfs, l, r, x, y, z, calc(l, r, x, y, z), eps, dep);
}
```
