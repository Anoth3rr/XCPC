### 最大公约数 `gcd` 

$\tt{GCC} 环境下 $`std::gcd` 和 `__gcd` 使用的都是 $\tt{Stein}$ 算法，单次时间与欧几里得算法几乎没有差别，大量运算时常数略优，相比于欧几里得算法不容易被卡掉。但是在极端环境，如 $\gcd(10^{18}, 1)$ ，表现略逊于欧几里得算法。

#### 位运算优化

手写 $\tt {Stein}$ 算法。**略快于内置函数，用于卡常。**

```cpp
int gcd(int a, int b) {
    if (!a || !b) return a | b;
    int t = __builtin_ctzll(a | b);
    a >>= __builtin_ctzll(a);
    while (b) {
        b >>= __builtin_ctzll(b);
        if (a > b) swap(a, b);
        b -= a;
    }
    return a << t;
}
```

#### 最小公倍数

```cpp
int lcm(int x, int y) { return x / gcd(x, y) * y; }
```

### 大整数类（高精度计算）

保留十进制读写、四则运算、余数、gcd/lcm 与 Karatsuba 乘法；零统一为正零。小整数乘除要求其绝对值不超过 $2^{31}-1$，除数非零；高精 gcd 的 Stein 备选见第十三章。

```c++
const int base = 1000000000;
const int base_digits = 9; // 分解为九个数位一个数字
struct bigint {
    vector<int> a;
    int sign;

    bigint() : sign(1) {}
    bigint operator-() const {
        bigint res = *this;
        if (!res.a.empty()) res.sign = -sign;
        return res;
    }
    bigint(long long v) {
        *this = v;
    }
    bigint(const string &s) {
        read(s);
    }
    void operator=(const bigint &v) {
        sign = v.sign;
        a = v.a;
    }
    void operator=(long long v) {
        a.clear();
        sign = v < 0 ? -1 : 1;
        u64 x = v < 0 ? -u64(v) : u64(v);
        for (; x; x /= base) a.push_back(x % base);
    }

    // 基础加减乘除
    bigint operator+(const bigint &v) const {
        if (sign == v.sign) {
            bigint res = v;
            for (int i = 0, carry = 0; i < max(a.size(), v.a.size()) || carry; ++i) {
                if (i == res.a.size()) {
                    res.a.push_back(0);
                }
                res.a[i] += carry + (i < a.size() ? a[i] : 0);
                carry = res.a[i] >= base;
                if (carry) {
                    res.a[i] -= base;
                }
            }
            return res;
        }
        return *this - (-v);
    }
    bigint operator-(const bigint &v) const {
        if (sign == v.sign) {
            if (abs() >= v.abs()) {
                bigint res = *this;
                for (int i = 0, carry = 0; i < v.a.size() || carry; ++i) {
                    res.a[i] -= carry + (i < v.a.size() ? v.a[i] : 0);
                    carry = res.a[i] < 0;
                    if (carry) {
                        res.a[i] += base;
                    }
                }
                res.trim();
                return res;
            }
            return -(v - *this);
        }
        return *this + (-v);
    }
    void operator*=(int v) {
        check(v);
        for (int i = 0, carry = 0; i < a.size() || carry; ++i) {
            if (i == a.size()) {
                a.push_back(0);
            }
            long long cur = a[i] * (long long)v + carry;
            carry = (int)(cur / base);
            a[i] = (int)(cur % base);
        }
        trim();
    }
    void operator/=(int v) {
        check(v);
        for (int n = a.size(), i = n - 1, rem = 0; i >= 0; --i) {
            long long cur = a[i] + rem * (long long)base;
            a[i] = (int)(cur / v);
            rem = (int)(cur % v);
        }
        trim();
    }
    int operator%(int v) const {
        if (v < 0) {
            v = -v;
        }
        int m = 0;
        for (int n = a.size(), i = n - 1; i >= 0; --i) {
            m = (a[i] + m * (long long)base) % v;
        }
        return m * sign;
    }

    void operator+=(const bigint &v) {
        *this = *this + v;
    }
    void operator-=(const bigint &v) {
        *this = *this - v;
    }
    bigint operator*(int v) const {
        bigint res = *this;
        res *= v;
        return res;
    }
    bigint operator/(int v) const {
        bigint res = *this;
        res /= v;
        return res;
    }
    void operator%=(const int &v) {
        *this = *this % v;
    }

    bool operator<(const bigint &v) const {
        if (sign != v.sign) return sign < v.sign;
        int n = a.size(), m = v.a.size();
        if (n != m) return sign > 0 ? n < m : n > m;
        for (int n = a.size(), i = n - 1; i >= 0; i--)
            if (a[i] != v.a[i]) return a[i] * sign < v.a[i] * sign;
        return false;
    }
    bool operator>(const bigint &v) const {
        return v < *this;
    }
    bool operator<=(const bigint &v) const {
        return !(v < *this);
    }
    bool operator>=(const bigint &v) const {
        return !(*this < v);
    }
    bool operator==(const bigint &v) const {
        return !(*this < v) && !(v < *this);
    }
    bool operator!=(const bigint &v) const {
        return *this < v || v < *this;
    }

    bigint abs() const {
        bigint res = *this;
        res.sign *= res.sign;
        return res;
    }
    void check(int &v) { // 检查输入的是否为负数
        if (v < 0) {
            sign = -sign;
            v = -v;
        }
    }
    void trim() { // 去除前导零
        while (!a.empty() && !a.back()) a.pop_back();
        if (a.empty()) sign = 1;
    }
    bool isZero() const { // 判断是否等于零
        return a.empty() || (a.size() == 1 && !a[0]);
    }
    friend bigint gcd(const bigint &a, const bigint &b) {
        return b.isZero() ? a.abs() : gcd(b, a % b);
    }
    friend bigint lcm(const bigint &a, const bigint &b) {
        if (a.isZero() || b.isZero()) return 0;
        return (a / gcd(a, b) * b).abs();
    }
    void read(const string &s) {
        sign = 1;
        a.clear();
        int pos = 0;
        while (pos < s.size() && (s[pos] == '-' || s[pos] == '+')) {
            if (s[pos] == '-') sign = -sign;
            ++pos;
        }
        for (int n = s.size(), i = n - 1; i >= pos; i -= base_digits) {
            int x = 0;
            for (int j = max(pos, i - base_digits + 1); j <= i; ++j) x = x * 10 + s[j] - '0';
            a.push_back(x);
        }
        trim();
    }
    friend istream &operator>>(istream &stream, bigint &v) {
        string s;
        stream >> s;
        v.read(s);
        return stream;
    }
    friend ostream &operator<<(ostream &stream, const bigint &v) {
        if (v.sign == -1) stream << '-';
        stream << (v.a.empty() ? 0 : v.a.back());
        for (int n = v.a.size(), i = n - 2; i >= 0; --i)
            stream << setw(base_digits) << setfill('0') << v.a[i];
        return stream;
    }

    /* 大整数乘除大整数部分 */
    typedef vector<long long> vll;
    bigint operator*(const bigint &v) const { // 大整数乘大整数
        vector<int> a6 = convert_base(this->a, base_digits, 6);
        vector<int> b6 = convert_base(v.a, base_digits, 6);
        vll a(a6.begin(), a6.end());
        vll b(b6.begin(), b6.end());
        while (a.size() < b.size()) a.push_back(0);
        while (b.size() < a.size()) b.push_back(0);
        while (a.size() & (a.size() - 1)) a.push_back(0), b.push_back(0);
        vll c = karatsubaMultiply(a, b);
        bigint res;
        res.sign = sign * v.sign;
        for (int i = 0, carry = 0; i < c.size(); ++i) {
            long long cur = c[i] + carry;
            res.a.push_back((int)(cur % 1000000));
            carry = (int)(cur / 1000000);
        }
        res.a = convert_base(res.a, 6, base_digits);
        res.trim();
        return res;
    }
    friend pair<bigint, bigint> divmod(const bigint &a1,
                                       const bigint &b1) { // 大整数除大整数，同时返回答案与余数
        assert(!b1.isZero());
        int norm = base / (b1.a.back() + 1);
        bigint a = a1.abs() * norm;
        bigint b = b1.abs() * norm;
        bigint q, r;
        q.a.resize(a.a.size());
        for (int n = a.a.size(), i = n - 1; i >= 0; i--) {
            r *= base;
            r += a.a[i];
            int s1 = r.a.size() <= b.a.size() ? 0 : r.a[b.a.size()];
            int s2 = r.a.size() <= b.a.size() - 1 ? 0 : r.a[b.a.size() - 1];
            int d = ((long long)base * s1 + s2) / b.a.back();
            r -= b * d;
            while (r < 0) r += b, --d;
            q.a[i] = d;
        }
        q.sign = a1.sign * b1.sign;
        r.sign = a1.sign;
        q.trim();
        r.trim();
        return make_pair(q, r / norm);
    }
    static vector<int> convert_base(const vector<int> &a, int old_digits, int new_digits) {
        vector<long long> p(max(old_digits, new_digits) + 1);
        p[0] = 1;
        for (int i = 1; i < p.size(); ++i) p[i] = p[i - 1] * 10;
        vector<int> res;
        long long cur = 0;
        int cur_digits = 0;
        for (int i = 0; i < a.size(); ++i) {
            cur += a[i] * p[cur_digits];
            cur_digits += old_digits;
            while (cur_digits >= new_digits) {
                res.push_back((int)(cur % p[new_digits]));
                cur /= p[new_digits];
                cur_digits -= new_digits;
            }
        }
        res.push_back((int)cur);
        while (!res.empty() && !res.back()) res.pop_back();
        return res;
    }
    static vll karatsubaMultiply(const vll &a, const vll &b) {
        int n = a.size();
        vll res(n + n);
        if (n <= 32) {
            for (int i = 0; i < n; ++i) {
                for (int j = 0; j < n; ++j) {
                    res[i + j] += a[i] * b[j];
                }
            }
            return res;
        }

        int k = n >> 1;
        vll a1(a.begin(), a.begin() + k);
        vll a2(a.begin() + k, a.end());
        vll b1(b.begin(), b.begin() + k);
        vll b2(b.begin() + k, b.end());

        vll a1b1 = karatsubaMultiply(a1, b1);
        vll a2b2 = karatsubaMultiply(a2, b2);

        for (int i = 0; i < k; ++i) a2[i] += a1[i];
        for (int i = 0; i < k; ++i) b2[i] += b1[i];

        vll r = karatsubaMultiply(a2, b2);
        for (int i = 0; i < a1b1.size(); ++i) r[i] -= a1b1[i];
        for (int i = 0; i < a2b2.size(); ++i) r[i] -= a2b2[i];

        for (int i = 0; i < r.size(); ++i) res[i + k] += r[i];
        for (int i = 0; i < a1b1.size(); ++i) res[i] += a1b1[i];
        for (int i = 0; i < a2b2.size(); ++i) res[i + n] += a2b2[i];
        return res;
    }

    void operator*=(const bigint &v) {
        *this = *this * v;
    }
    bigint operator/(const bigint &v) const {
        return divmod(*this, v).first;
    }
    void operator/=(const bigint &v) {
        *this = *this / v;
    }
    bigint operator%(const bigint &v) const {
        return divmod(*this, v).second;
    }
    void operator%=(const bigint &v) {
        *this = *this % v;
    }
};
```

#### 朴素高精乘法备选

沿用上方 `bigint`，不改默认 Karatsuba 乘法；两数较短或长度悬殊时可直接调用 `mulSimple`。设块数为 n、m，时间 $O(nm)$、额外空间 $O(n+m)$。

```c++
bigint mulSimple(const bigint &x, const bigint &y) {
    if (x.isZero() || y.isZero()) return 0;
    bigint r;
    r.sign = x.sign * y.sign;
    r.a.assign(x.a.size() + y.a.size(), 0);
    for (int i = 0; i < x.a.size(); ++i) {
        int c = 0;
        for (int j = 0; j < y.a.size() || c; ++j) {
            int v = r.a[i + j] + c;
            if (j < y.a.size()) v += x.a[i] * y.a[j];
            r.a[i + j] = v % base, c = v / base;
        }
    }
    r.trim();
    return r;
}
```

### 分数运算类

分母恒正、每次运算后既约，乘除先交叉约分；保留 `x/y`、`val()`、`norm()` 接口。整数中间结果仍须不溢出；可用本章 `bigint` 作底层类型，此时不调用转浮点的 `val()`。

```c++
template<class T = int> struct Frac {
    T x, y;
    Frac(T x = 0, T y = 1) : x(x), y(y) { reduce(); }
    static T ab(T v) { return v < 0 ? -v : v; }
    void reduce() {
        assert(y != T(0));
        if (y < 0) x = -x, y = -y;
        if (x == T(0)) { y = 1; return; }
        T g = gcd(ab(x), y);
        x = x / g, y = y / g;
    }
    Frac norm() const { return Frac(x, y); }
    double val() const { return 1. * x / y; }
    Frac &operator+=(const Frac &v) {
        T g = gcd(y, v.y);
        x = x * (v.y / g) + v.x * (y / g), y = y / g * v.y;
        reduce();
        return *this;
    }
    Frac &operator-=(const Frac &v) { return *this += -v; }
    Frac &operator*=(const Frac &v) {
        T g1 = gcd(ab(x), v.y), g2 = gcd(ab(v.x), y);
        x = x / g1 * (v.x / g2), y = y / g2 * (v.y / g1);
        return *this;
    }
    Frac &operator/=(const Frac &v) {
        assert(v.x != T(0));
        T g1 = gcd(ab(x), ab(v.x)), g2 = gcd(y, v.y), vx = v.x / g1;
        x = x / g1 * (v.y / g2), y = y / g2 * vx;
        if (y < 0) x = -x, y = -y;
        return *this;
    }
    friend Frac operator+(Frac a, const Frac &b) { return a += b; }
    friend Frac operator-(Frac a, const Frac &b) { return a -= b; }
    friend Frac operator*(Frac a, const Frac &b) { return a *= b; }
    friend Frac operator/(Frac a, const Frac &b) { return a /= b; }
    friend Frac operator-(Frac a) { return Frac(-a.x, a.y); }
    friend bool operator<(const Frac &a, const Frac &b) {
        using W = conditional_t<is_integral_v<T>, i128, T>;
        return W(a.x) * W(b.y) < W(b.x) * W(a.y);
    }
    friend bool operator>(const Frac &a, const Frac &b) { return b < a; }
    friend bool operator<=(const Frac &a, const Frac &b) { return !(b < a); }
    friend bool operator>=(const Frac &a, const Frac &b) { return !(a < b); }
    friend bool operator==(const Frac &a, const Frac &b) { return a.x == b.x && a.y == b.y; }
    friend bool operator!=(const Frac &a, const Frac &b) { return !(a == b); }
    friend ostream &operator<<(ostream &out, const Frac &a) {
        return a.y == T(1) ? out << a.x : out << a.x << '/' << a.y;
    }
};
```

### 快速幂

#### 常规

时间复杂度 $\mathcal O(\log N)$ 。

```c++
int power(int a, int b, int m) {
    int r = 1 % m;
    a %= m;
    for (; b; b >>= 1, a = a * a % m) 
        if (b & 1) r = r * a % m;
    return r;
}
```

#### 长整型 with 防爆乘法

使用 `i128` 精确计算，支持 $1\le m\le 2^{63}-1$；底数允许为负，指数非负。浮点估商法依赖浮点精度且容易触发有符号溢出，此处统一用整数版本。

```c++
int mul(int a, int b, int m) {
    int r = i128(a) * b % m;
    return r < 0 ? r + m : r;
}

int power(int a, int b, int m) {
    int r = 1 % m;
    a %= m;
    for (; b; b >>= 1, a = mul(a, a, m))
        if (b & 1) r = mul(r, a, m);
    return r;
}
```

### 质数判定

#### 试除法

标准 $\mathcal O(\sqrt N)$ ，有==常数优化版本==可达到 $\mathcal O(\frac {\sqrt N}{3})$ 。

```c++
bool isprime(int n) {
    if (n < 2) return false;
    for (int i = 2; i <= n / i; i++)
        if (n % i == 0) return false;
    return true;
}
```

#### 筛法1

使用欧拉筛（线性筛），预期时间复杂度为 $\mathcal O(N)$ 。

```c++
constexpr int N = 1E7;

vector<int> primes, spf(N + 1);

void init() {
    primes.clear();
    fill(spf.begin(), spf.end(), 0);
    for (int i = 2; i <= N; i++) {
        if (!spf[i]) spf[i] = i, primes.push_back(i);
        for (auto p : primes) {
            if (p > N / i) break;
            int y = p * i;
            spf[y] = p;
            if (p == spf[i]) break;
        }
    }
}

bool isprime(int n) {
    return n >= 2 && spf[n] == n;
}
```

#### 筛法2

由于 `bitset` 快速的连续读写效率，使得它非常适合用于与埃氏筛结合打质数表，性能略优于常规线性筛。

```cpp
constexpr int N = 1E7;

bitset<N + 1> isp;

void init() {
    isp.set(); // 将所有位置设为 1
    isp[0] = isp[1] = 0;
    for (int i = 2; i <= N / i; i++) {
        if (isp[i]) {
            for (int j = i * i; j <= N; j += i) isp[j] = 0;
        }
    }
}

bool isprime(int n) {
    return isp[n];
}
```

#### Miller-Rabin

使用 7 个固定底数，对有符号 64 位非负整数作确定性判定，时间 $O(\log n)$。模乘用 `i128` 保证精确；第十三章另有 Montgomery 版本。

```c++
int mul(int a, int b, int m) {
    return i128(a) * b % m;
}

bool isprime(int n) {
    if (n < 2 || n % 6 % 4 != 1) return (n | 1) == 3;
    int s = __builtin_ctzll(n - 1), d = n >> s;
    for (int a : {2, 325, 9375, 28178, 450775, 9780504, 1795265022}) {
        if (!(a %= n)) continue;
        int x = 1;
        for (int b = d; b; b >>= 1, a = mul(a, a, n))
            if (b & 1) x = mul(x, a, n);
        if (x == 1 || x == n - 1) continue;
        for (int i = 1; i < s && x != n - 1; ++i) x = mul(x, x, n);
        if (x != n - 1) return false;
    }
    return true;
}
```

### 质因子分解

#### 筛法

`spf`、`maxp` 分别维护最小、最大质因子；`fact` 返回质因子与指数。

使用欧拉筛（线性筛），预处理时间复杂度 $\mathcal O(N)$ ，单次查询 $\mathcal O(\textrm{Prime Numer})$ 。

```c++
constexpr int N = 1E7;

vector<int> primes, spf(N + 1), maxp(N + 1);

void init() {
    primes.clear();
    for (int i = 2; i <= N; i++) {
        if (!spf[i]) spf[i] = maxp[i] = i, primes.push_back(i);
        for (auto p : primes) {
            if (p > spf[i] || p > N / i) break;
            spf[i * p] = p;
            maxp[i * p] = maxp[i];
        }
    }
}

vector<pii> fact(int n) {
    vector<pii> r;
    while (n > 1) {
        int p = spf[n], e = 0;
        while (n % p == 0) n /= p, e++;
        r.push_back({p, e});
    }
    return r;
}
```

#### Pollard-Rho

沿用上方 Miller-Rabin 的 `mul`、`isprime` 与随机数引擎 `rng`，`fact(x)` 要求 $x\ge1$。若最小质因子为 p，寻找一个非平凡因子的期望迭代数为 $O(\sqrt p)$，最坏规模约为 $O(x^{1/4})$，另计素性判定开销。第十三章保留 Montgomery 优化版。

```c++
int rho(int n) {
    if (n % 2 == 0) return 2;
    int x = 0, y = 0, c = 0, p = 1, d;
    auto f = [&](int x) {
        int v = mul(x, x, n) - c;
        return v < 0 ? v + n : v;
    };
    for (int l = 1;; l *= 2) {
        if (x == y) c = rng() % (n - 1) + 1, y = 0, l = 1;
        x = y;
        for (int i = 0; i < l; ++i) {
            y = f(y);
            if (x == y) break;
            if (int q = mul(p, abs(x - y), n)) p = q;
            if (i % 128 == 0 && (d = gcd(p, n)) > 1) return d;
        }
    }
}

vector<int> fact(int x) {
    vector<int> r;
    auto f = [&](auto f, int x) {
        if (x == 1) return;
        if (isprime(x)) return r.push_back(x);
        int y = rho(x);
        f(f, y), f(f, x / y);
    };
    f(f, x), sort(r.begin(), r.end());
    return r;
}
```

> 求解二元一次不定方程 $A\cdot x + B\cdot y = C$ 。

若 $d\mid c$，把特解乘 $c/d$ 得 $(x_0,y_0)$，通解为
$$
x=x_0+t\frac bd,\qquad y=y_0-t\frac ad,\qquad t\in\mathbb Z.
$$
系数为负时，先对绝对值求解，再改变对应解的符号。缩放特解可能需要 `i128`。

当 $a,b>0$ 时，正整数解对应
$$
\left\lceil\frac{1-x_0}{b/d}\right\rceil\le t\le
\left\lfloor\frac{y_0-1}{a/d}\right\rfloor.
$$
```cpp
auto calc = [&](int a, int b, int c) {
    int x, y, d = exgcd(a, b, x, y);
    if (c % d != 0) return {-1, -1};
    int x0 = x * c / d, y0 = y * c / d;
    return {x0, y0};
    auto norm = [%](int x, int m) {
        x %= m;
        return x < 0 ? x + m : x;
    };
    int X = norm(x0, b), Y = norm(y0, a);
    int l = (X - x0) / b, r = (y0 - Y) / a;
    // x 为最小正整数
    return {X, y - l * a};
    // y 为最小正整数
    return {x + r * b, Y};
    // x, y 均为正整数的组数
    int cnt = max(0LL, r - l + 1);
}
```

### 离散对数 bsgs 与 exbsgs

以 $\mathcal O(\sqrt {P})$ 的复杂度求解 $a^x \equiv b(\bmod P)$  。其中标准 $\tt BSGS$ 算法不能计算 $a$ 与 $MOD$ 互质的情况，而 exbsgs 则可以。

```c++
namespace BSGS {
    int power(int a, int b, int m) {
        int r = 1 % m;
        a %= m;
        for (; b; b >>= 1, a = a * a % m)
            if (b & 1) r = r * a % m;
        return r;
    }

    int exgcd(int a, int b, int &x, int &y) {
        if (!b) {
            x = 1, y = 0;
            return a;
        }
        int d = exgcd(b, a % b, y, x);
        y -= a / b * x;
        return d;
    }
    int inv(int a, int m) {
        int x, y, d = exgcd(a, m, x, y);
        return d == 1 ? (x % m + m) % m : -1;
    }

    int bsgs(int a, int b, int p) {
        a %= p, b %= p;
        if (b == 1) return 0;

        int m = sqrtl(p) + 1, e = 1;
        map<int, int> mp;
        for (int j = 0; j < m; ++j) {
            mp.emplace(e, j);
            e = e * a % p;
        }

        int step = inv(power(a, m, p), p), cur = b;
        for (int i = 0; i <= m; ++i) {
            auto it = mp.find(cur);
            if (it != mp.end()) return i * m + it->second;
            cur = cur * step % p;
        }
        return -1;
    }

    int exbsgs(int a, int b, int p) {
        if (p == 1) return 0;
        a %= p, b %= p;
        if (a < 0) a += p;
        if (b < 0) b += p;
        if (b == 1) return 0;

        int k = 0, t = 1, g;
        while ((g = gcd(a, p)) > 1) {
            if (b % g) return -1;
            b /= g, p /= g, ++k;
            if (p == 1) return k;

            t = t * (a / g) % p % p;
            if (t == b) return k;
        }

        int it = inv(t, p);
        int x = bsgs(a, b * it % p, p);
        return x < 0 ? -1 : x + k;
    }
} // namespace BSGS

using namespace BSGS;
```

### 欧拉函数

#### 直接求解单个数的欧拉函数

$1$ 到 $N$ 中与 $N$ 互质数的个数称为欧拉函数，记作 $\varphi (N)$ 。求解欧拉函数的过程即为分解质因数的过程，复杂度 $\mathcal{O}(\sqrt{n})$ 。

```c++
int phi(int n) {
    int ans = n;
    for(int i = 2; i <= n / i; i ++) {
        if(n % i == 0) {
            ans = ans / i * (i - 1);
            while(n % i == 0) n /= i;
        }
    }
    if(n > 1) ans = ans / n * (n - 1);
    return ans;
} 
```

#### 求解 1 到 N 所有数的欧拉函数

利用上述第四条性质，我们可以快速递推出 $2-N$ 中每个数的欧拉函数，复杂度 $\mathcal{O}(N)$ ，而该算法**即是线性筛的算法**。

```c++
constexpr int N = 1E5;

vector<int> primes, spf(N + 1), phi(N + 1);

void init() {
    for (int i = 2; i <= N; ++i) {
        if (!spf[i]) spf[i] = i, primes.push_back(i), phi[i] = i - 1;
        for (auto p : primes) {
            if (p > N / i) break;
            int y = p * i;
            spf[y] = p;
            if (p == spf[i]) {
                phi[y] = phi[i] * p;
                break;
            }
            phi[y] = phi[i] * (p - 1);
        }
    }
}
```

#### 约数反演递推欧拉函数

由 $n=\sum_{d\mid n}\varphi(d)$ 扣除真因子的贡献，返回 $0$ 到 $n$ 的表。时间 $O(n\log n)$、空间 $O(n)$，线性筛仍是批量求值的首选。

```c++
vector<int> phiDivisors(int n) {

    iota(phi.begin(), phi.end(), 0LL);
    for (int i = 1; i <= n; i++)
        for (int j = i + i; j <= n; j += i) phi[j] -= phi[i];
```

```c++
        _ifac.resize(m + 1);
        _inv.resize(m + 1);
        for (int i = n + 1; i <= m; i++) {
            _fac[i] = _fac[i - 1] * i;

        _ifac[m] = _fac[m].inv();
        for (int i = m; i > n; i--) {
            _ifac[i - 1] = _ifac[i] * i;
            _inv[i] = _ifac[i] * _fac[i - 1];
        }
        n = m;
    }

        return _fac[m];
    }
    Z ifac(int m) {
        if (m > n) init(m);
        return _ifac[m];
    }
    Z inv(int m) {
        if (m > n) init(m);
        return _inv[m];

    }
```

### ExLucas

求解组合数模任意非质数的情况，复杂度为 $\mathcal{O}(P\log P)$ 。

```cpp
int exL(int n, int m, int mod) {
    if (m < 0 || m > n || mod == 1) return 0;
    int ans = 0, x = mod;
    auto solve = [&](int p, int pk) {
        vector<int> f(pk + 1, 1);
        for (int i = 1; i <= pk; ++i) f[i] = mul(f[i - 1], i % p ? i : 1, pk);

        auto fac = [&](auto &&self, int x) -> int {
            if (!x) return 1;
            return mul(mul(power(f[pk], x / pk, pk), f[x % pk], pk), self(self, x / p), pk);
        };

        int e = 0, phi = pk / p * (p - 1);
        for (int y = n; y; y /= p) e += y / p;
        for (int y = m; y; y /= p) e -= y / p;
        for (int y = n - m; y; y /= p) e -= y / p;

        int den = mul(fac(fac, m), fac(fac, n - m), pk);
        int r = mul(mul(fac(fac, n), power(den, phi - 1, pk), pk), power(p, e, pk), pk);
        int M = mod / pk;
        int add = mul(mul(r, M, mod), power(M, phi - 1, pk), mod);
        ans = (ans + add) % mod;
    };
    for (int p = 2; p <= x / p; ++p) {
        if (x % p) continue;
        int pk = 1;
        while (x % p == 0) x /= p, pk *= p;
        solve(p, pk);
    }
    if (x > 1) solve(x, x);
    return ans;
}
```

#### 质因数指数法（任意正模数）

先筛出 $1$ 到上限 $n$ 的质数，再用阶乘的质因子指数计算组合数，无需求逆。适用于 $n$ 可预处理、模数可变或为合数的情况；预处理 $O(n)$，单次至多 $O(\pi(n)\log n)$。

```c++
struct FactorComb {
    int n;
    vector<int> primes, spf;
    FactorComb(int n) : n(n), spf(n + 1) {
        for (int i = 2; i <= n; i++) {
            if (!spf[i]) spf[i] = i, primes.push_back(i);
            for (auto p : primes) {
                if (p > spf[i] || p > n / i) break;
                spf[i * p] = p;
            }
        }
    }
    int C(int x, int y, int mod) const {
        assert(mod > 0 && x <= n);
        if (x < 0 || y < 0 || y > x || mod == 1) return 0;
        auto count = [](int x, int p) {
            int r = 0;
            for (; x; x /= p) r += x / p;
            return r;
        };
        auto power = [&](int a, int b) {
            int r = 1 % mod;
            a %= mod;
            for (; b; b >>= 1, a = i128(a) * a % mod)
                if (b & 1) r = i128(r) * a % mod;
            return r;
        };
        int r = 1 % mod;
        for (auto p : primes) {
            if (p > x) break;
            int e = count(x, p) - count(y, p) - count(x - y, p);
            r = i128(r) * power(p, e) % mod;
        }
        return r;
    }
};
```

### 动态模数取模类（Barrett 约减）

运行期确定模数时选用。先调用 `MInt<id>::setMod(m)`，要求 $1<m<2^{31}$；同一 `id` 共享模数，换模后旧对象不可继续使用。加减乘可用合数模，除法／逆元要求质数模且除数非零；不同模数同时存在时用不同 `id`。

```c++
struct Barrett {
    uint32_t m;
    uint64_t im;
    Barrett(uint32_t m = 1) : m(m), im(uint64_t(-1) / m + 1) {}
    uint32_t mul(uint32_t a, uint32_t b) const {
        uint64_t z = uint64_t(a) * b;
        uint64_t q = __uint128_t(z) * im >> 64;
        uint32_t y = z - q * m;
        return y < m ? y : uint32_t(y + m);
    }
};

template <int id = 0> struct MInt {
    inline static Barrett B;
    uint32_t x;
    MInt(i64 v = 0) : x(norm(v)) {}
    static void setMod(int m) {
        assert(1 < m && m < (1LL << 31));
        B = Barrett(m);
    }
    static int getMod() {
        return B.m;
    }
    static uint32_t norm(int v) {
        v %= getMod();
        return v < 0 ? v + getMod() : v;
    }
    int val() const {
        return x;
    }
    MInt pow(int k) const {
        assert(k >= 0);
        MInt a = *this, r = 1;
        for (; k; k >>= 1, a *= a)
            if (k & 1) r *= a;
        return r;
    }
    MInt inv() const {
        assert(x);
        return pow(getMod() - 2);
    }
    MInt operator-() const { return x ? MInt(getMod() - x) : MInt(0); }
    MInt &operator+=(MInt v) {
        x += v.x;
        if (x >= B.m) x -= B.m;
        return *this;
    }
    MInt &operator-=(MInt v) {
        x = x >= v.x ? x - v.x : x + B.m - v.x;
        return *this;
    }
    MInt &operator*=(MInt v) {
        x = B.mul(x, v.x);
        return *this;
    }
    MInt &operator/=(MInt v) { return *this *= v.inv(); }
    friend MInt operator+(MInt a, MInt b) { return a += b; }
    friend MInt operator-(MInt a, MInt b) { return a -= b; }
    friend MInt operator*(MInt a, MInt b) { return a *= b; }
    friend MInt operator/(MInt a, MInt b) { return a /= b; }
    friend bool operator==(MInt a, MInt b) { return a.x == b.x; }
    friend bool operator!=(MInt a, MInt b) { return a.x != b.x; }
    friend istream &operator>>(istream &in, MInt &a) {
        i64 v;
        in >> v;
        a = v;
        return in;
    }
    friend ostream &operator<<(ostream &out, MInt a) { return out << a.x; }
};
```

### 容斥原理

> 定义：$\big|S_1 \cup S_2 \cup S_3 \cup … \cup S_n \big | =\sum_{i=1}^N|S_i|   -   \sum_{i,j=1}^N \big| S_i \cap S_j \big|   +   \sum_{i,j,k=1}^N \big| S_i \cap S_j \cap S_k \big| -…$ 

例题：给定一个整数 $n$ 和 $m$ 个不同的质数 $p_1, p_2, ..., p_m$，请你求出 $1 \sim n$ 中能被 $p_1, p_2, ..., p_m$ 中的至少一个数整除的整数有多少个。

#### 二进制枚举解

```cpp

}
```

### 线性基（高斯消元法）

设向量长度为 $N$（一般取 $63$），总数为 $M$，时间复杂度为 $\mathcal O(NM)$ 。

```c++
struct LB { // Linear Basis
    using int = long long;
    const int BASE = 63;
    vector<int> d, p;
    int cnt, flag;

    LB() {
        d.resize(BASE + 1);
        p.resize(BASE + 1);
        cnt = flag = 0;
    }
    bool insert(int val) {
        for (int i = BASE - 1; i >= 0; i--) {
            if (val & (1ll << i)) {
                if (!d[i]) {
                    d[i] = val;
                    return true;
                }
                val ^= d[i];
            }
        }
        flag = 1; //可以异或出0
        return false;
    }
    bool check(int val) { // 判断 val 是否能被异或得到
        for (int i = BASE - 1; i >= 0; i--) {
            if (val & (1ll << i)) {
                if (!d[i]) {
                    return false;
                }
                val ^= d[i];
            }
        }
        return true;
    }
    int ask_max() {
        int res = 0;
        for (int i = BASE - 1; i >= 0; i--) {
            if ((res ^ d[i]) > res) res ^= d[i];
        }
        return res;
    }
    int ask_min() {
        if (flag) return 0; // 特判 0
        for (int i = 0; i <= BASE - 1; ++i) {
            if (d[i]) return d[i];
        }
    }
    void rebuild() { // 第k小值独立预处理
        for (int i = BASE - 1; i >= 0; i--) {
            for (int j = i - 1; j >= 0; j--) {
                if (d[i] & (1ll << j)) d[i] ^= d[j];
            }
        }
        for (int i = 0; i <= BASE - 1; ++i) {
            if (d[i]) p[cnt++] = d[i];
        }
    }
    int kthquery(int k) { // 查询能被异或得到的第 k 小值, 如不存在则返回 -1
        if (flag) k--; // 特判 0, 如果不需要 0, 直接删去
        if (!k) return 0;
        int res = 0;
        if (k >= (1ll << cnt)) return -1;
        for (int i = BASE - 1; i >= 0; i--) {
            if (k & (1LL << i)) res ^= p[i];
        }
        return res;
    }
    void Merge(const LB &b) { // 合并两个线性基
        for (int i = BASE - 1; i >= 0; i--) {
            if (b.d[i]) {
                insert(b.d[i]);
            }
        }
    }
};
```

#### 正向展开树状数组解

给定一个全排列，求出它是 1 ~ $n$ 所有全排列的第几个，答案对 $998244353$ 取模。

答案就是 $\sum_{i = 1}^{n} res_{a_i} (n - i)!$ 。$res_x$ 表示剩下的比 $x$ 小的数字的数量，通过**树状数组**处理。

```c++

    LL n;
    vector <LL> a;
    fwt(LL n) : n(n), a(n + 1) {}
    LL sum(LL x){
        LL res = 0;
        for (; x; x -= x & -x)
            res += a[x];
        return res;
    }
    void add(LL x, LL k){
        for (; x <= n; x += x & -x)
            a[x] += k;
    }
    LL query(LL x, LL y){
        return sum(y) - sum(x - 1);
    }
};
```

### 莫比乌斯函数/反演

莫比乌斯函数定义：$\displaystyle {\mu(n) = \begin{cases} 1 &n = 1 \\ (-1)^k &n = \prod_{i = 1}^k p_i \text{ 且 } p_i \text{ 互质 } \\ 0 &else \end{cases}}$ 。

> 莫比乌斯函数性质：对于任意正整数 $n$ 满足 $\displaystyle {\sum_{d|n}\mu(d) = \begin{cases} 1 & n = 1 \\ 0 & n \neq 1\end{cases}}$ ；$\displaystyle {\sum_{d|n} \frac{\mu(d)}{d} = \frac{\varphi(n)}{n}}$ 。 

莫比乌斯反演定义：定义：$F(n)$ 和 $f(n)$ 是定义在非负整数集合上的两个函数，并且满足 $\displaystyle F(n) = \sum_{d|n}f(d)$ ，可得 $\displaystyle f(n) = \sum_{d|n}\mu(d)F(\left \lfloor \frac{n}{d} \right \rfloor)$ 。

```c++
const int N = 5e4 + 10;
bool st[N];

void getMu() {
    mu[1] = 1;
    for (int i = 2; i <= N - 10; i++) {
        if (!st[i]) {
            prime[++cnt] = i;
            mu[i] = -1;
        }
        for (int j = 1; j <= cnt && i * prime[j] <= N - 10; j++) {
            st[i * prime[j]] = true;
            if (i % prime[j] == 0) {
                mu[i * prime[j]] = 0;
                break;
            }
            mu[i * prime[j]] = -mu[i];
        }
    }
    for (int i = 1; i <= N - 10; i++) {
        sum[i] = sum[i - 1] + mu[i];
    }
}

    }
```

### 整除（数论）分块

$\displaystyle \left\lfloor \frac{n}{l} \right\rfloor = \left\lfloor \frac{n}{l + 1} \right\rfloor = ... = \left\lfloor \frac{n}{r} \right\rfloor \iff \left\lfloor \frac{n}{l} \right\rfloor \le \frac{n}{r} < \left\lfloor \frac{n}{l} \right\rfloor + 1$ ，根据不等式左侧，得到 $\displaystyle r \le \left\lfloor \frac{n}{\lfloor \frac{n}{l} \rfloor} \right\rfloor$ 。

```c++
void solve() {
    int n;
    cin >> n;
    int ans = 0;
    for (int i = 1, j; i <= n; i = j + 1) {
        j = n / (n / i);
        ans += (j - i + 1) * (n / i);
    }
    cout << ans << endl;
}
```
