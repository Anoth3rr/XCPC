## 02 - 常用函数

```cpp
// 根号，根据范围自行切换
template<T> int mysqrt(int n) {
    assert(n >= 0);
    int ans = sqrtl(n);
    while (T(ans + 1) * (ans + 1) <= n) ans++;
    while (T(ans) * ans > n) ans--;
    return ans;
}
// 最小公倍数
int mylcm(int x, int y) {
    return x / gcd(x, y) * y;
}
// log2向下取整
template <class T> int log2f(T n) {
    int ans = 0;
    while (n > 1) n >>= 1, ++ans;
    return ans;
}
int log2f(int n) {
    return 63 - __builtin_clzll(n);
}
// log2向上取整
template <class T> int log2c(T n) {
    return log2f(n) + ((n & (n - 1)) != 0);
}
int log2c(int n) {
    return log2f(n) + ((n & (n - 1)) != 0);
}
template <class T> T sign(const T &a) {
    return a == 0 ? 0 : (a < 0 ? -1 : 1);
}
template <class T> T floor(const T &a, const T &b) {
    T q = a / b, r = a % b;
    if (r != 0 && ((r > 0) != (b > 0))) --q;
    return q;
}
template <class T> T ceil(const T &a, const T &b) {
    T q = a / b, r = a % b;
    if (r != 0 && ((r > 0) == (b > 0))) ++q;
    return q;
}
// a <= x && x <= b
template <typename T> bool In(T x, T a, T b) {
    return std::min(a, b) <= x && x <= std::max(a, b);
}
// 矩阵边界（0-indexed）
inline bool inGrid(int x, int y, int n, int m) {
    return 0 <= x && x < n && 0 <= y && y < m;
}
inline bool inSq(int x, int y, int n) {
    return inGrid(x, y, n, n);
}
// 把若干位按 base 进制编码
template <class T, class... A> T encode(T bas, A... a) {
    T res = 0;
    ((res = res * bas + static_cast<T>(a)), ...);
    return res;
}
// ASCII 转大写/小写
inline char tomaj(char c) {
    if ('a' <= c && c <= 'z') return c - 32;
    return c;
}
inline char tolow(char c) {
    if ('A' <= c && c <= 'Z') return c + 32;
    return c;
}
// 若干字符等于c
template <typename... A> bool sameto(char c, A... a) {
    return ((a == c) && ...);
}
// 若干字符存在等于c
template <typename... A> bool exi(char c, A... a) {
    return ((a == c) || ...);
}
```

## 03 - 二分/三分查找 

### 03A - 整数域二分

```cpp
// x 或 x 的后继
template <class I, class F> optional<I> find(I l, I r, F f) {
    optional<I> a;
    while (l <= r) {
        I m = l + (r - l) / 2;
        if (f(m)) {
            a = m;
            r = m - 1;
        } else {
            l = m + 1;
        }
    }
    return a;
}
// x 或 x 的前驱
template <class I, class F> optional<I> find(I l, I r, F f) {
    optional<I> a;
    while (l <= r) {
        I m = l + (r - l) / 2;
        if (f(m)) {
            a = m;
            l = m + 1;
        } else {
            r = m - 1;
        }
    }
    return a;
}
```

### 03B - 实数域二分

限制二分次数。

```cpp
template <class I, class F> I find(I l, I r, F f, int cnt = 100) {
    for (int i = 0; i < cnt; ++i) {
        I m = (l + r) / 2;
        if (f(m)) r = m;
        else l = m;
    }
    return l;
}
```

### 03C - 整数域三分

```cpp
template <class I, class F> I find(I l, I r, F f) {
    while (l < r) {
        I m = l + (r - l) / 2;
        if (f(m) <= f(m + 1)) r = m;
        else l = m + 1;
    }
    return l;
}
```

### 03D - 实数域三分

限制次数实现。

```cpp
template <class I, class F> I find(I l, I r, F f, int cnt = 100) {
    for (int i = 0; i < cnt; ++i) {
        I m1 = (l * 2 + r) / 3;
        I m2 = (l + r * 2) / 3;
        if (f(m1) <= f(m2)) r = m2;
        else l = m1;
    }
    return l;
}
```
