```cpp
// 根号	
int mysqrt(int n) {
    int ans = sqrt(n);
    while ((ans + 1) * (ans + 1) <= n) ans++;
    while (ans * ans > n) ans--;
    return ans;
}
// 最小公倍数
int mylcm(int x, int y) {
    return x / gcd(x, y) * y;
}
// log2向下取整
template <class T> int log2f(T n) {
    int ans = 0;
    while (n > 1) n >>= 1, ++ans;
    return ans;
}
int log2f(int n) {
    return 63 - __builtin_clzll(n);
}
// log2向上取整
template <class T> int log2c(T n) {
    return log2f(n) + ((n & (n - 1)) != 0);
}
int log2c(int n) {
    return log2f(n) + ((n & (n - 1)) != 0);
}
template <class T> T sign(const T &a) {
    return a == 0 ? 0 : (a < 0 ? -1 : 1);
}
template <class T> T floor(const T &a, const T &b) {
    T q = a / b, r = a % b;
    if (r != 0 && ((r > 0) != (b > 0))) --q;
    return q;
}
template <class T> T ceil(const T &a, const T &b) {
    T q = a / b, r = a % b;
    if (r != 0 && ((r > 0) == (b > 0))) ++q;
    return q;
}
// a <= x && x <= b
template <typename T> bool In(T x, T a, T b) {
    return min(a, b) <= x && x <= max(a, b);
}
// 矩阵边界
inline bool check(int x, int y, int n, int m) {
    return 0 <= x && x < n && 0 <= y && y < m;
}
// 把若干位按 base 进制编码
template <class T, class... A> T encode(T bas, A... a) {
    T res = 0;
    ((res = res * bas + a), ...);
    return res;
}
// ASCII 转大写/小写
inline char tomaj(char c) {
    if ('a' <= c && c <= 'z') return c - 32;
    return c;
}
inline char tolow(char c) {
    if ('A' <= c && c <= 'Z') return c + 32;
    return c;
}
// 若干字符等于c
template <typename... A> bool sameto(char c, A... a) {
    return ((a == c) && ...);
}
// 若干字符存在等于c
template <typename... A> bool exi(char c, A... a) {
    return ((a == c) || ...);
}
```

### 整数域二分

- $x$ 或 $x$ 的后继

```cpp
int ans = r;
while (l <= r) {
    int m = (l + r) >> 1;
    if (check(m)) ans = m, r = m - 1;
    else l = m + 1;
}
```

### 实数域三分

限制次数实现。

```cpp
ld l = -1E9, r = 1E9;
for (int t = 1; t <= 100; t++) {
    ld mid1 = (l * 2 + r) / 3;
    ld mid2 = (l + r * 2) / 3;
    if (judge(mid1) < judge(mid2)) {
        r = mid2;
    } else {
        l = mid1;
    }
}
cout << l << endl;
```
